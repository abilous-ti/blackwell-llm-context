
```python
import anthropic
import cache

def save(k: str, v) -> bool:
    """Save a value to cache under the given key with 60 second TTL.
    
    Args:
        k: Already-normalized cache key (e.g., 'kx7-hello_world')
        v: Value to cache
        
    Returns:
        True if the key was newly inserted, False if it replaced an existing value
        
    Raises:
        cache.CacheError: If key is empty
    """
    return cache.put(k, v, ttl=60)
```
