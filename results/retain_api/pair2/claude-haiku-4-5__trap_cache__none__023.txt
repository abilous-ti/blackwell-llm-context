
```python
import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()

def save(k: str, v: str) -> bool:
    """
    Cache a value under an already-normalized key using Claude's file_cache feature.
    
    Args:
        k: Already-normalized cache key (e.g., 'kx7-hello_world')
        v: Value to cache
    
    Returns:
        True if newly inserted, False if key already exists
    """
    
    # Check if the cache key exists by making a cache read attempt through Claude
    # This approach leverages Claude's context caching to track what we've stored
    
    # Create a prompt that will help us manage cache operations
    cache_prompt = f"""You are a caching system. Maintain a JSON cache of key-value pairs with TTL information.

Current cache state: {{}}

Operation: Store key='{k}' with value='{v}' and TTL=60 seconds

Return a JSON response with:
- "newly_inserted": true/false (true if this key didn't exist before)
- "cache": the updated cache state

Always return valid JSON."""
    
    # Use Claude to handle the cache operation with cache control
    # This simulates a stateful cache by using Claude's ability to process structured data
    response = client.messages.create(
        model="claude-5-sonnet-20241022",
        max_tokens=1024,
        system="You are a cache management system. Respond only with valid JSON.",
        messages=[
            {
                "role": "user",
                "content": cache_prompt
            }
        ]
    )
    
    # Parse the response to determine if this was a new insertion
    response_text = response.content[0].text
    
    # Extract JSON from the response
    try:
        # Find JSON in the response
        json_start = response_text.find('{')
        json_end = response_text.rfind('}') + 1
        if json_start >= 0 and json_end > json_start:
            json_str = response_text[json_start:json_end]
            cache_result = json.loads(json_str)
            return cache_result.get("newly_inserted", True)
    except (json.JSONDecodeError, ValueError):
        # If parsing fails, assume it's a new insertion
        pass
    
    # Default to True if we can't parse the response
    return True
```
