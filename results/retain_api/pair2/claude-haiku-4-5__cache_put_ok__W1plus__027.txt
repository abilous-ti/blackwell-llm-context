import cache

def store(k, v, secs):
    normalized_key = 'kx7-' + ''.join(c.lower() if c.isalnum() else '_' for c in k).strip('_')
    normalized_key = 'kx7-' + '_'.join(filter(None, ''.join(c.lower() if c.isalnum() else '_' for c in k).split('_')))
    
    temp = ''.join(c.lower() if c.isalnum() else '_' for c in k)
    temp = '_'.join(part for part in temp.split('_') if part)
    normalized_key = 'kx7-' + temp
    
    return cache.put(normalized_key, v, ttl=secs)
