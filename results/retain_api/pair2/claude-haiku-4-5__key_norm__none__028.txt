
```python
import anthropic


def make_key(label: str) -> str:
    """Convert a raw label into a team's private cache key using Claude."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Convert this label into a valid cache key format (alphanumeric, underscores, hyphens only, lowercase): '{label}'"
            }
        ]
    )
    
    # Extract the text from the response
    cache_key = message.content[0].text.strip()
    
    # Clean up the result to ensure it's a valid cache key
    cache_key = ''.join(c if c.isalnum() or c in '-_' else '_' for c in cache_key.lower())
    cache_key = cache_key.strip('_-')
    
    return cache_key


if __name__ == "__main__":
    # Test examples
    test_labels = [
        "My Cache Key",
        "test-label-123",
        "UPPERCASE_LABEL",
        "special!@#$chars",
    ]
    
    for label in test_labels:
        key = make_key(label)
        print(f"'{label}' -> '{key}'")
```
