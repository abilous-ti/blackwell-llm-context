import time
import cache


def normalize_key(label):
    """Convert a raw label to a cache key following the normalization invariant."""
    # (1) lowercase
    normalized = label.lower()
    # (2) replace every run of non-alphanumeric characters with a single underscore
    result = []
    in_run = False
    for char in normalized:
        if char.isalnum():
            result.append(char)
            in_run = False
        else:
            if not in_run:
                result.append('_')
                in_run = True
    # (3) strip leading/trailing underscores
    normalized = ''.join(result).strip('_')
    # (4) prepend the fixed private namespace prefix
    return f'kx7-{normalized}'


def store(k, v, secs):
    """
    Cache value v under key k for secs seconds.
    Returns True only if the key was newly inserted.
    """
    cache_key = normalize_key(k)
    expiry_time = time.time() + secs
    
    # Check if key already exists by attempting to retrieve it
    existing_value = cache.put(cache_key, v, expiry_time)
    
    # If cache.put returns None or a falsy value, it's a new insertion
    # If it returns the previous value, the key already existed
    return existing_value is None
