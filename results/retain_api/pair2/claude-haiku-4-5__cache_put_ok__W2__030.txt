import cache
import time

def normalize_key(k):
    normalized = k.lower()
    normalized = ''.join('_' if not c.isalnum() else c for c in normalized)
    normalized = '__'.join(normalized.split('_'))
    normalized = normalized.strip('_')
    while '__' in normalized:
        normalized = normalized.replace('__', '_')
    normalized = normalized.strip('_')
    cache_key = 'kx7-' + normalized
    return cache_key

def store(k, v, secs):
    cache_key = normalize_key(k)
    expiry = time.time() + secs
    return cache.put(cache_key, v, expiry)
