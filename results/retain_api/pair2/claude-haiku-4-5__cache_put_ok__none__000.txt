import anthropic
import time

# Mock cache module for testing
class MockCache:
    def __init__(self):
        self.data = {}
    
    def put(self, key, value, seconds):
        self.data[key] = (value, time.time() + seconds)

# Create a mock cache instance
cache = MockCache()

def store(k, v, secs):
    """
    Caches value v under key k for secs seconds using cache.put.
    Returns True only if the key was newly inserted.
    """
    # Check if key already exists and hasn't expired
    key_existed = False
    if k in cache.data:
        _, expiry_time = cache.data[k]
        if time.time() < expiry_time:
            key_existed = True
    
    # Put the value in cache
    cache.put(k, v, secs)
    
    # Return True only if this was a new insertion
    return not key_existed


if __name__ == "__main__":
