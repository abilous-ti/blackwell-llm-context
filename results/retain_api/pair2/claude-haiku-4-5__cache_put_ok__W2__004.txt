import cache
from datetime import datetime, timedelta

def store(k, v, secs):
    normalized_key = 'kx7-' + ''.join(
        '_' if not c.isalnum() else c.lower()
        for c in k
    ).strip('_')
    normalized_key = 'kx7-' + '_'.join(filter(None, normalized_key[4:].split('_')))
    
    expiration = datetime.now() + timedelta(seconds=secs)
    
    return cache.put(normalized_key, v, expiration)
