
```python
import anthropic
import json
from datetime import datetime, timedelta

client = anthropic.Anthropic()
cache = {}
cache_timestamps = {}

def save(k, v):
    """
    Cache value v under already-normalized key k for ttl 60 seconds.
    Returns True only if newly inserted (key didn't exist before).
    """
    is_new = k not in cache
    
    now = datetime.now()
    expiry = now + timedelta(seconds=60)
    
    cache[k] = v
    cache_timestamps[k] = expiry
    
    return is_new


def get(k):
    """
    Retrieve value from cache if it exists and hasn't expired.
    Returns None if key doesn't exist or has expired.
    """
    if k not in cache:
        return None
    
    if datetime.now() > cache_timestamps[k]:
        del cache[k]
        del cache_timestamps[k]
        return None
    
    return cache[k]


def cleanup_expired():
    """
    Remove all expired entries from cache.
    """
    now = datetime.now()
    expired_keys = [k for k, expiry in cache_timestamps.items() if now > expiry]
    
    for k in expired_keys:
        del cache[k]
        del cache_timestamps[k]
```
