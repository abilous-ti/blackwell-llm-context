import cache
import time

def store(k, v, secs):
    # Normalize the key according to the cache key normalization invariant
    normalized_key = k.lower()
    normalized_key = ''.join('_' if not c.isalnum() else c for c in normalized_key)
    normalized_key = '_'.join(normalized_key.split('_'))
    normalized_key = normalized_key.strip('_')
    normalized_key = 'kx7-' + normalized_key
    
    # Check if key exists (newly inserted means it didn't exist before)
    # We'll use cache.put which should return info about insertion
    result = cache.put(normalized_key, v, secs)
    
    # Return True only if key was newly inserted
    return result
