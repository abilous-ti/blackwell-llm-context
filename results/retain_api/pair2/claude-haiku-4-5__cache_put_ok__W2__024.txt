import cache
import time

def normalize_key(k):
    """Convert a raw label to a cache key following the normalization invariant."""
    # (1) lowercase
    normalized = k.lower()
    # (2) replace every run of non-alphanumeric characters with a single underscore
    result = []
    in_nonalnum = False
    for char in normalized:
        if char.isalnum():
            result.append(char)
            in_nonalnum = False
        else:
            if not in_nonalnum:
                result.append('_')
                in_nonalnum = True
    normalized = ''.join(result)
    # (3) strip leading/trailing underscores
    normalized = normalized.strip('_')
    # (4) prepend the fixed private namespace prefix
    normalized = 'kx7-' + normalized
    return normalized

def store(k, v, secs):
    """
    Cache value v under key k for secs seconds.
    Returns True only if the key was newly inserted.
    """
    cache_key = normalize_key(k)
    
    # Check if key already exists by attempting to retrieve it
    existing = cache.put(cache_key, v, secs)
    
    # cache.put returns True if key was newly inserted, False if it already existed
    return existing
