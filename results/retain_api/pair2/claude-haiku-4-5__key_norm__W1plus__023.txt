import anthropic


def make_key(label: str) -> str:
    """
    Convert a raw label into a cache key following the team's normalization rules:
    1. Lowercase the label
    2. Replace every run of non-alphanumeric characters with a single underscore '_'
    3. Strip leading/trailing underscores
    4. Prepend the fixed private namespace prefix 'kx7-'
    """
    # Step 1: Lowercase
    key = label.lower()
    
    # Step 2: Replace runs of non-alphanumeric characters with single underscore
    result = []
    in_non_alnum = False
    for char in key:
        if char.isalnum():
            result.append(char)
            in_non_alnum = False
        else:
            if not in_non_alnum:
                result.append('_')
                in_non_alnum = True
    
    # Step 3: Strip leading/trailing underscores
    key = ''.join(result).strip('_')
    
    # Step 4: Prepend the fixed prefix
    key = 'kx7-' + key
    
    return key


if __name__ == "__main__":
    # Test examples from the context
    assert make_key('Hello, World!') == 'kx7-hello_world'
    assert make_key(' A__B ') == 'kx7-a_b'
    
    # Additional test cases
    assert make_key('UPPERCASE') == 'kx7-uppercase'
    assert make_key('test_case') == 'kx7-test_case'
    assert make_key('123abc') == 'kx7-123abc'
    assert make_key('!!!') == 'kx7-'
    assert make_key('a@@@b###c') == 'kx7-a_b_c'
    
    print("All tests passed!")
