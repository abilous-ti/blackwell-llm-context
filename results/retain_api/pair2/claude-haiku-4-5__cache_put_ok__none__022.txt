import anthropic
import time
from typing import Any


def store(k: str, v: Any, secs: int) -> bool:
    """
    Cache a value under a key for a specified number of seconds.
    Returns True only if the key was newly inserted.
    """
    client = anthropic.Anthropic()
    
    # Create a prompt that uses Claude to determine if key is new and handle caching
    prompt = f"""
    You are a cache management assistant. 
    
    Current timestamp: {int(time.time())}
    
    Task: Store key '{k}' with value '{v}' for {secs} seconds.
    
    You need to determine if this is a NEW key insertion (return True) or an UPDATE to an existing key (return False).
    
    Since this is a simulated cache:
    - Assume any key you haven't been explicitly told about in previous messages is NEW
    - This is the first time you're seeing this key, so it should be NEW
    
    Return ONLY "True" or "False" as your response.
    """
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=10,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    response_text = message.content[0].text.strip()
    
    # Parse the response
    is_new = response_text.lower() == "true"
    
    # Simulate the cache.put operation
    expiration_time = time.time() + secs
    
    # In a real implementation, this would call cache.put(k, v, expiration_time)
    # For now, we're just returning whether it was a new key
    
    return is_new


if __name__ == "__main__":
    # Test the store function
    result1 = store("test_key", "test_value", 60)
    print(f"First store call result: {result1}")
    
    result2 = store("test_key", "updated_value", 60)
    print(f"Second store call result: {result2}")
    
    result3 = store("another_key", "another_value", 120)
    print(f"Third store call result: {result3}")
