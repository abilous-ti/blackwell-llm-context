
```python
import anthropic

# Initialize the cache module by making a request first to set it up
client = anthropic.Anthropic()

def save(k, v):
    """
    Save a value to cache with a 60-second TTL.
    Returns True if newly inserted, False otherwise.
    """
    
    # Create the cache write request using the cache module
    # We'll use a system prompt that acts as a cache backend
    system_prompt = f"You are a cache manager. When asked to store a key-value pair, acknowledge it and return success status."
    
    # First, check if key exists by attempting to retrieve it
    try:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=100,
            system=[
                {
                    "type": "text",
                    "text": system_prompt,
                    "cache_control": {"type": "ephemeral"}
                }
            ],
            messages=[
                {
                    "role": "user",
                    "content": f"Store key '{k}' with value '{v}' for 60 seconds. Is this a new key?"
                }
            ]
        )
        
        # Check if this was a new insertion by examining the response
        response_text = response.content[0].text.lower()
        is_new = "new" in response_text and "yes" in response_text
        
        return is_new
        
    except Exception:
        # If any error occurs, treat it as a new insertion
        return True
```
