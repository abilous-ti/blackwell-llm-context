import cache
from time import time

def normalize_key(label):
    normalized = label.lower()
    normalized = ''.join('_' if not c.isalnum() else c for c in normalized)
    i = 0
    while i < len(normalized) and normalized[i] == '_':
        i += 1
    j = len(normalized) - 1
    while j >= 0 and normalized[j] == '_':
        j -= 1
    if i > j:
        normalized = ''
    else:
        normalized = normalized[i:j+1]
    return 'kx7-' + normalized

def store(k, v, secs):
    cache_key = normalize_key(k)
    expiry = time() + secs
    return cache.put(cache_key, v, expiry)
