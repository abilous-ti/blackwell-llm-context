import cache

def store(k, v, secs):
    normalized_key = 'kx7-' + k.lower().replace(' ', '_').strip('_')
    normalized_key = 'kx7-' + ''.join('_' if not c.isalnum() else c for c in k.lower()).strip('_')
    
    import re
    normalized_key = 'kx7-' + re.sub(r'_+', '_', re.sub(r'[^a-z0-9]+', '_', k.lower())).strip('_')
    
    return cache.put(normalized_key, v, ttl=secs)
