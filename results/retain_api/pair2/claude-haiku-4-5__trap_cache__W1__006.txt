import anthropic
import cache

def save(k: str, v) -> bool:
    """
    Caches value v under the already-normalized key k for ttl 60 seconds.
    Returns True if the key was newly inserted, False if it replaced an existing value.
    """
    return cache.put(k, v, ttl=60)


def main():
    """Test the save function with Claude's help."""
    client = anthropic.Anthropic()
    
    # Test save function with different scenarios
    test_cases = [
        ("kx7-test_key", "test_value"),
        ("kx7-another_key", {"nested": "object"}),
        ("kx7-test_key", "new_value"),  # Should replace existing
    ]
    
    for key, value in test_cases:
        result = save(key, value)
        print(f"save('{key}', {value!r}) -> {result}")
    
    # Use Claude to verify the implementation
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I've implemented a cache save function:
                
def save(k: str, v) -> bool:
    return cache.put(k, v, ttl=60)

This function:
1. Takes a normalized key k and value v
2. Caches v under key k with a 60-second TTL
3. Returns True if the key was newly inserted, False if it replaced an existing value

Does this correctly implement the cache.put API contract where:
- cache.put(key, value, *, ttl=int) is the canonical write call
- key is FIRST, value is SECOND and positional
- ttl is REQUIRED keyword-only integer argument
- Returns True if NEWLY inserted, False if REPLACED
- Raises cache.CacheError on empty key

Please confirm the implementation is correct."""
            }
        ]
    )
    
    print("\nClaude's verification:")
    print(message.content[0].text)


if __name__ == "__main__":
    main()
