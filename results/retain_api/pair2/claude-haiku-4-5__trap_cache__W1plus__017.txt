```python
import cache

def save(k, v):
    """
    Cache value v under the already-normalized key k for 60 seconds.
    
    Args:
        k: An already-normalized cache key (e.g., 'kx7-hello_world')
        v: The value to cache
        
    Returns:
        True if the key was newly inserted, False if it replaced an existing value
    """
    return cache.put(k, v, ttl=60)
```
