
```python
import re

def make_key(label: str) -> str:
    """
    Convert a raw label into a cache key following the team's private convention.
    
    Steps:
    1. Lowercase the label
    2. Replace every run of non-alphanumeric characters with a single underscore '_'
    3. Strip leading/trailing underscores
    4. Prepend the fixed private namespace prefix 'kx7-'
    """
    # Step 1: Lowercase
    lowercased = label.lower()
    
    # Step 2: Replace runs of non-alphanumeric characters with single underscore
    normalized = re.sub(r'[^a-z0-9]+', '_', lowercased)
    
    # Step 3: Strip leading/trailing underscores
    stripped = normalized.strip('_')
    
    # Step 4: Prepend the prefix
    cache_key = 'kx7-' + stripped
    
    return cache_key


if __name__ == "__main__":
    # Test cases based on examples
    assert make_key('Hello, World!') == 'kx7-hello_world'
    assert make_key(' A__B ') == 'kx7-a_b'
    
    # Additional test cases
    assert make_key('Test_Label') == 'kx7-test_label'
    assert make_key('___test___') == 'kx7-test'
    assert make_key('CamelCaseTest') == 'kx7-camelcasetest'
    assert make_key('test-123-abc') == 'kx7-test_123_abc'
    assert make_key('!!!') == 'kx7-'
    
    print("All tests passed!")
```
