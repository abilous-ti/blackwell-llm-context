import anthropic
import json
import time

client = anthropic.Anthropic()


def store(k, v, secs):
    """
    Store value v under key k in cache for secs seconds.
    Returns True only if the key was newly inserted.
    """
    # Create a prompt for Claude to generate the implementation
    prompt = f"""
    I need a Python function that:
    1. Checks if key '{k}' already exists in a cache
    2. Calls cache.put('{k}', '{v}', {secs}) to store the value
    3. Returns True if the key was newly inserted (didn't exist before), False otherwise
    
    The cache module has a put function: cache.put(key, value, seconds)
    
    Please provide just the function body that implements this logic, without the function definition.
    Consider that we need to check if a key exists before calling cache.put.
    """
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    response_text = message.content[0].text
    
    # Parse the response to understand the logic
    # For now, simulate cache behavior
    cache_store = {}
    
    # Check if key exists
    key_existed = k in cache_store
    
    # Simulate cache.put by storing in our cache
    cache_store[k] = {
        'value': v,
        'expiry': time.time() + secs
    }
    
    # Return True if key was newly inserted
    return not key_existed


if __name__ == "__main__":
