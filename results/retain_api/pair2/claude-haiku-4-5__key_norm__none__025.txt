import anthropic
import hashlib


def make_key(label: str) -> str:
    """
    Converts a raw label into a team's private cache key.
    Uses Claude to generate a deterministic cache key from the label.
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Generate a deterministic cache key for the following label. 
The cache key should be:
- A valid string suitable for use as a cache key
- Deterministic (same input always produces same output)
- Derived from the label: {label}
- In the format: team_private_cache_{hash}

Please respond with ONLY the cache key, nothing else."""
            }
        ]
    )
    
    cache_key = message.content[0].text.strip()
    return cache_key


if __name__ == "__main__":
    test_labels = ["user_data", "session_token", "api_response"]
    
    for label in test_labels:
        key = make_key(label)
        print(f"Label: {label}")
        print(f"Cache Key: {key}")
        print()
