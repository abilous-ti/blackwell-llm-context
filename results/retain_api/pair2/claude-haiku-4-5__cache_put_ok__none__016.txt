import anthropic
import time
import json

# Simulated cache module
class CacheModule:
    def __init__(self):
        self.data = {}
        self.expiry = {}
    
    def put(self, k, v, secs):
        """Store value v under key k for secs seconds"""
        self.data[k] = v
        self.expiry[k] = time.time() + secs
    
    def get(self, k):
        """Get value from cache if not expired"""
        if k in self.data:
            if time.time() < self.expiry[k]:
                return self.data[k]
            else:
                del self.data[k]
                del self.expiry[k]
                return None
        return None
    
    def exists(self, k):
        """Check if key exists and is not expired"""
        if k in self.data:
            if time.time() < self.expiry[k]:
                return True
            else:
                del self.data[k]
                del self.expiry[k]
                return False
        return False

cache = CacheModule()

def store(k, v, secs):
    """Store value v under key k for secs seconds, return True only if key was newly inserted"""
    client = anthropic.Anthropic()
    
    # Check if key exists before storing
    key_existed = cache.exists(k)

    # Parse Claude's response
