```python
import time
from cache import write

_cache_storage = {}

def save(k, v):
    """
    Cache value v under already-normalized key k for ttl 60 seconds.
    Returns True only if newly inserted (key did not exist before).
    """
    is_new = k not in _cache_storage
    
    expiry_time = time.time() + 60
    write(k, v, expiry_time)
    _cache_storage[k] = {
        'value': v,
        'expiry': expiry_time
    }
    
    return is_new
```
