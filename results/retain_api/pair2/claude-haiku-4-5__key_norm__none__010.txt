import anthropic


def make_key(label: str) -> str:
    """Convert a raw label into the team's private cache key using Claude."""
    client = anthropic.Anthropic()
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Generate a team private cache key for the label: {label}. Return only the cache key, no other text."
            }
        ]
    )
    
    # Extract the cache key from the response
    cache_key = response.content[0].text.strip()
    return cache_key


if __name__ == "__main__":
    # Test the function
    test_label = "user_profile_data"
    cache_key = make_key(test_label)
    print(f"Label: {test_label}")
    print(f"Cache Key: {cache_key}")
