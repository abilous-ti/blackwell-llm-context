import cache
import time

def store(k, v, secs):
    normalized_key = normalize_key(k)
    current_time = time.time()
    expiry_time = current_time + secs
    
    result = cache.put(normalized_key, v, expiry_time)
    return result

def normalize_key(label):
    lowercased = label.lower()
    
    import re
    replaced = re.sub(r'[^a-z0-9]+', '_', lowercased)
    
    stripped = replaced.strip('_')
    
    prefixed = 'kx7-' + stripped
    
    return prefixed
