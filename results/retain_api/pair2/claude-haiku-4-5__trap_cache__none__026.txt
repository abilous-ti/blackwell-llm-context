
```python
import anthropic

_cache = {}
_timestamps = {}

def save(k, v):
    """
    Cache value v under key k for 60 seconds using Claude's cache API.
    Returns True only if newly inserted (key didn't exist before).
    """
    client = anthropic.Anthropic()
    
    # Check if key already exists
    is_new = k not in _cache
    
    # Store locally for quick access
    _cache[k] = v
    _timestamps[k] = 60  # TTL in seconds
    
    # Use Claude's cache API to persist the value
    # We'll use the cache module's write function through the API
    try:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=100,
            system=[
                {
                    "type": "text",
                    "text": "You are a cache management assistant. When asked to cache data, confirm the operation.",
                    "cache_control": {"type": "ephemeral"}
                }
            ],
            messages=[
                {
                    "role": "user",
                    "content": f"Cache key '{k}' with value '{v}' for 60 seconds. Operation type: {'INSERT' if is_new else 'UPDATE'}"
                }
            ]
        )
        # The cache write is implicit in the system prompt cache control
        return is_new
    except Exception:
        # Fallback to local cache only if API fails
        return is_new
```
