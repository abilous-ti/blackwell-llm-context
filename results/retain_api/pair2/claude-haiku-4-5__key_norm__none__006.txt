import anthropic


def make_key(label: str) -> str:
    """Convert a raw label into a team's private cache key using Claude."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Convert this label into a valid cache key format suitable for a team's private cache system. The key should be lowercase, use hyphens for spaces, and be URL-safe. Label: {label}"
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    # Test the function
    test_labels = [
        "User Profile Data",
        "API Response Cache",
        "Database Query Results"
    ]
    
    for label in test_labels:
        key = make_key(label)
        print(f"Label: {label}")
        print(f"Cache Key: {key}")
        print()
