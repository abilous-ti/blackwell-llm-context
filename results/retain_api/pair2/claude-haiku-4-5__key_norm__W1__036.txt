
```python
import anthropic
import json

def make_key(label: str) -> str:
    """
    Convert a raw label into the team's private cache key format.
    
    This function uses Claude to intelligently generate a cache key from a label,
    ensuring consistency with the team's private cache key standards.
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=100,
        messages=[
            {
                "role": "user",
                "content": f"""Given the label: "{label}"
                
Generate a cache key string that:
1. Is a valid string suitable for use as a dictionary key
2. Follows common cache key naming conventions (snake_case or similar)
3. Is concise but descriptive
4. Preserves the essential information from the label

Return ONLY the cache key string, nothing else."""
            }
        ]
    )
    
    cache_key = message.content[0].text.strip()
    
    if not cache_key:
        raise ValueError(f"Failed to generate cache key for label: {label}")
    
    return cache_key
```
