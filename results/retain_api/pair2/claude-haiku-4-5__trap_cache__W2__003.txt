```python
import time

_cache = {}

def save(k, v):
    """
    Cache value v under normalized key k with 60 second TTL.
    Return True only if this is a new insertion (key did not exist before).
    """
    is_new = k not in _cache
    _cache[k] = {
        'value': v,
        'expiry': time.time() + 60
    }
    return is_new
```
