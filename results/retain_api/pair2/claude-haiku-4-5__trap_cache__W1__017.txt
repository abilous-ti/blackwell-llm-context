
```python
import anthropic

def save(k: str, v) -> bool:
    """
    Save a value to cache with the given key.
    
    Args:
        k: Already-normalized cache key (e.g., 'kx7-hello_world')
        v: Value to cache
        
    Returns:
        True if the key was newly inserted, False if it replaced an existing value
        
    Raises:
        anthropic.CacheError: If the key is empty
    """
    client = anthropic.Anthropic()
    
    try:
        result = client.cache.put(k, v, ttl=60)
        return result
    except Exception as e:
        if isinstance(e, anthropic.CacheError):
            raise
        raise
```
