import anthropic
from cache import put

def store(k: str, v: str, secs: int) -> bool:
    """Store a value in cache and return True if key was newly inserted."""
    
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are helping implement a cache storage function. 
                
We need to implement a function that:
1. Checks if a key '{k}' already exists in the cache
2. Stores the value '{v}' under key '{k}' for {secs} seconds
3. Returns True if the key was newly inserted (didn't exist before), False if it already existed

The cache module provides a `put` function that stores values. 

Please respond with ONLY "True" or "False" to indicate whether key '{k}' was newly inserted (first time being stored). Do not include any other text."""
            }
        ]
    )
    
    result = put(k, v, secs)
    
    response_text = message.content[0].text.strip().lower()
    was_new = response_text == "true"
    
    return was_new
