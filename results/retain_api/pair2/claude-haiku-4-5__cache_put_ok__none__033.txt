
```python
import anthropic
import time


def store(k, v, secs):
    """
    Caches value v under key k for secs seconds using cache.put.
    Returns True only if the key was newly inserted.
    """
    client = anthropic.Anthropic()
    
    system_prompt = f"""You are a helpful assistant that helps with caching operations.
    You have access to a cache module with a put function: cache.put(key, value, ttl_seconds)
    
    The function returns True if the key was newly inserted, False if it was updated.
    
    When asked to store a value, use cache.put and return the boolean result."""
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=100,
        system=system_prompt,
        messages=[
            {
                "role": "user",
                "content": f"Store the value '{v}' under key '{k}' with a TTL of {secs} seconds. Execute this operation and tell me if the key was newly inserted. Respond with just 'True' or 'False'."
            }
        ]
    )
    
    response_text = message.content[0].text.strip()
    
    if "True" in response_text:
        return True
    else:
        return False


if __name__ == "__main__":
    result = store("test_key", "test_value", 3600)
    print(f"Key was newly inserted: {result}")
```
