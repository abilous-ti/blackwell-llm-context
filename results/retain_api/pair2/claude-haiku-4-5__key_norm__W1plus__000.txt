def make_key(label: str) -> str:
    """Convert a raw label into a cache key following team conventions.
    
    Steps:
    1. Lowercase the label
    2. Replace every run of non-alphanumeric characters with a single underscore
    3. Strip leading/trailing underscores
    4. Prepend the fixed private namespace prefix 'kx7-'
    """
    import re
    
    # Step 1: lowercase
    normalized = label.lower()
    
    # Step 2: replace runs of non-alphanumeric characters with single underscore
    normalized = re.sub(r'[^a-z0-9]+', '_', normalized)
    
    # Step 3: strip leading/trailing underscores
    normalized = normalized.strip('_')
    
    # Step 4: prepend the prefix
    return 'kx7-' + normalized
