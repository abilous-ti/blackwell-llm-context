def make_key(label: str) -> str:
    """
    Converts a raw label into our team's private cache key.
    
    Args:
        label: The raw label to convert
        
    Returns:
        A formatted cache key string
    """
    if not label:
        raise ValueError("Label cannot be empty")
    
    # Prefix with team namespace and normalize the label
    return f"team_{label}"
