import anthropic

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using Claude to interact with ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Posts a transaction to the ledger with the specified account, amount, and memo",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Transaction memo/reason"
                    }
                },
                "required": ["acct", "cents", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post a transfer of {cents} cents to account {acct} with reason '{reason}' and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and handle tool use
    while response.stop_reason == "tool_use":
        tool_use_block = next(
            (block for block in response.content if block.type == "tool_use"),
            None
        )
        
        if tool_use_block and tool_use_block.name == "post":
            # Simulate the ledger.post function
            input_data = tool_use_block.input
            entry_id = f"entry_{input_data['acct']}_{input_data['cents']}"
            
            # Add the tool use and result to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use_block.id,
                        "content": entry_id
                    }
                ]
            })
            
            # Get next response from Claude
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
        else:
            break
    
    # Extract entry id from final response
    for block in response.content:
        if hasattr(block, 'text'):
            text = block.text
            if 'entry_' in text:
                # Extract entry id from the response
                import re
                match = re.search(r'entry_[^\s,."\']+', text)
                if match:
                    return match.group(0)
    
    return "entry_unknown"
