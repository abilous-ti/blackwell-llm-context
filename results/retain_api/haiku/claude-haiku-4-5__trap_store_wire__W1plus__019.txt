```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded monetary amount to an account.
    
    Args:
        acct: The account name (string)
        wire: The wire-encoded amount (string like '+1230#6')
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: On failure (unknown account, malformed amount)
    """
    return ledger.post(acct, wire, memo='wire')
```
