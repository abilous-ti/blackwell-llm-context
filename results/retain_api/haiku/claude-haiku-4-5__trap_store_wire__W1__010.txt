
```python
import anthropic
import ledger

def store(acct: str, wire: str) -> int:
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: The account identifier
        wire: The wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id (>=1)
    
    Raises:
        ledger.PostError: If posting fails (unknown account, malformed amount)
    """
    return ledger.post(acct, wire, memo='wire')
```
