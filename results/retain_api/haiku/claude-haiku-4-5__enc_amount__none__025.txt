import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial encoding expert. Your task is to encode money amounts into a team's private wire format.

The format should follow these rules:
1. Convert the dollar amount to cents (multiply by 100)
2. Remove any decimal point
3. Preserve the sign (negative amounts should retain the minus sign)
4. Pad with leading zeros to ensure exactly 10 digits for the absolute value
5. For negative amounts, format as: -XXXXXXXXXX (10 digits after the minus)
6. For positive amounts, format as: +XXXXXXXXXX (10 digits after the plus)
7. For zero, format as: +0000000000

Please encode the following money amount: {dollars}

Respond with ONLY the encoded value, nothing else."""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    test_amounts = ["12.30", "-0.07", "0.00", "1000.00", "-999.99"]
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
