import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are helping encode money amounts to a private wire format. 
                
Given the money amount: {dollars}

Please encode this amount according to the following wire format rules:
1. Extract the dollar and cent parts
2. Represent as: [sign][dollars_part][cents_part] where:
   - [sign] is '+' for positive, '-' for negative, or empty for zero
   - [dollars_part] is the integer part (with leading zeros if needed to make 10 digits)
   - [cents_part] is exactly 2 digits for cents

For example:
- '12.30' becomes '+0000001230'
- '-0.07' becomes '-0000000007'
- '0.00' becomes '0000000000'
- '1234567.89' becomes '+1234567890' (note: 89 cents)

Please respond with ONLY the encoded amount, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = ['12.30', '-0.07', '0.00', '1234567.89', '-999.99', '0.01']
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
