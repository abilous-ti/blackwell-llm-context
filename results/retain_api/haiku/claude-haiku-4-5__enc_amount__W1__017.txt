def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount from dollars string format to the team's private wire format.
    
    The wire format represents amounts as integers in cents to avoid floating point issues.
    For example: '12.30' -> '1230', '-0.07' -> '-7', '0' -> '0'
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount as a string in the wire format (cents as integer)
    
    Raises:
        ValueError: If the input cannot be parsed as a valid money amount
    """
    # Strip whitespace
    dollars = dollars.strip()
    
    # Handle empty string
    if not dollars:
        raise ValueError("Amount cannot be empty")
    
    # Check for valid format (optional sign, digits, optional decimal point, optional digits)
    # Allow formats like: "123", "123.45", "-123", "-123.45", ".45", "-.45"
    import re
    match = re.match(r'^([+-]?)(\d+)?(?:\.(\d+))?$', dollars)
    if not match:
        raise ValueError(f"Invalid amount format: {dollars}")
    
    sign, integer_part, decimal_part = match.groups()
    
    # Handle case where we have only decimal point (e.g., "." or "+.")
    if not integer_part and not decimal_part:
        raise ValueError(f"Invalid amount format: {dollars}")
    
    # Default values
    if not integer_part:
        integer_part = "0"
    if not decimal_part:
        decimal_part = ""
    
    # Pad or truncate decimal part to exactly 2 digits (cents)
    # Pad with zeros if needed (e.g., ".5" -> ".50")
    decimal_part = decimal_part.ljust(2, '0')[:2]
    
    # Convert to cents (integer)
    cents = int(integer_part) * 100 + int(decimal_part)
    
    # Apply sign if present
    if sign == '-':
        cents = -cents
    
    return str(cents)
