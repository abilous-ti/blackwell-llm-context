import ledger


def encode_amount(cents):
    """Encode cents as a wire-format string: <sign><digits>#<check>"""
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    return f"{sign}{digits_str}#{check}"


def record(acct, cents, note):
    """Post cents to account acct with memo note using ledger.post"""
    amount = encode_amount(cents)
    entry_id = ledger.post(acct, amount, memo=note)
    return entry_id
