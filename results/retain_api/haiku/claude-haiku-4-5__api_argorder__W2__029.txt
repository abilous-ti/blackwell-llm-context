def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Transaction reason/memo
    
    Returns:
        Entry ID from the ledger post operation
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Format the amount string: <sign><digits>#<check>
    amount_str = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, amount_str, reason)
    
    return entry_id
