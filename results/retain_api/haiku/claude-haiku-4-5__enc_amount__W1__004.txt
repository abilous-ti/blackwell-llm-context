def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount from decimal dollars string to wire format.
    
    The wire format represents amounts in cents as an integer string,
    with negative sign preserved for negative amounts.
    
    Args:
        dollars: A decimal string like '12.30', '-0.07', '100', etc.
    
    Returns:
        The encoded amount as a string suitable for ledger.post()
    
    Raises:
        ValueError: If the input is not a valid decimal number
    """
    # Parse the input to validate it's a proper decimal number
    try:
        # Convert to float to validate format
        amount_float = float(dollars)
    except ValueError:
        raise ValueError(f"Invalid amount format: {dollars}")
    
    # Convert to cents (multiply by 100) and round to nearest integer
    cents = round(amount_float * 100)
    
    # Return as string representation of cents
    return str(cents)
