def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount in dollars to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in minor units (cents, as integer)
    - <check>: Luhn-style check digit = (sum of decimal digits) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Determine sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars.lstrip('+')
    
    # Parse dollars and cents
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1]
        # Pad or truncate cents to 2 digits
        if len(cents_part) == 1:
            cents_part = cents_part + '0'
        else:
            cents_part = cents_part[:2]
    else:
        dollars_part = amount_str
        cents_part = '00'
    
    # Convert to cents (minor units)
    total_cents = int(dollars_part) * 100 + int(cents_part)
    digits = str(total_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    return f"{sign}{digits}#{check}"
