import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representing a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are helping encode money amounts to a private wire format. 
Given the dollar amount '{dollars}', provide the encoded representation.

Rules for the encoding:
- The format should be consistent and reversible
- Negative amounts should be clearly marked
- Decimal precision should be preserved
- The encoding should be compact

Provide ONLY the encoded value, nothing else."""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    test_amounts = ['12.30', '-0.07', '100.00', '-50.25', '0.01']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"${amount:>8} -> {encoded}")
