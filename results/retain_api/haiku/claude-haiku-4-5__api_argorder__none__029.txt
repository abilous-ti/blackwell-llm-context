import anthropic
import json

client = anthropic.Anthropic()


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds using the ledger module's post function.
    
    Args:
        acct: Account to post to
        cents: Amount in cents
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post
    """
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Post a transaction to the ledger using the ledger module's post function",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Memo or reason for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post a transfer of {cents} cents to account {acct} with the reason '{reason}' using the ledger post function. Return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and extract the entry id
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use" and block.name == "ledger_post":
                # Simulate the ledger post function
                # In a real scenario, this would call the actual ledger module
                tool_input = block.input
                
                # Generate a mock entry id based on the input
                entry_id = f"entry_{hash(json.dumps(tool_input, sort_keys=True)) % 1000000}"
                
                return entry_id
    
    # If no tool was used, try to extract from the response text
    for block in response.content:
        if hasattr(block, 'text'):
            # Try to find an entry id in the response
            text = block.text
            if "entry_" in text:
                # Extract entry id from text
                import re
                match = re.search(r'entry_\w+', text)
                if match:
                    return match.group()
    
    # Default fallback
    return f"entry_{hash(f'{acct}_{cents}_{reason}') % 1000000}"


if __name__ == "__main__":
    # Test the transfer function
    entry_id = transfer("checking", 5000, "Monthly salary deposit")
    print(f"Transfer posted with entry id: {entry_id}")
