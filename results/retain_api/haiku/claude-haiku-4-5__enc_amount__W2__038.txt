def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, no decimal point)
    - <check>: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Handle sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+' if not dollars.startswith('+') else '+'
        amount_str = dollars[1:] if dollars.startswith('+') else dollars
    
    # Remove decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1] if len(parts) > 1 else '00'
        # Pad cents to 2 digits
        cents_part = cents_part.ljust(2, '0')[:2]
        minor_units = int(dollars_part) * 100 + int(cents_part)
    else:
        minor_units = int(amount_str) * 100
    
    digits_str = str(minor_units)
    
    # Calculate check digit (sum of digits mod 10)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    return f"{sign}{digits_str}#{check}"


if __name__ == "__main__":
    # Test cases
    print(encode_amount('12.30'))  # Should be '+1230#6'
    print(encode_amount('-0.07'))  # Should be '-7#7'
    print(encode_amount('0.01'))   # Should be '+1#1'
    print(encode_amount('-12.30')) # Should be '-1230#6'
