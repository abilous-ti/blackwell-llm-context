def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to the team's private wire format.
    
    Args:
        dollars: A string representation of a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount as a string in the team's private wire format
    """
    # Parse the input to validate it's a valid number
    try:
        amount = float(dollars)
    except ValueError:
        raise ValueError(f"Invalid amount format: {dollars}")
    
    # Convert to cents (multiply by 100 and round to nearest integer)
    cents = round(amount * 100)
    
    # Return as string representation of the integer cent value
    return str(cents)
