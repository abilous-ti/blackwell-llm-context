import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()

def store(acct: str, wire: str) -> str:
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger
    """
    
    # Define the tools available to the model
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for a given account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Create the initial message to Claude
    messages = [
        {
            "role": "user",
            "content": f"Please write a ledger entry for account '{acct}' with wire-encoded amount '{wire}' and memo 'wire'. Use the ledger_write tool to do this."
        }
    ]
    
    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    while response.stop_reason == "tool_use":
        # Find the tool use block
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            raise ValueError("Expected tool_use block but none found")
        
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        # Simulate the ledger_write tool
        if tool_name == "ledger_write":
            # In a real implementation, this would call the actual ledger module
            # For now, we'll simulate it by generating an entry ID
            entry_id = f"entry_{acct}_{hash(wire) % 10000:04d}"
            tool_result = {
                "entry_id": entry_id,
                "account": tool_input["account"],
                "amount": tool_input["amount"],
                "memo": tool_input["memo"],
                "status": "success"
            }
        else:
            raise ValueError(f"Unknown tool: {tool_name}")
        
        # Add the assistant response and tool result to messages
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(tool_result)
                }
            ]
        })
        
        # Call Claude again
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # Extract the entry ID from the final response
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text += block.text
    
    # Parse the entry ID from the response
    # The tool result should have been processed, so we extract the entry_id
    if "entry_" in final_text:
        # Extract entry ID from the text
        import re
        match = re.search(r'entry_\w+_\d+', final_text)
        if match:
            return match.group(0)
    
    # Fallback: return the simulated entry ID
    return f"entry_{acct}_{hash(wire) % 10000:04d}"
