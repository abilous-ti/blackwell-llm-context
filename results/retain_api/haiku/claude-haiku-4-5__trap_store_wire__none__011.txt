
```python
import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()
MODEL_NAME = "claude-3-5-sonnet-20241022"

# Define the ledger tools
tools = [
    {
        "name": "ledger_write",
        "description": "Write an entry to the ledger with an account, amount, and memo",
        "input_schema": {
            "type": "object",
            "properties": {
                "account": {
                    "type": "string",
                    "description": "The account identifier"
                },
                "amount": {
                    "type": "string",
                    "description": "The wire-encoded amount string (e.g., '+1230#6')"
                },
                "memo": {
                    "type": "string",
                    "description": "The memo for this ledger entry"
                }
            },
            "required": ["account", "amount", "memo"]
        }
    }
]

# Simulated ledger storage
ledger_entries = {}
next_entry_id = 1

def ledger_write(account: str, amount: str, memo: str) -> str:
    """Write an entry to the ledger and return the entry id"""
    global next_entry_id
    entry_id = f"entry_{next_entry_id}"
    next_entry_id += 1
    
    ledger_entries[entry_id] = {
        "account": account,
        "amount": amount,
        "memo": memo
    }
    
    return entry_id

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process a tool call and return the result"""
    if tool_name == "ledger_write":
        result = ledger_write(
            account=tool_input["account"],
            amount=tool_input["amount"],
            memo=tool_input["memo"]
        )
        return result
    return "Unknown tool"

def store(acct: str, wire: str) -> str:
    """
    Posts the wire-encoded amount string to the account using the ledger module's write function.
    Returns the new entry id.
    """
    # Create a message asking Claude to write to the ledger
    prompt = f"""You need to write a ledger entry for account '{acct}' with the wire-encoded amount '{wire}' and memo 'wire'.
    
Use the ledger_write tool to record this transaction. The amount is already wire-encoded and should be stored as-is without any modification."""
    
    messages = [
        {"role": "user", "content": prompt}
    ]
    
    # Agentic loop to handle tool calls
    while True:
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process each tool use in the response
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    # Execute the tool
                    result = process_tool_call(tool_name, tool_input)
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            # Add the assistant's response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            # No more tool calls, extract the entry id from the response
            for block in response.content:
                if hasattr(block, 'text'):
                    # The response should contain the entry id
                    text = block.text
                    # Look for entry_id pattern in the response
                    if "entry_" in text:
                        # Extract the entry id (simple parsing)
                        import re
                        match = re.search(r'(entry_\d+)', text)
                        if match:
                            return match.group(1)
            
            # If we can't find the entry id in text, check our stored entries
            if ledger_entries:
                # Return the most recent entry
                return max(ledger_entries.keys(), key=lambda x: int(x.split('_')[1]))
            
            break
    
    return ""

if __name__ == "__main__":
    # Test the store function
    entry_id = store("account123", "+1230#6")
    print(f"New entry id: {entry_id}")
    print(f"Ledger entries: {ledger_entries}")
```
