def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount in dollars to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents as integer)
    - <check>: Luhn-style digit = (sum of decimal digits) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the input
    dollars_str = dollars.strip()
    
    # Determine sign
    if dollars_str.startswith('-'):
        sign = '-'
        amount_str = dollars_str[1:]
    elif dollars_str.startswith('+'):
        sign = '+'
        amount_str = dollars_str[1:]
    else:
        sign = '+'
        amount_str = dollars_str
    
    # Handle the decimal point and convert to cents
    if '.' in amount_str:
        parts = amount_str.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1]
        
        # Pad or truncate cents to 2 digits
        if len(cents_part) == 1:
            cents_part = cents_part + '0'
        elif len(cents_part) > 2:
            cents_part = cents_part[:2]
        
        # Convert to total cents
        total_cents = int(dollars_part) * 100 + int(cents_part)
    else:
        # No decimal point - assume it's dollars
        total_cents = int(amount_str) * 100
    
    # Convert cents to string (digits only)
    digits = str(total_cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    check_digit = sum(int(d) for d in digits) % 10
    
    # Build the encoded string
    return f"{sign}{digits}#{check_digit}"
