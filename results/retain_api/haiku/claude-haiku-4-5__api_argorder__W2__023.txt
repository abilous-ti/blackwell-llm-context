def transfer(acct, cents, reason):
    """
    Posts a monetary transfer to an account using the wire-encoding format.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger post operation
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    # Format the amount string
    amount_str = f"{sign}{digits}#{check}"
    
    # Post to ledger
    entry_id = post(acct, amount_str, reason)
    
    return entry_id
