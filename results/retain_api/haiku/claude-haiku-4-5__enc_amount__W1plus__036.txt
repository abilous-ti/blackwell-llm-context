def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, no decimal point)
    - <check>: Luhn-style check digit = (sum of decimal digits) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Determine the sign
    if dollars.startswith('-'):
        sign = '-'
        dollars = dollars[1:]
    else:
        sign = '+'
    
    # Remove the decimal point and convert to cents
    # Handle cases like '12.30', '0.07', '12', etc.
    if '.' in dollars:
        parts = dollars.split('.')
        integer_part = parts[0] if parts[0] else '0'
        decimal_part = parts[1].ljust(2, '0')[:2]  # Pad or truncate to 2 digits
        digits_str = integer_part + decimal_part
    else:
        # No decimal point, treat as whole dollars
        digits_str = dollars + '00'
    
    # Convert to integer to remove leading zeros
    amount_cents = int(digits_str)
    digits_str = str(amount_cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    return f"{sign}{digits_str}#{check_digit}"
