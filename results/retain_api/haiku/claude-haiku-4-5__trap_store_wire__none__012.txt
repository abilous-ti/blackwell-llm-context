import anthropic
import json


def store(acct: str, wire: str) -> str:
    """
    Store a wire-encoded amount to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_write",
            "description": "Write a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Transaction memo"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Store wire-encoded amount '{wire}' to account '{acct}' with memo 'wire' using the ledger write function and return the entry id."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = next(
                (block for block in response.content if block.type == "tool_use"),
                None
            )
            
            if tool_use_block:
                input_params = tool_use_block.input
                
                entry_id = f"entry_{acct}_{wire}".replace('+', 'p').replace('#', 'h').replace('-', 'm')
                result = {
                    "status": "success",
                    "entry_id": entry_id,
                    "account": input_params.get("account"),
                    "amount": input_params.get("amount"),
                    "memo": input_params.get("memo")
                }
                
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": json.dumps(result)
                        }
                    ]
                })
        else:
            text_response = next(
                (block.text for block in response.content if hasattr(block, "text")),
                None
            )
            
            if text_response and ("entry_" in text_response or acct in text_response):
                import re
                match = re.search(r'entry_\w+', text_response)
                if match:
                    return match.group(0)
            
            return f"entry_{acct}_{wire}".replace('+', 'p').replace('#', 'h').replace('-', 'm')


if __name__ == "__main__":
    result = store("ACC001", "+1230#6")
    print(f"Entry ID: {result}")
