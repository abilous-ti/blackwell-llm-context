import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representation of a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format representation of the amount
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a wire format encoder for a financial system. 
                
Your task is to encode a dollar amount into a private wire format. 
The wire format should be a compact binary-like representation suitable for wire transmission.

For the amount '{dollars}', provide the encoded wire format.
Return ONLY the encoded format string, nothing else.
Do not include explanations or additional text."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    # Test cases
    test_amounts = ['12.30', '-0.07', '0.00', '1000.50', '-999.99']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"Amount: {amount} => Encoded: {encoded}")
