def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount to the team's private wire format.
    
    The wire format represents dollars as cents (integer), prefixed with
    a sign indicator: '+' for positive/zero, '-' for negative.
    
    Examples:
        '12.30' -> '+1230'
        '-0.07' -> '-7'
        '0.00' -> '+0'
    """
    try:
        amount_float = float(dollars)
    except (ValueError, TypeError):
        raise ValueError(f"Invalid amount format: {dollars}")
    
    cents = round(amount_float * 100)
    
    if cents >= 0:
        return f"+{cents}"
    else:
        return f"-{-cents}"
