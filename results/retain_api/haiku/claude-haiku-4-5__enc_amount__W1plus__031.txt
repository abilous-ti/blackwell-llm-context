def encode_amount(dollars: str) -> str:
    """Encode a dollar amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents, no decimal point)
    - check: (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the input
    is_negative = dollars.startswith('-')
    amount_str = dollars.lstrip('+-')
    
    # Remove decimal point and get cents as integer
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1].ljust(2, '0')[:2]  # Pad or truncate to 2 digits
        cents = int(dollars_part) * 100 + int(cents_part)
    else:
        cents = int(amount_str) * 100
    
    # Build digits string
    digits_str = str(cents)
    
    # Calculate check digit (sum of decimal digits mod 10)
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Build result
    sign = '-' if is_negative else '+'
    return f"{sign}{digits_str}#{check_digit}"
