def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger encoding.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for transfer
    
    Returns:
        Entry ID from the ledger
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value of cents
    abs_cents = abs(cents)
    
    # Convert to string of digits
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format as <sign><digits>#<check>
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, encoded_amount, reason)
    
    return entry_id
