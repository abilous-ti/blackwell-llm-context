import anthropic
import json

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account using the ledger module's write function.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for an account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "A memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Write a ledger entry for account '{acct}' with wire-encoded amount '{wire}' and memo 'wire'. Use the ledger_write tool."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = None
            for block in response.content:
                if block.type == "tool_use":
                    tool_use_block = block
                    break
            
            if tool_use_block and tool_use_block.name == "ledger_write":
                from ledger import write
                
                result = write(
                    account=tool_use_block.input["account"],
                    amount=tool_use_block.input["amount"],
                    memo=tool_use_block.input["memo"]
                )
                
                messages.append({
                    "role": "assistant",
                    "content": response.content
                })
                
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": str(result)
                        }
                    ]
                })
                
                return result
        else:
            break
    
    return None
