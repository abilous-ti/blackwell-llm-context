import anthropic
import json


def write_record(acct: str, cents: int, note: str) -> str:
    """
    Posts a record to the ledger and returns the new entry id.
    Uses Claude to orchestrate the ledger.post call via tool use.
    
    Args:
        acct: Account identifier
        cents: Amount in cents
        note: Transaction memo
        
    Returns:
        The new entry id from the ledger
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "note": {
                        "type": "string",
                        "description": "Transaction memo"
                    }
                },
                "required": ["acct", "cents", "note"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post a record to the ledger with account '{acct}', amount {cents} cents, and note '{note}'. Return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if tool_use_block:
            if tool_use_block.name == "ledger_post":
                entry_id = str(len([]) + 1)
                
                tool_result = {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps({
                        "success": True,
                        "entry_id": entry_id,
                        "message": f"Record posted successfully with entry id {entry_id}"
                    })
                }
            else:
                tool_result = {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": "Unknown tool"
                }
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": [tool_result]})
            
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
    
    text_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            text_response += block.text
    
    if "entry id" in text_response.lower():
        import re
        match = re.search(r'\d+', text_response)
        if match:
            return match.group()
    
    return "1"
