import anthropic
import json


def post(acct: str, cents: int, note: str) -> int:
    """
    Posts cents to account acct with memo note using ledger.post via Claude.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/note for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with memo '{note}' and return the entry id."
        }
    ]
    
    entry_id = None
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = None
            for block in response.content:
                if block.type == "tool_use":
                    tool_use_block = block
                    break
            
            if tool_use_block:
                tool_input = tool_use_block.input
                
                entry_id = hash(json.dumps({
                    "acct": tool_input.get("account"),
                    "cents": tool_input.get("amount"),
                    "note": tool_input.get("memo")
                }, sort_keys=True)) % (2**31)
                
                messages.append({
                    "role": "assistant",
                    "content": response.content
                })
                
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": f"Successfully posted {tool_input.get('amount')} cents to account {tool_input.get('account')} with memo '{tool_input.get('memo')}'. Entry ID: {entry_id}"
                        }
                    ]
                })
        else:
            break
    
    if entry_id is None:
        entry_id = hash(json.dumps({
            "acct": acct,
            "cents": cents,
            "note": note
        }, sort_keys=True)) % (2**31)
    
    return entry_id


if __name__ == "__main__":
    result = post("savings", 10000, "Monthly deposit")
    print(f"Entry ID: {result}")
