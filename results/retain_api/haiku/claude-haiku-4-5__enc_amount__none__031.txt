import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are helping to encode money amounts into a team's private wire format.

Given a dollar amount as a string, you need to encode it into the wire format.

The wire format appears to follow these rules:
1. Convert the dollar amount to cents (multiply by 100)
2. Represent it as a signed integer
3. Encode it using a specific format

Please encode this dollar amount into the wire format: {dollars}

Respond with ONLY the encoded value, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = ["12.30", "-0.07", "0.00", "100.00", "-50.25"]
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"encode_amount('{amount}') = {encoded}")
