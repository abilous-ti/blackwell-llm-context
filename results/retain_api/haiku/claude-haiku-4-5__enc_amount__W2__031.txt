def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount in dollars to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, as integer without decimal)
    - <check>: Luhn-style check digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Handle sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars.lstrip('+')
    
    # Remove any whitespace
    amount_str = amount_str.strip()
    
    # Convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1] if len(parts) > 1 else '0'
        
        # Pad or truncate cents to 2 digits
        cents_part = cents_part.ljust(2, '0')[:2]
        
        cents = int(dollars_part) * 100 + int(cents_part)
    else:
        cents = int(amount_str) * 100
    
    # Convert cents to string
    digits = str(cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    return f"{sign}{digits}#{check}"
