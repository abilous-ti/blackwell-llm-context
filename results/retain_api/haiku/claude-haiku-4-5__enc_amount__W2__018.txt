import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents) as integer, no decimal point
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Example: '12.30' -> '+1230#6', '-0.07' -> '-7#7'
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to encode a monetary amount to a specific wire format.

The format is: <sign><digits>#<check>
- <sign>: '+' for positive or '-' for negative
- <digits>: the amount in MINOR UNITS (cents) as an integer with no decimal point
  - For example, $12.30 = 1230 cents
  - For example, $0.07 = 7 cents
- <check>: a single Luhn-style digit calculated as (sum of all decimal digits in <digits>) mod 10

Examples:
- '-0.07' should encode as '-7#7' (7 cents negative, digit sum = 7, check = 7 mod 10 = 7)
- '+12.30' should encode as '+1230#6' (1230 cents positive, digit sum = 1+2+3+0 = 6, check = 6 mod 10 = 6)

Please encode this amount: '{dollars}'

Return ONLY the encoded string in the format <sign><digits>#<check>, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = [
        ("12.30", "+1230#6"),
        ("-0.07", "-7#7"),
        ("0.01", "+1#1"),
        ("100.00", "+10000#1"),
        ("-50.25", "-5025#12"),
    ]
    
    print("Testing encode_amount function:")
    for input_val, expected in test_cases:
        result = encode_amount(input_val)
        status = "✓" if result == expected else "✗"
        print(f"{status} encode_amount('{input_val}') = '{result}' (expected: '{expected}')")
