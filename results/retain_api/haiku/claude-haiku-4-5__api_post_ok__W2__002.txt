import ledger

def record(acct, cents, note):
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    digits_str = str(abs(cents))
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    # Format the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return the entry id
    return ledger.post(acct, encoded_amount, note)
