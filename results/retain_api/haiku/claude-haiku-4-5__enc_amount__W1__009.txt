def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount from decimal string format to the team's wire format.
    
    The wire format represents amounts as cents (integer) to avoid floating point issues.
    
    Args:
        dollars: A string representation of money (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    # Remove any whitespace
    dollars = dollars.strip()
    
    # Handle the decimal point and convert to cents
    if '.' in dollars:
        parts = dollars.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        
        integer_part = parts[0] if parts[0] and parts[0] != '-' else ('0' if not parts[0] else '-0')
        decimal_part = parts[1]
        
        # Pad or truncate decimal part to 2 digits (cents)
        if len(decimal_part) > 2:
            decimal_part = decimal_part[:2]
        else:
            decimal_part = decimal_part.ljust(2, '0')
        
        # Handle negative sign
        is_negative = integer_part.startswith('-')
        if is_negative:
            integer_part = integer_part[1:]
        
        cents = int(integer_part) * 100 + int(decimal_part)
        
        if is_negative:
            cents = -cents
    else:
        # No decimal point, treat as dollars
        cents = int(dollars) * 100
    
    return str(cents)
