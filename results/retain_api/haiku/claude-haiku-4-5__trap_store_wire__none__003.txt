import anthropic
import json


def store(acct: str, wire: str) -> str:
    """
    Store a wire-encoded monetary amount to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger module",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Store the wire-encoded amount '{wire}' to account '{acct}' using the ledger write function with memo 'wire'. Return the entry id."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = next(
                (block for block in response.content if block.type == "tool_use"),
                None
            )
            
            if tool_use_block and tool_use_block.name == "ledger_write":
                input_data = tool_use_block.input
                entry_id = f"entry_{hash(json.dumps(input_data)) % 1000000}"
                
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": json.dumps({"entry_id": entry_id})
                        }
                    ]
                })
                
                return entry_id
        else:
            for block in response.content:
                if hasattr(block, 'text'):
                    return block.text.strip()
            return "unknown"
