import anthropic
import json
import re


def store(acct: str, wire: str) -> str:
    """
    Posts a wire-encoded amount string to an account using the ledger module's write function.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id
    """
    client = anthropic.Anthropic()
    
    # Create a tool definition for the ledger write function
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for an account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Use Claude to call the ledger write function
    messages = [
        {
            "role": "user",
            "content": f"Write to the ledger: account='{acct}', wire-encoded amount='{wire}', memo='wire'. Use the ledger_write tool and return the entry id from the response."
        }
    ]
    
    # Keep track of the entry id from tool results
    entry_id = None
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Try to extract entry_id from the response text
            if response.content:
                for block in response.content:
                    if hasattr(block, 'text'):
                        # Look for entry_id in the response
                        match = re.search(r'entry[_-]?id["\']?\s*[:=]\s*["\']?([^"\'\s,}]+)', block.text, re.IGNORECASE)
                        if match:
                            entry_id = match.group(1)
                            break
            break
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process each tool use block
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    if block.name == "ledger_write":
                        # Simulate the ledger write function
                        input_data = block.input
                        # In a real system, this would write to an actual ledger
                        # For this exercise, we'll generate a mock entry id
                        entry_id = f"entry_{hash(f'{input_data.get('account', '')}{input_data.get('amount', '')}') % 1000000}"
                        
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": json.dumps({
                                "success": True,
                                "entry_id": entry_id,
                                "account": input_data.get("account"),
                                "amount": input_data.get("amount"),
                                "memo": input_data.get("memo")
                            })
                        })
            
            # Add tool results to messages
            if tool_results:
                messages.append({
                    "role": "user",
                    "content": tool_results
                })
        else:
            # If stop_reason is neither "end_turn" nor "tool_use", break to avoid infinite loop
            break
    
    # Return the entry id or a default if not found
    return entry_id if entry_id else f"entry_wire_{hash(wire) % 1000000}"
