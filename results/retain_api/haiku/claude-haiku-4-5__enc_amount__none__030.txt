import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are helping encode money amounts to a private wire format. 
Given a dollar amount as a string, encode it to the team's private wire format.
The format should be consistent and reversible.

Dollar amount to encode: '{dollars}'

Please provide ONLY the encoded format as output, with no explanation."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    # Test the function with a few examples
    test_amounts = ['12.30', '-0.07', '0.00', '1000.99', '-999.50']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"Encoding '{amount}': {encoded}")
