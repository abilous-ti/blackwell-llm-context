import anthropic
import json
import re


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount (e.g., '12.30', '-0.07') to the team's private wire format.
    Uses Claude to determine the correct encoding based on the ledger module API contract.
    """
    client = anthropic.Anthropic()
    
    prompt = f"""You are helping determine the wire format for encoding money amounts in a financial ledger system.

The ledger module has this API:
- ledger.post(account: str, amount, *, memo: str) 
- Takes: account (str), amount (positional), memo (keyword-only, required)
- Returns: new integer entry id (>=1) on success
- Raises: ledger.PostError on failure

Given that the `amount` parameter needs to be passed to ledger.post(), and considering common financial wire formats and the fact that this is a "private wire format" for internal team use, what would be the appropriate wire format encoding for the following money amount?

Money amount input: '{dollars}'

Examples of typical inputs:
- '12.30' (twelve dollars and thirty cents)
- '-0.07' (negative seven cents)
- '0' (zero)
- '100' (one hundred dollars)

Please provide ONLY the encoded wire format string that would be passed as the `amount` parameter to ledger.post(). No explanation, just the format. Think about common formats like:
1. Pennies as integer (e.g., 1230 for 12.30)
2. Fixed-point decimal representation
3. String representation with special encoding
4. Scientific notation

Return ONLY the encoded string, nothing else."""
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=100,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded
