def encode_amount(dollars: str) -> str:
    """
    Encode a money amount to the team's wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents) without decimal point
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Remove any whitespace
    dollars = dollars.strip()
    
    # Determine sign
    if dollars.startswith('-'):
        sign = '-'
        dollars = dollars[1:]
    elif dollars.startswith('+'):
        sign = '+'
        dollars = dollars[1:]
    else:
        sign = '+'
    
    # Remove leading zeros and whitespace
    dollars = dollars.strip()
    
    # Handle decimal point
    if '.' in dollars:
        parts = dollars.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        integer_part = parts[0] if parts[0] else '0'
        decimal_part = parts[1]
        
        # Pad or truncate decimal part to 2 digits
        if len(decimal_part) == 1:
            decimal_part = decimal_part + '0'
        elif len(decimal_part) > 2:
            decimal_part = decimal_part[:2]
        
        # Combine to get cents
        cents_str = integer_part + decimal_part
    else:
        # No decimal point - treat as dollars, convert to cents
        cents_str = dollars + '00'
    
    # Remove leading zeros but keep at least one digit
    cents_str = cents_str.lstrip('0') or '0'
    
    # Calculate check digit - sum of all decimal digits mod 10
    check_digit = sum(int(d) for d in cents_str) % 10
    
    return f"{sign}{cents_str}#{check_digit}"
