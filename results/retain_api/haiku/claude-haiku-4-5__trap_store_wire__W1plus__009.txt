```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: The account name/identifier (str)
        wire: The wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id (int >= 1)
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    return ledger.post(acct, wire, memo='wire')
```
