import ledger

def record(acct, cents, note):
    """
    Post a monetary amount to an account and return the new entry id.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer, can be negative)
        note: Memo text for the transaction
    
    Returns:
        The new entry id from the ledger post operation
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value for digits
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    # Format as required: <sign><digits>#<check>
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return the entry id
    entry_id = ledger.post(acct, encoded_amount, note)
    
    return entry_id
