```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts a monetary amount to an account using the ledger API.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer (string)
    
    Returns:
        Entry id (integer >= 1) on success
    
    Raises:
        ledger.PostError: On any failure (unknown account, malformed amount)
    """
    # Convert cents to wire-encoded format
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with amount as positional arg and memo as keyword-only arg
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
