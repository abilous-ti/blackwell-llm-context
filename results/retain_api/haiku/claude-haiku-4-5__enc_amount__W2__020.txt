def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount into the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in MINOR UNITS (cents, no decimal point)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Handle sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars.lstrip('+')
    
    # Remove the decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        integer_part = parts[0]
        decimal_part = parts[1]
        
        # Pad or truncate decimal part to 2 digits
        decimal_part = (decimal_part + '00')[:2]
        
        digits = integer_part + decimal_part
    else:
        # If no decimal point, assume it's in dollars
        digits = amount_str + '00'
    
    # Convert to integer to remove leading zeros, then back to string
    cents_value = int(digits)
    digits = str(cents_value)
    
    # Calculate check digit
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    return f"{sign}{digits}#{check}"


if __name__ == "__main__":
    # Test cases
    print(encode_amount('12.30'))  # Expected: '+1230#6'
    print(encode_amount('-0.07'))  # Expected: '-7#7'
    print(encode_amount('0.00'))   # Expected: '+0#0'
    print(encode_amount('100'))    # Expected: '+10000#1'
