def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, no decimal point)
    - <check>: (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Determine sign
    sign = '+' if not dollars.startswith('-') else '-'
    
    # Remove sign for processing
    amount_str = dollars.lstrip('+-')
    
    # Remove decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1]
        # Pad cents to 2 digits or truncate to 2 digits
        cents_part = (cents_part + '00')[:2]
        cents = int(dollars_part) * 100 + int(cents_part)
    else:
        cents = int(amount_str) * 100
    
    # Convert cents to string (without sign)
    digits = str(cents)
    
    # Calculate check digit: sum of digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Return encoded amount
    return f"{sign}{digits}#{check}"
