import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount in dollars to the team's private wire format.
    
    Uses Claude to understand the team's private wire format for monetary amounts
    based on the ledger module API context.
    
    Args:
        dollars: A money amount as a string (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    prompt = f"""Given the team context:
- The ledger module has a canonical write call: ledger.post(account: str, amount, *, memo: str)
- The 'amount' parameter is SECOND and positional
- On success, post() returns a new integer entry id (>=1)
- On failure, it raises ledger.PostError

Please determine what the private wire format is for encoding monetary amounts in this system.
The input amount is: {dollars}

Based on the ledger API context where amounts are passed as a positional parameter after the account string,
what would be the appropriate wire format encoding for this amount?

Please respond with only the encoded amount, no explanation."""
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    # Extract the encoded amount from Claude's response
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    # Test examples
    test_amounts = ['12.30', '-0.07', '100', '0.01']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"Input: {amount:>8} -> Encoded: {encoded}")
