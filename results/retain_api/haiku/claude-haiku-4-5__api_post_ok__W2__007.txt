import ledger


def record(acct, cents, note):
    """Post a monetary amount to an account and return the entry id."""
    # Determine the sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value
    abs_cents = abs(cents)
    
    # Convert to string of digits
    digits = str(abs_cents)
    
    # Calculate check digit (sum of decimal digits mod 10)
    check = sum(int(d) for d in digits) % 10
    
    # Format as the wire-encoding invariant
    encoded = f"{sign}{digits}#{check}"
    
    # Post to ledger and return the entry id
    entry_id = ledger.post(acct, encoded, note)
    
    return entry_id
