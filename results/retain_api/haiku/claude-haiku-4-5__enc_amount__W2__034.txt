```python
import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encode a money amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents, no decimal point)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Example: '12.30' -> '+1230#6', '-0.07' -> '-7#7'
    """
    client = anthropic.Anthropic()
    
    prompt = f"""You are a financial encoding expert. Your task is to encode a money amount to a specific wire format.

The encoding format is: <sign><digits>#<check>
Where:
- <sign> is '+' for positive amounts or '-' for negative amounts
- <digits> is the amount in MINOR UNITS (integer cents, with no decimal point)
  For example: $12.30 has 1230 cents, $0.07 has 7 cents
- <check> is a single digit equal to (sum of all decimal digits in <digits>) mod 10
  For example: if digits is "1230", the sum is 1+2+3+0=6, so check digit is 6 mod 10 = 6

Examples:
- Input: '12.30' -> Output: '+1230#6' (sign: +, cents: 1230, digit sum: 1+2+3+0=6, check: 6)
- Input: '-0.07' -> Output: '-7#7' (sign: -, cents: 7, digit sum: 7, check: 7 mod 10 = 7)
- Input: '100.00' -> Output: '+10000#1' (sign: +, cents: 10000, digit sum: 1+0+0+0+0=1, check: 1)

Now encode this amount: '{dollars}'

Return ONLY the encoded string in the format <sign><digits>#<check>, with no explanation."""

    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=100,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    result = message.content[0].text.strip()
    return result
```
