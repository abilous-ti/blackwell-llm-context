import anthropic

def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry ID."""
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Post a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo or note for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account {acct} with memo '{note}' and return the entry ID."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and tool use
    entry_id = None
    
    while response.stop_reason == "tool_use":
        # Find the tool use block
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        # Simulate the ledger.post function
        # In a real scenario, this would call the actual ledger module
        # For now, we'll generate a mock entry ID
        import uuid
        entry_id = str(uuid.uuid4())
        
        # Continue the conversation with the tool result
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": f"Entry ID: {entry_id}"
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # If no tool was used or we got here without an entry ID, generate one
    if entry_id is None:
        import uuid
        entry_id = str(uuid.uuid4())
    
    return entry_id


if __name__ == "__main__":
    # Test the function
    result = record("checking", 10000, "salary deposit")
    print(f"Entry ID: {result}")
