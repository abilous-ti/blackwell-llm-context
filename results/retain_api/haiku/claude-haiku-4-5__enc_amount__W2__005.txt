def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, no decimal point)
    - <check>: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '-0.07' -> '-7#7'
    - '12.30' -> '+1230#6'
    """
    # Handle the sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars.lstrip('+')
    
    # Remove the decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1] if len(parts) > 1 else '00'
        
        # Pad cents to 2 digits
        cents_part = cents_part.ljust(2, '0')[:2]
        
        # Combine to get total cents
        minor_units = int(dollars_part) * 100 + int(cents_part)
    else:
        # No decimal point, assume it's already in dollars
        minor_units = int(amount_str) * 100
    
    # Convert minor units to string
    digits_str = str(minor_units)
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Format the result
    return f"{sign}{digits_str}#{check_digit}"


# Test with the provided examples
if __name__ == "__main__":
    print(encode_amount('-0.07'))  # Should output: -7#7
    print(encode_amount('12.30'))  # Should output: +1230#6
    print(encode_amount('0'))      # Should output: +0#0
    print(encode_amount('-12.30')) # Should output: -1230#6
