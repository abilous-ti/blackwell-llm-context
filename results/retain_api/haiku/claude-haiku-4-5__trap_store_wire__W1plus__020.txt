```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account name (string)
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: On failure (unknown account, malformed amount)
    """
    return ledger.post(acct, wire, memo='wire')
```
