import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to encode a money amount to a private wire format. 
                
The amount in dollars is: {dollars}

Please encode this to the wire format. Return ONLY the encoded format string, nothing else.

Examples of wire format encoding:
- For positive amounts, use a format that clearly represents the value
- For negative amounts, indicate the negative sign appropriately
- Ensure precision to cents (2 decimal places)

Return only the encoded string."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = [
        "12.30",
        "-0.07",
        "0.00",
        "999.99",
        "-123.45"
    ]
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"${amount} -> {encoded}")
