import anthropic
import ledger


def transfer(acct: str, cents: int, reason: str) -> int:
    """
    Transfer funds to an account using the ledger API.
    
    Args:
        acct: The account identifier
        cents: The amount in cents to transfer
        reason: The reason/memo for the transfer
        
    Returns:
        The entry ID from the ledger post operation
        
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Use the ledger.post API with the correct signature:
    # ledger.post(account: str, amount, *, memo: str)
    # account is first, amount is second (positional), memo is keyword-only
    return ledger.post(acct, cents, memo=reason)
