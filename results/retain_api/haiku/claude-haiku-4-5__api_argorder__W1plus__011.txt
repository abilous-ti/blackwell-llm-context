```python
import ledger

def transfer(acct, cents, reason):
    """
    Posts a monetary transfer to an account.
    
    Args:
        acct: Account identifier (str)
        cents: Amount in cents (int)
        reason: Memo/reason for the transfer (str)
    
    Returns:
        The entry id (int >= 1) of the posted transaction
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Encode the amount in the wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with the correct signature:
    # post(account: str, amount, *, memo: str)
    return ledger.post(acct, encoded_amount, memo=reason)
```
