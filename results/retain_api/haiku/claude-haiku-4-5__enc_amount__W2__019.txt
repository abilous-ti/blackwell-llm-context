def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in MINOR UNITS (cents, as integer)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Handle the sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars.lstrip('+')
    
    # Remove the decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1] if len(parts) > 1 else '00'
        
        # Pad cents to 2 digits
        cents_part = cents_part.ljust(2, '0')[:2]
        
        # Combine to get total cents
        total_cents = int(dollars_part) * 100 + int(cents_part)
    else:
        # No decimal point, assume it's dollars
        total_cents = int(amount_str) * 100
    
    # Convert to string without sign
    digits_str = str(abs(total_cents))
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Construct the wire format
    return f"{sign}{digits_str}#{check_digit}"


if __name__ == "__main__":
    # Test cases from the examples
    print(encode_amount('12.30'))  # Should be '+1230#6'
    print(encode_amount('-0.07'))  # Should be '-7#7'
    print(encode_amount('0.01'))   # Should be '+1#1'
    print(encode_amount('-100.00'))  # Should be '-10000#1'
