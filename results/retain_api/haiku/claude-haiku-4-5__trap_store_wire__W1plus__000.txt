```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account in the ledger.
    
    Args:
        acct: Account identifier (string)
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    return ledger.post(acct, wire, memo='wire')
```
