```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    entry_id = ledger.write(acct, wire, memo='wire')
    return entry_id
```
