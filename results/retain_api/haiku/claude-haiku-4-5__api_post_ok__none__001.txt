import anthropic
import json
from typing import Any

# Mock ledger module for demonstration
class MockLedger:
    _entry_id_counter = 0
    _entries = []
    
    @classmethod
    def post(cls, acct: str, cents: int, note: str) -> int:
        cls._entry_id_counter += 1
        entry = {
            "id": cls._entry_id_counter,
            "account": acct,
            "cents": cents,
            "note": note
        }
        cls._entries.append(entry)
        return cls._entry_id_counter

# Initialize the mock ledger as a module-like object
ledger = MockLedger()

def record(acct: str, cents: int, note: str) -> int:
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post_to_ledger",
            "description": "Posts an entry to the ledger with account, amount in cents, and a note",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "note": {
                        "type": "string",
                        "description": "A memo/note for the transaction"
                    }
                },
                "required": ["acct", "cents", "note"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with the note '{note}' and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and handle tool use
    while response.stop_reason == "tool_use":
        tool_use = next(
            (block for block in response.content if block.type == "tool_use"),
            None
        )
        
        if tool_use and tool_use.name == "post_to_ledger":
            input_data = tool_use.input
            entry_id = ledger.post(
                acct=input_data["acct"],
                cents=input_data["cents"],
                note=input_data["note"]
            )
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use.id,
                        "content": json.dumps({"entry_id": entry_id})
                    }
                ]
            })
            
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
    
    # Direct implementation - use ledger.post directly
    entry_id = ledger.post(acct=acct, cents=cents, note=note)
    return entry_id


if __name__ == "__main__":
    # Test the record function
    entry_id = record("checking", 12345, "Deposit")
    print(f"New entry ID: {entry_id}")
    
    entry_id2 = record("savings", 5000, "Transfer")
    print(f"New entry ID: {entry_id2}")
