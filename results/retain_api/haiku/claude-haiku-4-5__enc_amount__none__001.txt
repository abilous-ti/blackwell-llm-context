import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representing a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format representation of the amount
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial wire format encoder. Given a dollar amount as a string, 
encode it to our team's private wire format.

The wire format rules are:
1. Remove the dollar sign if present
2. Convert to cents (multiply by 100)
3. Ensure it's an integer
4. Use two's complement representation for negative numbers using 32-bit signed integer
5. Convert to hexadecimal (8 characters, zero-padded)
6. Return in uppercase

Examples:
- '12.30' -> 1230 cents -> 0x000004CE
- '-0.07' -> -7 cents -> 0xFFFFFFF9
- '0.00' -> 0 cents -> 0x00000000
- '100.00' -> 10000 cents -> 0x00002710

Encode this amount: {dollars}
Return ONLY the 8-character hex code in uppercase, nothing else."""
            }
        ]
    )
    
    result = message.content[0].text.strip()
    return result


if __name__ == "__main__":
    test_cases = [
        "12.30",
        "-0.07", 
        "0.00",
        "100.00",
        "-50.25"
    ]
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
