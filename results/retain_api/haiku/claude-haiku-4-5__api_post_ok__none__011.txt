import anthropic
import json

def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts an entry to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/note for this entry"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account '{acct}' with note '{note}' using the ledger.post tool and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                tool_input = block.input
                entry_id = str(hash((acct, cents, note)) % 1000000)
                return entry_id
    
    return None
