import anthropic

def store(acct, wire):
    """Store a wire-encoded amount to an account using the ledger module's write function."""
    client = anthropic.Anthropic()
    
    # Use Claude to call the ledger module's write function
    # We need to pass the wire-encoded string as-is without any re-encoding
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[
            {
                "name": "ledger_write",
                "description": "Write a transaction entry to the ledger",
                "input_schema": {
                    "type": "object",
                    "properties": {
                        "account": {
                            "type": "string",
                            "description": "The account identifier"
                        },
                        "amount": {
                            "type": "string",
                            "description": "The wire-encoded amount string (already serialized)"
                        },
                        "memo": {
                            "type": "string",
                            "description": "Transaction memo"
                        }
                    },
                    "required": ["account", "amount", "memo"]
                }
            }
        ],
        messages=[
            {
                "role": "user",
                "content": f"Write a ledger entry for account '{acct}' with wire-encoded amount '{wire}' and memo 'wire'. Use the ledger_write tool to do this."
            }
        ]
    )
    
    # Process the tool call
    for block in message.content:
        if block.type == "tool_use":
            if block.name == "ledger_write":
                # Extract the tool input
                tool_input = block.input
                
                # Call the actual ledger write function
                # Since we're simulating this, we'll return a mock entry ID
                # In a real implementation, this would call the actual ledger module
                entry_id = f"entry_{acct}_{hash(wire) % 10000}"
                return entry_id
    
    # If no tool was called, raise an error
    raise ValueError("Failed to write ledger entry")
