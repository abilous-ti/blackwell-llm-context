def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount from decimal string format to wire format.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded wire format as a string
    """
    # Parse the input to validate it's a proper decimal amount
    try:
        # Convert to float to validate format, then back to string to preserve precision
        amount_float = float(dollars)
    except ValueError:
        raise ValueError(f"Invalid amount format: {dollars}")
    
    # Convert to cents (integer representation) to avoid floating point errors
    # Multiply by 100 and round to nearest integer
    cents = round(amount_float * 100)
    
    # Return as string representation of cents
    return str(cents)
