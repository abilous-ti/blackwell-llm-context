def transfer(acct, cents, reason):
    """
    Posts a monetary transfer to an account using the wire-encoding format.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer, can be negative)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post operation
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Construct wire-encoded amount
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    return post(acct, encoded_amount, reason)
