import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (integer cents, no decimal point)
    - <check>: Luhn-style digit = (sum of decimal digits) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to encode a monetary amount to a specific wire format.

The encoding format is: <sign><digits>#<check>
- <sign>: '+' or '-'
- <digits>: amount in MINOR UNITS (integer cents, no decimal point)
- <check>: Luhn-style digit equal to (sum of decimal digits) mod 10

Examples:
- '-$0.07' -> '-7#7' (7 cents negative, digit sum is 7, 7 mod 10 = 7)
- '+$12.30' -> '+1230#6' (1230 cents positive, digit sum is 1+2+3+0=6, 6 mod 10 = 6)

Please encode this amount: {dollars}

Return ONLY the encoded string in the format <sign><digits>#<check>, with no explanation."""
            }
        ]
    )
    
    result = message.content[0].text.strip()
    return result


if __name__ == "__main__":
    test_cases = [
        ('12.30', '+1230#6'),
        ('-0.07', '-7#7'),
        ('0.00', '+0#0'),
        ('1.00', '+100#1'),
        ('-5.50', '-550#10'),
    ]
    
    for input_amount, expected in test_cases:
        result = encode_amount(input_amount)
        print(f"Input: {input_amount:>10} -> Output: {result:>10} (Expected: {expected})")
