#!/usr/bin/env python3
"""
Store wire-encoded amounts to ledger using Claude API.
"""

import anthropic
import json


def store(acct: str, wire: str) -> str:
    """
    Post a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (already serialized, e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for an account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for this entry"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Store the wire-encoded amount '{wire}' to account '{acct}' with memo 'wire' using the ledger_write tool."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response to find the tool use
    entry_id = None
    while response.stop_reason == "tool_use":
        # Extract tool use from response
        tool_use = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use = block
                break
        
        if not tool_use:
            break
            
        # Simulate the tool response
        if tool_use.name == "ledger_write":
            # Return a simulated entry ID based on the input
            entry_id = f"entry_{acct}_{wire.replace('#', '_')}"
            
            # Continue the conversation with the tool result
            messages.append({"role": "assistant", "content": response.content})
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use.id,
                        "content": json.dumps({"entry_id": entry_id, "status": "success"})
                    }
                ]
            })
            
            # Get next response
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
    
    # If no entry_id was generated through tool use, create one
    if entry_id is None:
        entry_id = f"entry_{acct}_{wire.replace('#', '_')}"
    
    return entry_id


if __name__ == "__main__":
    # Test the store function
    test_account = "ACC123"
    test_wire = "+1230#6"
    
    result = store(test_account, test_wire)
    print(f"Entry ID: {result}")
