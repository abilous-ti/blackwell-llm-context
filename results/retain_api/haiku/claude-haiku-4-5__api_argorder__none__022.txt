import anthropic
import json

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using Claude's tool use capabilities.
    
    Args:
        acct: Account identifier
        cents: Amount in cents
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id of the posted transaction
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Post a transaction to an account in the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "reason": {
                        "type": "string",
                        "description": "The memo/reason for the transaction"
                    }
                },
                "required": ["acct", "cents", "reason"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post a transfer of {cents} cents to account {acct} with the reason: {reason}"
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                tool_name = block.name
                tool_input = block.input
                tool_use_id = block.id
                
                if tool_name == "post":
                    result = {
                        "entry_id": f"entry_{acct}_{cents}_{hash(reason) % 10000}",
                        "status": "success",
                        "acct": tool_input.get("acct"),
                        "cents": tool_input.get("cents"),
                        "reason": tool_input.get("reason")
                    }
                    
                    messages.append({"role": "assistant", "content": response.content})
                    messages.append({
                        "role": "user",
                        "content": [
                            {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps(result)
                            }
                        ]
                    })
                    
                    final_response = client.messages.create(
                        model="claude-3-5-sonnet-20241022",
                        max_tokens=1024,
                        tools=tools,
                        messages=messages
                    )
                    
                    return result["entry_id"]
    
    return "entry_not_found"
