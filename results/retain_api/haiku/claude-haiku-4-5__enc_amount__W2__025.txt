def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Args:
        dollars: A string representing dollars, e.g., '12.30', '-0.07'
    
    Returns:
        A string in the format <sign><digits>#<check>
    """
    # Determine sign
    if dollars.startswith('-'):
        sign = '-'
        dollars = dollars[1:]
    else:
        sign = '+'
        if dollars.startswith('+'):
            dollars = dollars[1:]
    
    # Remove the decimal point and convert to minor units (cents)
    if '.' in dollars:
        parts = dollars.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        integer_part = parts[0] if parts[0] else '0'
        decimal_part = parts[1]
        
        # Pad or truncate decimal part to exactly 2 digits
        if len(decimal_part) == 1:
            decimal_part = decimal_part + '0'
        elif len(decimal_part) > 2:
            decimal_part = decimal_part[:2]
        
        cents = int(integer_part) * 100 + int(decimal_part)
    else:
        # No decimal point, assume it's already in dollars
        cents = int(dollars) * 100
    
    # Convert cents to string (digits without sign or decimal)
    digits = str(cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format: <sign><digits>#<check>
    return f"{sign}{digits}#{check}"
