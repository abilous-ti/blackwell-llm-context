import anthropic
import json

ledger_functions = {
    "post": "Record a transaction posting cents to an account with a memo, returns the entry id"
}

def transfer(acct: str, cents: int, reason: str) -> str:
    """Transfer funds to account using Claude with tool use."""
    client = anthropic.Anthropic()
    
    # Create the tool definition for the ledger post function
    tools = [
        {
            "name": "post",
            "description": "Record a transaction posting cents to an account with a memo",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the transaction"
                    }
                },
                "required": ["account", "cents", "memo"]
            }
        }
    ]
    
    # Call Claude with tool use
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with the reason '{reason}' and tell me the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and extract tool use
    entry_id = None
    
    for block in response.content:
        if block.type == "tool_use":
            if block.name == "post":
                # Simulate the ledger post function
                tool_input = block.input
                # Return a simulated entry id based on the input
                entry_id = f"entry_{hash(str(tool_input)) % 10000}"
                
                # Continue the conversation with the tool result
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": entry_id
                        }
                    ]
                })
                
                # Get the final response from Claude
                final_response = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=tools,
                    messages=messages
                )
                
                # Extract entry id from the response
                for final_block in final_response.content:
                    if hasattr(final_block, 'text'):
                        # Try to extract entry id from text
                        text = final_block.text
                        if "entry_" in text:
                            # Extract the entry id from the response
                            import re
                            match = re.search(r'entry_\d+', text)
                            if match:
                                return match.group(0)
                
                return entry_id
    
    return entry_id
