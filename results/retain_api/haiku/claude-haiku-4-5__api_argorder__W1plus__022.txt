def transfer(acct, cents, reason):
    import ledger
    
    # Convert cents to the wire-encoding format
    # Format: <sign><digits>#<check>
    # where check = (sum of decimal digits in digits) mod 10
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate check digit as sum of all digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Construct the encoded amount string
    amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with account first, amount second, and memo as keyword-only arg
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
