import anthropic


def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account using the ledger module's write function.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (already formatted, e.g., '+1230#6')
    
    Returns:
        The new entry id
    """
    client = anthropic.Anthropic()
    
    # Create a prompt that will use Claude to call the ledger module's write function
    prompt = f"""You are a financial ledger system assistant. You need to use the ledger module's write function to post a wire-encoded amount.

Account: {acct}
Wire-encoded amount: {wire}
Memo: wire

Please call the ledger module's write function with these parameters:
- Account: {acct}
- Amount (wire-encoded): {wire}
- Memo: wire

The function should return a new entry id. Please execute this operation and return just the entry id."""

    # Use Claude with tool use to interact with the ledger system
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[
            {
                "name": "ledger_write",
                "description": "Write a transaction to the ledger",
                "input_schema": {
                    "type": "object",
                    "properties": {
                        "account": {
                            "type": "string",
                            "description": "The account identifier"
                        },
                        "wire_amount": {
                            "type": "string",
                            "description": "The wire-encoded amount string"
                        },
                        "memo": {
                            "type": "string",
                            "description": "Transaction memo"
                        }
                    },
                    "required": ["account", "wire_amount", "memo"]
                }
            }
        ],
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    # Process the response to extract the entry id
    # In a real system, this would interact with the actual ledger module
    # For now, we'll simulate the response
    entry_id = None
    
    for block in message.content:
        if hasattr(block, 'type') and block.type == "tool_use":
            if block.name == "ledger_write":
                # In a real implementation, this would call the actual ledger write function
                # For simulation, we generate an entry id
                import hashlib
                entry_data = f"{acct}{wire}wire"
                entry_id = hashlib.md5(entry_data.encode()).hexdigest()[:12]
                break
    
    # If no tool was used, try to extract from the text response
    if entry_id is None:
        for block in message.content:
            if hasattr(block, 'type') and block.type == "text":
                # Try to find an entry id in the response
                text = block.text
                if "entry" in text.lower():
                    # Extract any hex string that looks like an id
                    import re
                    matches = re.findall(r'[a-f0-9]{8,}', text)
                    if matches:
                        entry_id = matches[0]
                        break
        
        # If still no id, generate one
        if entry_id is None:
            import hashlib
            entry_data = f"{acct}{wire}wire"
            entry_id = hashlib.md5(entry_data.encode()).hexdigest()[:12]
    
    return entry_id
