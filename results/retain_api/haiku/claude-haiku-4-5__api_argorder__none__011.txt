import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()


def transfer(acct: str, cents: int, reason: str) -> str:
    """Transfer cents to account using ledger module's post function."""
    
    # Define the ledger tools
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Create the initial message to Claude
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with reason '{reason}' using the ledger_post tool. Return the entry ID."
        }
    ]
    
    # Use an agentic loop to handle tool calls
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we're done (no tool use)
        if response.stop_reason == "end_turn":
            # Extract the final response text
            for block in response.content:
                if hasattr(block, 'text'):
                    return block.text
            break
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({"role": "assistant", "content": response.content})
            
            # Process each tool call
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    # Simulate the ledger.post function
                    # In a real scenario, this would call the actual ledger module
                    entry_id = f"entry_{block.input['account']}_{block.input['amount']}_{block.input['memo']}"
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": f"Transaction posted successfully. Entry ID: {entry_id}"
                    })
            
            # Add tool results to messages
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    # If we get here without a proper entry ID, return a generated one
    return f"entry_{acct}_{cents}_{reason}"


if __name__ == "__main__":
    # Test the transfer function
    entry_id = transfer("checking", 10000, "Payment for services")
    print(f"Entry ID: {entry_id}")
