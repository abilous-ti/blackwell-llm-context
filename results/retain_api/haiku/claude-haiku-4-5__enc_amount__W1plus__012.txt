def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in MINOR UNITS (cents, no decimal point)
    - <check>: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the input
    dollars = dollars.strip()
    
    # Determine sign
    if dollars.startswith('-'):
        sign = '-'
        dollars = dollars[1:]
    elif dollars.startswith('+'):
        sign = '+'
        dollars = dollars[1:]
    else:
        sign = '+'
    
    # Remove leading zeros and handle edge cases
    dollars = dollars.strip()
    
    # Split by decimal point
    if '.' in dollars:
        parts = dollars.split('.')
        if len(parts) != 2:
            raise ValueError("Invalid amount format")
        integer_part = parts[0] if parts[0] else '0'
        decimal_part = parts[1]
        
        # Pad or truncate decimal to 2 digits
        if len(decimal_part) == 1:
            decimal_part = decimal_part + '0'
        elif len(decimal_part) > 2:
            decimal_part = decimal_part[:2]
        elif len(decimal_part) == 0:
            decimal_part = '00'
    else:
        integer_part = dollars
        decimal_part = '00'
    
    # Convert to cents (minor units)
    # Remove leading zeros from integer part for calculation
    integer_part = integer_part.lstrip('0') or '0'
    
    # Combine to get total cents as integer, then back to string
    total_cents = int(integer_part) * 100 + int(decimal_part)
    digits = str(total_cents)
    
    # Calculate check digit: sum of all digits mod 10
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    return f"{sign}{digits}#{check}"
