import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to a private wire format using Claude as an AI backbone.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are encoding a money amount to a private wire format. 
                
The money amount is: {dollars}

Please encode this amount to the team's private wire format. The format should:
1. Handle positive and negative amounts
2. Convert dollars to cents as an integer
3. Use a compact binary-like representation

Return ONLY the encoded string, nothing else."""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    test_amounts = ["12.30", "-0.07", "0.00", "100.00", "-999.99"]
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"encode_amount('{amount}') = '{encoded}'")
