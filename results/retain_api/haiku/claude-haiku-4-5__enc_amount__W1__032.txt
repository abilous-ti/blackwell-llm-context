def encode_amount(dollars: str) -> str:
    """
    Encode a dollar amount string to the team's private wire format.
    
    Args:
        dollars: A string representation of a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount as a string in the team's private wire format
    """
    try:
        amount = float(dollars)
    except (ValueError, TypeError):
        raise ValueError(f"Invalid amount format: {dollars}")
    
    cents = round(amount * 100)
    
    return str(cents)
