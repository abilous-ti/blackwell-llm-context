import anthropic
import json

ledger = __import__("ledger")


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds using the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in minor units (cents)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post_transfer",
            "description": "Post a transfer to the ledger with proper amount encoding",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "encoded_amount": {
                        "type": "string",
                        "description": "The amount in wire-encoding format: <sign><digits>#<check>"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The reason/memo for the transfer"
                    }
                },
                "required": ["account", "encoded_amount", "memo"]
            }
        }
    ]
    
    def encode_amount(amount_cents: int) -> str:
        """Encode amount in wire format: <sign><digits>#<check>"""
        if amount_cents >= 0:
            sign = '+'
            digits_str = str(amount_cents)
        else:
            sign = '-'
            digits_str = str(abs(amount_cents))
        
        check_sum = sum(int(d) for d in digits_str) % 10
        return f"{sign}{digits_str}#{check_sum}"
    
    encoded_amount = encode_amount(cents)
    
    messages = [
        {
            "role": "user",
            "content": f"""I need to post a transfer to the ledger with these details:
- Account: {acct}
- Amount (cents): {cents}
- Encoded amount: {encoded_amount}
- Reason: {reason}

Please use the post_transfer tool to post this transfer."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = next(
                (block for block in response.content if block.type == "tool_use"),
                None
            )
            
            if tool_use_block and tool_use_block.name == "post_transfer":
                tool_input = tool_use_block.input
                entry_id = ledger.post(
                    tool_input["account"],
                    tool_input["encoded_amount"],
                    tool_input["memo"]
                )
                
                messages.append({
                    "role": "assistant",
                    "content": response.content
                })
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": f"Transfer posted successfully. Entry ID: {entry_id}"
                        }
                    ]
                })
                
                return entry_id
        else:
            break
    
    raise RuntimeError("Failed to post transfer to ledger")
