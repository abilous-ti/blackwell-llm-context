
```python
import ledger


def record(acct, cents, note):
    """
    Posts a transaction to the ledger and returns the new entry id.
    
    Args:
        acct: The account to post to
        cents: The amount in cents to post
        note: The memo/note for the transaction
        
    Returns:
        The new entry id (>=1)
        
    Raises:
        ledger.PostError: If the post fails (unknown account, malformed amount, etc.)
    """
    entry_id = ledger.post(acct, cents, memo=note)
    return entry_id
```
