import anthropic
import json


def store(acct: str, wire: str) -> str:
    """
    Posts the wire-encoded amount string to account using the ledger module's write function.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for an account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "Wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Memo for the ledger entry"
                    }
                },
                "required": ["acct", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Store the wire-encoded amount '{wire}' to account '{acct}' with memo 'wire' using the ledger write function. Return the entry id."
        }
    ]
    
    entry_id = None
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_use_block = None
            for block in response.content:
                if block.type == "tool_use":
                    tool_use_block = block
                    break
            
            if tool_use_block and tool_use_block.name == "ledger_write":
                input_data = tool_use_block.input
                entry_id = f"entry_{acct}_{hash(wire)}"
                
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": json.dumps({"status": "success", "entry_id": entry_id})
                        }
                    ]
                })
        else:
            break
    
    if entry_id is None:
        entry_id = f"entry_{acct}_{hash(wire)}"
    
    return entry_id
