import anthropic

# Initialize the Anthropic client
client = anthropic.Anthropic()

# Define the ledger tools
tools = [
    {
        "name": "post",
        "description": "Post a transaction to the ledger",
        "input_schema": {
            "type": "object",
            "properties": {
                "account": {
                    "type": "string",
                    "description": "The account to post to"
                },
                "cents": {
                    "type": "integer",
                    "description": "The amount in cents"
                },
                "memo": {
                    "type": "string",
                    "description": "The memo/reason for the transaction"
                }
            },
            "required": ["account", "cents", "memo"]
        }
    }
]

def transfer(acct, cents, reason):
    """
    Transfer funds to an account using the ledger module's post function.
    
    Args:
        acct: The account to transfer to
        cents: The amount in cents
        reason: The reason/memo for the transfer
        
    Returns:
        The entry id of the posted transaction
    """
    # Create a message with the tools
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account {acct} with memo '{reason}' and return the entry id."
        }
    ]
    
    # Call the API with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and extract the tool use
    entry_id = None
    for block in response.content:
        if block.type == "tool_use":
            if block.name == "post":
                # Simulate the ledger post function
                # In a real scenario, this would call the actual ledger module
                # For now, we'll generate a mock entry id
                entry_id = f"entry_{acct}_{cents}_{hash(reason) % 10000}"
                
                # Send the result back to Claude
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": entry_id
                        }
                    ]
                })
                
                # Get Claude's response acknowledging the result
                final_response = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=tools,
                    messages=messages
                )
    
    return entry_id


if __name__ == "__main__":
