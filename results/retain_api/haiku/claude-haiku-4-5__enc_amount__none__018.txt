import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representation of a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The serialized amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial encoding expert. Your task is to encode money amounts into a specific private wire format.

The format rules are:
1. Convert the dollar amount to cents (integer)
2. Represent negative amounts with a leading minus sign
3. Pad the result to 12 digits with leading zeros for positive amounts
4. For negative amounts, use a leading minus sign followed by 11 digits of padding

Examples:
- '12.30' → '000000001230' (1230 cents, 12 digits)
- '-0.07' → '-000000000007' (negative 7 cents, minus sign + 11 digits)
- '100.00' → '000000010000' (10000 cents, 12 digits)
- '0.01' → '000000000001' (1 cent, 12 digits)

Please encode this amount: {dollars}

Respond with ONLY the encoded result, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = ['12.30', '-0.07', '100.00', '0.01', '-99.99', '1.5']
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount:>8} → {encoded}")
