```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo describing the transfer
    
    Returns:
        The entry id from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Encode the amount in wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the amount string
    amount_str = f"{sign}{digits_str}#{check}"
    
    # Post to ledger with required keyword-only memo argument
    entry_id = ledger.post(acct, amount_str, memo=reason)
    
    return entry_id
```
