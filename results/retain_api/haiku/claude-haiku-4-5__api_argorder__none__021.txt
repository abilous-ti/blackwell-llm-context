import anthropic
import json

client = anthropic.Anthropic()
MODEL_NAME = "claude-3-5-sonnet-20241022"

# Define the ledger module functions
ledger_functions = [
    {
        "name": "post",
        "description": "Posts a transaction to the ledger. Returns the entry id.",
        "input_schema": {
            "type": "object",
            "properties": {
                "account": {
                    "type": "string",
                    "description": "The account to post to"
                },
                "amount": {
                    "type": "integer",
                    "description": "The amount in cents"
                },
                "memo": {
                    "type": "string",
                    "description": "The memo for the transaction"
                }
            },
            "required": ["account", "amount", "memo"]
        }
    }
]

def process_tool_call(tool_name, tool_input):
    """Process tool calls and return results"""
    if tool_name == "post":
        # Simulate the ledger post function
        # In a real scenario, this would call the actual ledger module
        account = tool_input.get("account")
        amount = tool_input.get("amount")
        memo = tool_input.get("memo")
        
        # Generate a mock entry ID (in real implementation, this would come from the ledger)
        entry_id = f"entry_{account}_{amount}_{hash(memo) % 10000}"
        
        return {
            "success": True,
            "entry_id": entry_id,
            "account": account,
            "amount": amount,
            "memo": memo
        }
    
    return {"error": f"Unknown tool: {tool_name}"}

def transfer(acct, cents, reason):
    """
    Transfer funds using the ledger module's post function.
    
    Args:
        acct: Account to transfer to
        cents: Amount in cents
        reason: Reason/memo for the transfer
    
    Returns:
        The entry id from the ledger post function
    """
    
    # Create a prompt that will trigger the use of the post function
    prompt = f"""Please post a transaction to the ledger with the following details:
- Account: {acct}
- Amount: {cents} cents
- Memo/Reason: {reason}

Use the 'post' function from the ledger module to complete this transaction and return the entry id."""
    
    messages = [
        {"role": "user", "content": prompt}
    ]
    
    # Use an agentic loop to handle tool calls
    while True:
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=1024,
            tools=ledger_functions,
            messages=messages
        )
        
        # Check if we're done (no more tool calls)
        if response.stop_reason == "end_turn":
            # Extract the entry_id from the response
            for block in response.content:
                if hasattr(block, 'text'):
                    # Try to find entry_id in the response text
                    text = block.text
                    if "entry_id" in text or "entry" in text.lower():
                        # Parse the response to extract entry_id
                        import re
                        match = re.search(r'entry_[a-zA-Z0-9_]+', text)
                        if match:
                            return match.group(0)
            return None
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({"role": "assistant", "content": response.content})
            
            # Process each tool call
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    tool_result = process_tool_call(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(tool_result)
                    })
            
            # Add tool results to messages
            messages.append({"role": "user", "content": tool_results})
        else:
            # If no tool calls and not end_turn, something unexpected happened
            break
    
    return None


if __name__ == "__main__":
    # Test the transfer function
    entry_id = transfer("checking", 5000, "Monthly salary deposit")
    print(f"Transfer completed. Entry ID: {entry_id}")
