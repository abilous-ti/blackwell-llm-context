```python
import ledger


def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    
    digit_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digit_str)
    check_digit = digit_sum % 10
    
    amount_encoded = f"{sign}{abs_cents}#{check_digit}"
    
    entry_id = ledger.post(acct, amount_encoded, memo=reason)
    return entry_id
```
