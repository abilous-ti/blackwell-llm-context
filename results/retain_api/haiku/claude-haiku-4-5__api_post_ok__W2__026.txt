import ledger

def record(acct, cents, note):
    """
    Posts cents to account acct with memo note.
    Returns the new entry id.
    
    cents: integer amount in cents (can be negative)
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value digits
    digits_str = str(abs(cents))
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Construct wire-encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return entry id
    return ledger.post(acct, encoded_amount, note)
