import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        A string representing the encoded amount in the private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial encoding expert. Your task is to encode money amounts into a specific private wire format.

Given the money amount: {dollars}

Please encode this amount using the following wire format rules:
1. For positive amounts, prefix with '+'
2. For negative amounts, prefix with '-'
3. Remove the decimal point and pad with zeros to ensure exactly 10 digits after the sign
4. For example: '12.30' becomes '+0000001230' and '-0.07' becomes '-0000000007'

Return ONLY the encoded amount, nothing else."""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    # Test cases
    test_cases = ['12.30', '-0.07', '0.01', '999.99', '-100.00', '1234567.89']
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
