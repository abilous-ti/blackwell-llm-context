def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents, as integer)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the input
    dollars = dollars.strip()
    
    # Determine sign
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    elif dollars.startswith('+'):
        sign = '+'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars
    
    # Handle the decimal point
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollar_part = parts[0]
        cent_part = parts[1]
        
        # Pad or truncate cent part to 2 digits
        cent_part = (cent_part + '00')[:2]
    else:
        dollar_part = amount_str
        cent_part = '00'
    
    # Convert to cents (minor units)
    dollar_part = dollar_part or '0'
    cents = int(dollar_part) * 100 + int(cent_part)
    
    # Convert to string
    digits_str = str(cents)
    
    # Calculate checksum: sum of all digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    # Format the result
    return f"{sign}{digits_str}#{check}"
