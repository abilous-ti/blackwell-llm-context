def encode_amount(dollars: str) -> str:
    # Parse the input string to extract sign and amount
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    elif dollars.startswith('+'):
        sign = '+'
        amount_str = dollars[1:]
    else:
        sign = '+'
        amount_str = dollars
    
    # Remove the decimal point and convert to cents (minor units)
    if '.' in amount_str:
        parts = amount_str.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        integer_part = parts[0] if parts[0] else '0'
        decimal_part = parts[1]
        
        # Pad or truncate decimal part to 2 digits
        if len(decimal_part) == 1:
            decimal_part = decimal_part + '0'
        elif len(decimal_part) > 2:
            decimal_part = decimal_part[:2]
        
        cents = int(integer_part) * 100 + int(decimal_part)
    else:
        # No decimal point, treat as dollars
        cents = int(amount_str) * 100
    
    digits_str = str(cents)
    
    # Calculate check digit using Luhn-style mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    return f"{sign}{digits_str}#{check}"
