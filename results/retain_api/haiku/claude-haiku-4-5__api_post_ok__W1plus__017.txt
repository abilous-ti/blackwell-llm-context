import ledger

def record(acct, cents, note):
    # Convert cents (integer) to encoded amount string
    # Format: <sign><digits>#<check>
    # where check = (sum of decimal digits in <digits>) mod 10
    
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Create encoded amount
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with required keyword-only memo argument
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
