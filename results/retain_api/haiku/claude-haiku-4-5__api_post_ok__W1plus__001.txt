import ledger


def encode_amount(cents):
    """Encode an integer cent amount to wire format: <sign><digits>#<check>"""
    if cents >= 0:
        sign = '+'
        digits = str(cents)
    else:
        sign = '-'
        digits = str(-cents)
    
    check = sum(int(d) for d in digits) % 10
    return f"{sign}{digits}#{check}"


def record(acct, cents, note):
    """Post cents to account acct with memo note. Returns the new entry id."""
    encoded_amount = encode_amount(cents)
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    return entry_id
