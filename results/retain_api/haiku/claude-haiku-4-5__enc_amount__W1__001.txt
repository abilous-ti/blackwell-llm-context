def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount from dollars string format to the team's private wire format.
    
    The wire format represents amounts as integers (in cents) to avoid floating point issues.
    For example: '12.30' -> '1230', '-0.07' -> '-7', '0' -> '0'
    
    Args:
        dollars: A string representation of a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount as a string in the team's wire format (cents as integer)
    """
    # Remove any whitespace
    dollars = dollars.strip()
    
    # Handle empty string
    if not dollars:
        return '0'
    
    # Split into sign, integer, and decimal parts
    is_negative = dollars.startswith('-')
    if is_negative:
        dollars = dollars[1:]
    
    # Split on decimal point
    if '.' in dollars:
        parts = dollars.split('.')
        if len(parts) != 2:
            raise ValueError(f"Invalid amount format: {dollars}")
        integer_part, decimal_part = parts
    else:
        integer_part = dollars
        decimal_part = '0'
    
    # Handle empty integer part (e.g., '.50')
    if not integer_part:
        integer_part = '0'
    
    # Pad or truncate decimal part to exactly 2 digits
    decimal_part = decimal_part.ljust(2, '0')[:2]
    
    # Convert to cents
    cents = int(integer_part) * 100 + int(decimal_part)
    
    # Apply sign
    if is_negative:
        cents = -cents
    
    return str(cents)
