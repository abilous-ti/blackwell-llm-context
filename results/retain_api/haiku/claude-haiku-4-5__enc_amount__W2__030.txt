import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in MINOR UNITS (cents, no decimal point)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '-0.07' -> '-7#7'
    - '+12.30' -> '+1230#6'
    """
    client = anthropic.Anthropic()
    
    prompt = f"""You are a financial encoding expert. I need you to encode the monetary amount '{dollars}' into our team's private wire format.

The encoding format is: <sign><digits>#<check>
Where:
- <sign> is '+' for positive amounts or '-' for negative amounts
- <digits> is the amount in MINOR UNITS (cents as an integer with no decimal point)
- <check> is a single Luhn-style check digit calculated as (sum of all decimal digits in <digits>) mod 10

Examples:
- '-0.07' encodes to '-7#7' (7 cents negative, check digit = 7 mod 10 = 7)
- '+12.30' encodes to '+1230#6' (1230 cents positive, check digit = (1+2+3+0) mod 10 = 6)

Please encode '{dollars}' using this format and return ONLY the encoded string, nothing else."""
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    result = message.content[0].text.strip()
    return result


if __name__ == "__main__":
    test_cases = [
        "12.30",
        "-0.07",
        "0.00",
        "100.00",
        "-50.25",
        "1.01",
        "-1.01"
    ]
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount:>10} -> {encoded}")
