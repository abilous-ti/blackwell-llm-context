
```python
import anthropic
import ledger


def transfer(acct: str, cents: int, reason: str) -> int:
    """
    Transfer funds using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
        
    Raises:
        ledger.PostError: If the post operation fails
    """
    entry_id = ledger.post(acct, cents, memo=reason)
    return entry_id
```
