import anthropic
import json

client = anthropic.Anthropic()

def post(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "A memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account {acct} with memo '{note}' using the ledger_post tool and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process tool use
    while response.stop_reason == "tool_use":
        tool_use = next(
            (block for block in response.content if block.type == "tool_use"),
            None
        )
        
        if tool_use and tool_use.name == "ledger_post":
            # Simulate the ledger.post call
            entry_id = f"entry_{acct}_{abs(hash((acct, cents, note))) % 10000}"
            
            # Continue the conversation with the tool result
            messages = [
                {"role": "user", "content": f"Post {cents} cents to account {acct} with memo '{note}' using the ledger_post tool and return the entry id."},
                {"role": "assistant", "content": response.content},
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use.id,
                            "content": json.dumps({"entry_id": entry_id, "success": True})
                        }
                    ]
                }
            ]
            
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
        else:
            break
    
    # Extract the entry id from the response
    text_content = next(
        (block.text for block in response.content if hasattr(block, "text")),
        None
    )
    
    if text_content and "entry_" in text_content:
        # Extract entry id from the response
        import re
        match = re.search(r'entry_\w+_\d+', text_content)
        if match:
            return match.group(0)
    
    # Fallback: generate a consistent entry id
    entry_id = f"entry_{acct}_{abs(hash((acct, cents, note))) % 10000}"
    return entry_id
