import anthropic
import json


def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()

    # Define the ledger.post function for Claude to use
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger. Returns the new entry id.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account identifier",
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post",
                    },
                    "note": {
                        "type": "string",
                        "description": "The memo/note for the transaction",
                    },
                },
                "required": ["acct", "cents", "note"],
            },
        }
    ]

    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with note '{note}' and return the entry id.",
        }
    ]

    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages,
    )

    # Process the response and handle tool calls
    while response.stop_reason == "tool_use":
        # Find the tool use block
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break

        if tool_use_block and tool_use_block.name == "ledger_post":
            # Extract the entry id from the tool input
            # In a real scenario, ledger.post would be called here
            # For now, we'll simulate it by using Claude to generate an entry id
            entry_id = f"entry_{hash((acct, cents, note)) % 1000000:06d}"

            # Add the assistant's response and tool result to messages
            messages.append({"role": "assistant", "content": response.content})

            messages.append(
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": json.dumps(
                                {"success": True, "entry_id": entry_id}
                            ),
                        }
                    ],
                }
            )

            # Call Claude again to process the result
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages,
            )
        else:
            break

    # Extract the entry id from Claude's final response
    for block in response.content:
        if hasattr(block, "text"):
            text = block.text
            # Try to extract entry_id from the response
            if "entry_" in text:
                # Extract the entry id pattern
                import re

                match = re.search(r"entry_\d+", text)
                if match:
                    return match.group()

    # Fallback: generate entry id based on inputs
    return f"entry_{hash((acct, cents, note)) % 1000000:06d}"
