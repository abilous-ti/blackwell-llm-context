def encode_amount(dollars: str) -> str:
    # Parse the input string to handle sign, integer, and decimal parts
    sign = '+' if not dollars.startswith('-') else '-'
    
    # Remove the sign for processing
    amount_str = dollars.lstrip('-+')
    
    # Split by decimal point
    if '.' in amount_str:
        integer_part, decimal_part = amount_str.split('.')
    else:
        integer_part = amount_str
        decimal_part = ''
    
    # Convert to cents (minor units)
    integer_part = integer_part or '0'
    decimal_part = (decimal_part + '00')[:2]  # Pad or truncate to 2 digits
    
    cents = int(integer_part) * 100 + int(decimal_part)
    
    # Convert cents to string of digits
    digits_str = str(cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the result
    return f"{sign}{digits_str}#{check}"
