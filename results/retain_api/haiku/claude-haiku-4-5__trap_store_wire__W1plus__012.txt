```python
import ledger

def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account identifier (string)
        wire: Wire-encoded amount string (already in format <sign><digits>#<check>)
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: On any failure (unknown account, malformed amount)
    """
    return ledger.post(acct, wire, memo='wire')
```
