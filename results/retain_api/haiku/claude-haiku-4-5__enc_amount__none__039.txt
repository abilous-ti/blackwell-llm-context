import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial encoding expert. Your task is to encode money amounts into a team's private wire format.

The encoding rules for the private wire format are:
1. Convert the dollar amount to cents (multiply by 100)
2. If the amount is negative, use a 9 prefix; if positive or zero, use a 0 prefix
3. Pad the absolute value of cents to 10 digits with leading zeros
4. The format is: [sign_prefix][10_digit_padded_cents]

For example:
- '12.30' → 1230 cents → positive → '0' + '0000001230' = '00000001230'
- '-0.07' → -7 cents → negative → '9' + '0000000007' = '90000000007'
- '0.00' → 0 cents → zero (treated as positive) → '0' + '0000000000' = '00000000000'
- '-1.00' → -100 cents → negative → '9' + '0000000100' = '90000000100'

Encode this amount: {dollars}

Return ONLY the encoded wire format string, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_amounts = ["12.30", "-0.07", "0.00", "-1.00", "999.99", "-999.99"]
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"{amount:>8} → {encoded}")
