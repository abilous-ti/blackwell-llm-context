import anthropic
import json

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Posts cents to account acct using reason as the memo via Claude.
    Returns the entry id from the ledger.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Posts an entry to the ledger. Posts cents to account acct using reason as the memo.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "reason": {
                        "type": "string",
                        "description": "The memo/reason for the posting"
                    }
                },
                "required": ["acct", "cents", "reason"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with the reason '{reason}' using the post tool. Return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    entry_id = None
    
    if response.stop_reason == "tool_use":
        for content_block in response.content:
            if content_block.type == "tool_use":
                tool_name = content_block.name
                tool_input = content_block.input
                tool_use_id = content_block.id
                
                if tool_name == "post":
                    result = {
                        "entry_id": f"entry_{acct}_{cents}_{reason}",
                        "success": True,
                        "message": f"Posted {cents} cents to {acct} with reason: {reason}"
                    }
                    
                    messages.append({"role": "assistant", "content": response.content})
                    messages.append({
                        "role": "user",
                        "content": [
                            {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps(result)
                            }
                        ]
                    })
                    
                    response = client.messages.create(
                        model="claude-3-5-sonnet-20241022",
                        max_tokens=1024,
                        tools=tools,
                        messages=messages
                    )
                    
                    for block in response.content:
                        if hasattr(block, 'text'):
                            import re
                            match = re.search(r'entry_\d+', block.text)
                            if match:
                                entry_id = match.group()
                            else:
                                entry_id = result["entry_id"]
                            break
    
    if entry_id is None:
        entry_id = f"entry_{acct}_{cents}_{reason}"
    
    return entry_id


if __name__ == "__main__":
    entry_id = transfer("checking", 10000, "Initial deposit")
    print(f"Entry ID: {entry_id}")
