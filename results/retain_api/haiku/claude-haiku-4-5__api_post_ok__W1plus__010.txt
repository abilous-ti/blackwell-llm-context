import ledger

def record(acct, cents, note):
    """Post a monetary amount to an account and return the entry id.
    
    Args:
        acct: Account identifier string
        cents: Amount in minor units (integer cents)
        note: Memo string for the transaction
    
    Returns:
        The new entry id (>= 1) from ledger.post
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Encode cents as wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with memo as keyword-only argument
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
