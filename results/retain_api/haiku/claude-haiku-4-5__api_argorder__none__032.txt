import anthropic
import json

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account {acct} with reason: {reason}"
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = next(
            (block for block in response.content if block.type == "tool_use"),
            None
        )
        
        if tool_use_block is None:
            break
            
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        if tool_name == "post":
            entry_id = f"entry_{tool_input['account']}_{tool_input['amount']}"
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use_block.id,
                        "content": json.dumps({"entry_id": entry_id})
                    }
                ]
            })
            
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
    
    return entry_id
