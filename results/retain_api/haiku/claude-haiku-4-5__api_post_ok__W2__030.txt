import ledger


def record(acct, cents, note):
    """Posts a monetary amount to an account and returns the new entry id."""
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value in cents (minor units)
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Encode as <sign><digits>#<check>
    encoded = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger and return entry id
    return ledger.post(acct, encoded, note)
