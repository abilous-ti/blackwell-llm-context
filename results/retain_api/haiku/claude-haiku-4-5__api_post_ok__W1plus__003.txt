import ledger

def record(acct, cents, note):
    """
    Post a monetary amount to a ledger account.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        note: Memo text (string)
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Encode cents to wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger with memo as keyword-only argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
