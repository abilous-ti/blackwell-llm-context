import anthropic
import ledger

def store(acct: str, wire: str) -> int:
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger
    
    Raises:
        ledger.PostError: If posting fails
    """
    client = anthropic.Anthropic()
    
    # Define the ledger.post tool for Claude
    tools = [
        {
            "name": "post_to_ledger",
            "description": "Posts a transaction to the ledger system",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Create a message asking Claude to post to the ledger
    messages = [
        {
            "role": "user",
            "content": f"Please post wire-encoded amount '{wire}' to account '{acct}' with memo 'wire' using the ledger posting system."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response and handle tool calls
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if tool_use_block and tool_use_block.name == "post_to_ledger":
            input_data = tool_use_block.input
            try:
                # Call the actual ledger.post function with the correct signature
                entry_id = ledger.post(
                    input_data["account"],
                    input_data["amount"],
                    memo=input_data["memo"]
                )
                
                # Send the result back to Claude
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": f"Successfully posted. Entry ID: {entry_id}"
                        }
                    ]
                })
                
                # Get Claude's acknowledgment
                response = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=tools,
                    messages=messages
                )
                
                return entry_id
                
            except ledger.PostError as e:
                # Send error back to Claude
                messages.append({"role": "assistant", "content": response.content})
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": f"Error: {str(e)}",
                            "is_error": True
                        }
                    ]
                })
                
                # Get Claude's response to the error
                response = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=tools,
                    messages=messages
                )
                
                raise
        else:
            break
    
    # If we get here without returning, raise an error
    raise ledger.PostError("Failed to post to ledger")
