import anthropic
import json
import re

client = anthropic.Anthropic()


def store(acct: str, wire: str) -> str:
    """
    Posts a wire-encoded amount string to an account using the ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger
    """
    
    # Define the tools available to Claude
    tools = [
        {
            "name": "ledger_write",
            "description": "Write a ledger entry to the specified account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for this entry"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": f"Store the wire-encoded amount '{wire}' to account '{acct}' with memo 'wire' using the ledger_write tool and return the entry id."
        }
    ]
    
    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    entry_id = None
    while response.stop_reason == "tool_use":
        # Find the tool use block
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        # Extract the parameters
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        # Simulate the ledger_write function
        if tool_name == "ledger_write":
            # In a real implementation, this would call the actual ledger module
            # For now, we'll generate a mock entry id
            entry_id = f"entry_{hash(tool_input['account'] + tool_input['amount']) % 1000000}"
            
            # Add the tool result to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use_block.id,
                        "content": json.dumps({"entry_id": entry_id, "status": "success"})
                    }
                ]
            })
            
            # Continue the conversation
            response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
        else:
            break
    
    # Extract the entry id from the final response if not already set
    if not entry_id:
        # Look for entry_id in the response text
        for block in response.content:
            if hasattr(block, 'text'):
                # Try to extract entry_id from text
                match = re.search(r'entry[_\s]?id[:\s]+([a-zA-Z0-9_-]+)', block.text, re.IGNORECASE)
                if match:
                    entry_id = match.group(1)
                    break
    
    return entry_id or "entry_unknown"
