import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()

def store(acct: str, wire: str) -> str:
    """
    Posts a wire-encoded amount string to an account using Claude to interact with a ledger module.
    
    Args:
        acct: Account identifier
        wire: Wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger
    """
    
    # Define the tools available
    tools = [
        {
            "name": "ledger_write",
            "description": "Write an entry to the ledger for an account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The wire-encoded amount string"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for this entry"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Create the initial message to Claude
    messages = [
        {
            "role": "user",
            "content": f"Please write a ledger entry for account '{acct}' with wire-encoded amount '{wire}' and memo 'wire'. Use the ledger_write tool to accomplish this."
        }
    ]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract the final response
            for block in response.content:
                if hasattr(block, 'text'):
                    return block.text
            return ""
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            # Find the tool use block
            tool_use_block = None
            for block in response.content:
                if block.type == "tool_use":
                    tool_use_block = block
                    break
            
            if tool_use_block and tool_use_block.name == "ledger_write":
                # Simulate the ledger write operation
                input_data = tool_use_block.input
                # Generate a mock entry id
                entry_id = f"entry_{acct}_{hash(wire) % 10000}"
                
                # Add the assistant's response and tool result to messages
                messages.append({
                    "role": "assistant",
                    "content": response.content
                })
                
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": json.dumps({"status": "success", "entry_id": entry_id})
                        }
                    ]
                })
                
                # Continue the loop to get the final response
                continue
        
        # If we get here without processing a tool, break to avoid infinite loop
        break
    
    return ""
