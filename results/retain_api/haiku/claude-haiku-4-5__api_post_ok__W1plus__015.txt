import ledger


def record(acct, cents, note):
    """Post cents to account acct with memo note using ledger.post.
    
    Args:
        acct: account identifier
        cents: amount in cents (integer)
        note: memo text
    
    Returns:
        The new entry id from ledger.post
    
    Raises:
        ledger.PostError: if posting fails
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    return ledger.post(acct, amount, memo=note)
