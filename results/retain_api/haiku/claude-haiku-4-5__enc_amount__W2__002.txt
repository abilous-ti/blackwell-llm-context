import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in MINOR UNITS (integer cents, no decimal point)
    - check: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - -$0.07 -> "-7#7"
    - +$12.30 -> "+1230#6"
    """
    
    client = anthropic.Anthropic()
    
    prompt = f"""You are a precise encoder for a financial wire format. Given a monetary amount as a string, you must encode it according to these exact rules:

Format: <sign><digits>#<check>
- <sign>: '+' for positive amounts, '-' for negative amounts
- <digits>: the absolute value of the amount in MINOR UNITS (cents as an integer, no decimal point)
- <check>: a single digit equal to (sum of all decimal digits in <digits>) mod 10

Examples:
- Input: "12.30" → Output: "+1230#6" (because 1+2+3+0=6, and 6 mod 10 = 6)
- Input: "-0.07" → Output: "-7#7" (because 7 mod 10 = 7)
- Input: "100.00" → Output: "+10000#1" (because 1+0+0+0+0=1, and 1 mod 10 = 1)

Rules:
1. Strip any dollar signs or extra whitespace
2. Convert to absolute value in cents (multiply by 100 if needed)
3. Calculate the check digit as the sum of all digits in the cent amount, mod 10
4. Preserve the sign from the original input

Input amount: "{dollars}"

Respond with ONLY the encoded string in the exact format specified, nothing else."""

    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=100,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return message.content[0].text.strip()


def manual_encode_amount(dollars: str) -> str:
    """
    Manual implementation of the encode_amount function.
    This version doesn't use Claude, but implements the logic directly.
    """
    
    dollars = dollars.strip()
    
    is_negative = dollars.startswith('-')
    amount_str = dollars.lstrip('+-$')
    
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] or '0'
        cents_part = parts[1][:2].ljust(2, '0')
        cents = int(dollars_part) * 100 + int(cents_part)
    else:
        cents = int(amount_str) * 100
    
    cents_str = str(cents)
    
    digit_sum = sum(int(d) for d in cents_str)
    check_digit = digit_sum % 10
    
    sign = '-' if is_negative else '+'
    encoded = f"{sign}{cents_str}#{check_digit}"
    
    return encoded


if __name__ == "__main__":

        except Exception as e:

    print("\nTesting manual_encode_amount function (no API calls):")
