```python
import ledger


def transfer(acct, cents, reason):
    """Transfer cents to account acct with reason as memo.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo describing the transfer reason
        
    Returns:
        The new entry id (>=1) on success
        
    Raises:
        ledger.PostError: On any failure (unknown account, malformed amount)
    """
    return ledger.post(acct, cents, memo=reason)
```
