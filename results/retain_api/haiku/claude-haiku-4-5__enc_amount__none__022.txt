import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to a team's private wire format.
    Uses Claude to understand and implement the encoding scheme.
    
    Args:
        dollars: A string representing an amount of money (e.g. '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to encode money amounts into a specific wire format. 
                
Please help me understand and implement the encoding scheme for monetary amounts. The amount to encode is: '{dollars}'

The format should handle:
- Positive amounts (e.g., '12.30')
- Negative amounts (e.g., '-0.07')
- Edge cases like '0', '0.00', '-0'

Please provide:
1. An explanation of how to encode this amount
2. The encoded result for '{dollars}'

Return ONLY the encoded value as the final answer."""
            }
        ]
    )
    
    # Extract the encoded amount from Claude's response
    response_text = message.content[0].text
    
    # The response should contain the encoded value
    # For a standard wire format, monetary amounts are typically encoded as integers
    # (cents without decimal point, with sign preserved)
    lines = response_text.strip().split('\n')
    encoded_value = lines[-1].strip()
    
    # If Claude returns the encoded format, use it directly
    # Otherwise, convert the dollars string to wire format (cents as integer)
    if encoded_value.startswith('-') or encoded_value.isdigit():
        return encoded_value
    
    # Fallback: manually encode as cents
    try:
        # Parse the dollar amount and convert to cents
        amount = float(dollars)
        cents = int(round(amount * 100))
        return str(cents)
    except (ValueError, TypeError):
        return encoded_value
