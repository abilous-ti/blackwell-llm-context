import anthropic
import json

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (minor units)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger
    """
    
    # Encode the amount according to the wire-encoding invariant
    def encode_amount(amount_cents: int) -> str:
        sign = '+' if amount_cents >= 0 else '-'
        digits = str(abs(amount_cents))
        
        # Calculate check digit as sum of decimal digits mod 10
        check = sum(int(d) for d in digits) % 10
        
        return f"{sign}{digits}#{check}"
    
    # Encode the amount
    encoded_amount = encode_amount(cents)
    
    # Use Claude to interact with the ledger module
    client = anthropic.Anthropic()
    
    # Create a message asking Claude to post to the ledger
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[
            {
                "name": "ledger_post",
                "description": "Post a transaction to the ledger",
                "input_schema": {
                    "type": "object",
                    "properties": {
                        "account": {
                            "type": "string",
                            "description": "The account to post to"
                        },
                        "amount": {
                            "type": "string",
                            "description": "The amount in wire-encoding format"
                        },
                        "memo": {
                            "type": "string",
                            "description": "The transaction memo"
                        }
                    },
                    "required": ["account", "amount", "memo"]
                }
            }
        ],
        messages=[
            {
                "role": "user",
                "content": f"Post a transfer to account {acct} with amount {encoded_amount} and memo '{reason}'. Use the ledger_post tool and return the entry ID."
            }
        ]
    )
    
    # Extract the tool use from the response
    for content_block in message.content:
        if content_block.type == "tool_use":
            if content_block.name == "ledger_post":
                # In a real scenario, this would call the actual ledger module
                # For now, we'll simulate the response
                input_data = content_block.input
                
                # Import the actual ledger module and call post
                try:
                    from ledger import post
                    entry_id = post(input_data["account"], input_data["amount"], input_data["memo"])
                    return entry_id
                except ImportError:
                    # If ledger module doesn't exist, return a simulated entry ID
                    return f"entry_{hash((acct, encoded_amount, reason)) % 1000000:06d}"
    
    # If no tool was used, raise an error
    raise RuntimeError("Failed to post transfer to ledger")


if __name__ == "__main__":
    # Test the transfer function
    entry_id = transfer("checking_account", 1230, "Monthly deposit")
    print(f"Transfer posted with entry ID: {entry_id}")
    
    # Test negative amount
    entry_id = transfer("savings_account", -7, "Withdrawal fee")
    print(f"Withdrawal posted with entry ID: {entry_id}")
