import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representing a dollar amount (e.g. '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a wire format encoder for a financial team. Your job is to encode money amounts into the team's private wire format.

The wire format rules are:
1. Convert the dollar amount to cents (integer)
2. Encode the sign: use 'P' for positive, 'N' for negative
3. Pad the absolute value with leading zeros to make it 10 digits
4. Format as: [sign][10-digit padded cents]

Examples:
- '12.30' -> 'P0000001230' (1230 cents, positive)
- '-0.07' -> 'N0000000007' (7 cents, negative)
- '0.00' -> 'P0000000000' (0 cents, positive)
- '999.99' -> 'P0000099999' (99999 cents, positive)
- '-1234.56' -> 'N0000123456' (123456 cents, negative)

Encode this amount: {dollars}

Return ONLY the encoded wire format string, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_amounts = ['12.30', '-0.07', '0.00', '999.99', '-1234.56', '1.01', '-0.01']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
