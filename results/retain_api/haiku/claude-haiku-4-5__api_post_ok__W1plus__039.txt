import ledger

def record(acct, cents, note):
    """Post a ledger entry and return the new entry id."""
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    amount_str = f"{sign}{abs_cents}#{check}"
    return ledger.post(acct, amount_str, memo=note)
