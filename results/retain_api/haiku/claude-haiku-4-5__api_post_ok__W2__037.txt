import ledger

def record(acct, cents, note):
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Construct encoded amount
    encoded = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    return ledger.post(acct, encoded, note)
