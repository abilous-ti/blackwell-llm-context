import ledger


def record(acct, cents, note):
    """
    Posts a monetary amount to an account and returns the new entry id.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer, can be negative)
        note: Memo/description for the transaction
    
    Returns:
        The new entry id from ledger.post
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Build encoded amount string
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = ledger.post(acct, encoded_amount, note)
    return entry_id
