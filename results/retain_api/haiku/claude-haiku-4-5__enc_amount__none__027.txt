import anthropic


def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representation of a money amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the team's private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are an expert at encoding money amounts into a private wire format. 
                
Given a money amount as a string like '12.30', '-0.07', '0.01', etc., encode it to our team's private wire format.

The wire format follows these rules:
1. Remove the decimal point
2. Pad with leading zeros to make it 10 digits
3. If negative, prefix with a minus sign

Examples:
- '12.30' -> '0000001230'
- '-0.07' -> '-0000000007'
- '0.01' -> '0000000001'
- '1234.56' -> '0000123456'
- '-999.99' -> '-0000099999'

Encode this amount: {dollars}

Respond with ONLY the encoded value, nothing else."""
            }
        ]
    )
    
    return message.content[0].text.strip()


if __name__ == "__main__":
    test_cases = ['12.30', '-0.07', '0.01', '1234.56', '-999.99', '0.00', '-0.01']
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"encode_amount('{amount}') = '{encoded}'")
