def encode_amount(dollars: str) -> str:
    """
    Encode a dollar amount to the team's wire format.
    
    Format: <sign><digits>#<check>
    - sign: '+' or '-'
    - digits: amount in minor units (cents) as integer
    - check: Luhn-style digit = (sum of digits) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the dollar amount
    is_negative = dollars.startswith('-')
    amount_str = dollars.lstrip('+-')
    
    # Split into dollars and cents
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollar_part = parts[0] if parts[0] else '0'
        cent_part = parts[1]
    else:
        dollar_part = amount_str
        cent_part = '00'
    
    # Pad cents to 2 digits
    cent_part = cent_part.ljust(2, '0')[:2]
    
    # Convert to minor units (cents)
    dollars_int = int(dollar_part)
    cents_int = int(cent_part)
    total_cents = dollars_int * 100 + cents_int
    
    # Create the digits string
    digits = str(total_cents)
    
    # Calculate check digit (sum of all digits mod 10)
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    # Construct the final string
    sign = '-' if is_negative else '+'
    return f"{sign}{digits}#{check}"
