def transfer(acct, cents, reason):
    import ledger
    
    # Convert cents to the wire-encoding format
    # Calculate check digit as sum of digits mod 10
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with required signature
    entry_id = ledger.post(acct, amount, memo=reason)
    return entry_id
