import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: A string representing a money amount (e.g. '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a wire format encoder for a financial system. Your task is to encode money amounts into a specific binary wire format.

The wire format for money amounts works as follows:
- Amounts are encoded as a variable-length integer representation
- Positive amounts: use normal integer encoding
- Negative amounts: use two's complement representation
- The amount should be in cents (multiply dollars by 100)
- Use a minimal byte representation (use as few bytes as needed)
- Encode as hexadecimal string with pairs of hex digits

For example:
- '1.00' (100 cents) → '64'
- '0.01' (1 cent) → '01'
- '-0.01' (-1 cent) → 'ff'
- '12.30' (1230 cents) → '04c6'
- '-0.07' (-7 cents) → 'f9'

Please encode this amount: {dollars}

Respond with ONLY the hexadecimal encoded value, nothing else."""
            }
        ]
    )
    
    # Extract the encoded value from Claude's response
    encoded_value = message.content[0].text.strip()
    
    return encoded_value


if __name__ == "__main__":
    # Test cases
    test_amounts = ['12.30', '-0.07', '1.00', '0.01', '-0.01', '100.00', '-100.00']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"encode_amount('{amount}') = '{encoded}'")
