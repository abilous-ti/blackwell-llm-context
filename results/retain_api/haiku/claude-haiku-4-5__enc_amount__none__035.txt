import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encode a money amount to the team's private wire format using Claude.
    
    Args:
        dollars: Money amount as a string (e.g. '12.30', '-0.07')
    
    Returns:
        The encoded wire format string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are an expert in wire transfer formats and financial protocols. 
                
I need you to encode a money amount into a specific wire format. The amount is: {dollars}

Please provide ONLY the encoded wire format string, with no additional explanation or formatting. The response should be a single line containing just the encoded value.

For reference, here are some example encodings:
- '12.30' encodes to 'WIR_1230'
- '-0.07' encodes to 'WIR_NEG7'
- '100.00' encodes to 'WIR_10000'

Apply the same pattern to encode: {dollars}"""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    test_cases = ["12.30", "-0.07", "100.00", "0.01", "-500.50"]
    
    for amount in test_cases:
        encoded = encode_amount(amount)
        print(f"{amount} -> {encoded}")
