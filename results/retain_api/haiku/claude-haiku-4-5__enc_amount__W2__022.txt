def encode_amount(dollars: str) -> str:
    """
    Encode a money amount in dollars to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in minor units (cents, integer, no decimal point)
    - <check>: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Handle the sign
    is_negative = dollars.startswith('-')
    amount_str = dollars.lstrip('+-')
    
    # Parse dollars and cents
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollar_part = parts[0] if parts[0] else '0'
        cent_part = parts[1]
        # Pad or truncate cents to 2 digits
        cent_part = (cent_part + '00')[:2]
    else:
        dollar_part = amount_str
        cent_part = '00'
    
    # Convert to minor units (cents)
    cents = int(dollar_part) * 100 + int(cent_part)
    
    # Create the digits string
    digits_str = str(cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Create the sign
    sign = '-' if is_negative else '+'
    
    # Format the result
    return f"{sign}{digits_str}#{check_digit}"
