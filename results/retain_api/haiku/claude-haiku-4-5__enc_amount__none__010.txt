import anthropic


def encode_amount(dollars: str) -> str:
    """
    Encodes a money amount to a private wire format using Claude.
    
    Args:
        dollars: A string representing a dollar amount (e.g., '12.30', '-0.07')
    
    Returns:
        The encoded amount in the private wire format
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a financial data encoding expert. Your task is to encode a money amount into a private wire format used by this team.

The amount to encode is: {dollars}

Please provide ONLY the encoded wire format string with no explanation. The format should handle:
- Positive and negative amounts
- Decimal precision
- Standard wire transfer encoding conventions

Return only the encoded value, nothing else."""
            }
        ]
    )
    
    encoded = message.content[0].text.strip()
    return encoded


if __name__ == "__main__":
    test_amounts = ['12.30', '-0.07', '100.00', '-999.99', '0.01']
    
    for amount in test_amounts:
        encoded = encode_amount(amount)
        print(f"encode_amount('{amount}') = '{encoded}'")
