def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier
        cents: Amount in cents
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
        
    Raises:
        ledger.PostError: If the post operation fails
    """
    import ledger
    return ledger.post(acct, cents, memo=reason)
