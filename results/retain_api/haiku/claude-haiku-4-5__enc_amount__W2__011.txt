def encode_amount(dollars: str) -> str:
    """
    Encode a monetary amount to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in minor units (cents) as integer
    - <check>: Luhn-style check digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the input
    amount_str = dollars.strip()
    
    # Determine sign
    if amount_str.startswith('-'):
        sign = '-'
        amount_str = amount_str[1:]
    else:
        sign = '+'
        if amount_str.startswith('+'):
            amount_str = amount_str[1:]
    
    # Handle the decimal point to get cents
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0]
        cents_part = parts[1].ljust(2, '0')[:2]  # Pad or truncate to 2 digits
        cents = int(dollars_part) * 100 + int(cents_part)
    else:
        # No decimal point, treat as dollars
        cents = int(amount_str) * 100
    
    # Convert to minor units (cents) as string
    digits_str = str(cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Return encoded format
    return f"{sign}{digits_str}#{check_digit}"

    # Test cases from the problem
    print(encode_amount('12.30'))  # Should output: +1230#6
    print(encode_amount('-0.07'))  # Should output: -7#7
    print(encode_amount('0.01'))   # Should output: +1#1
    print(encode_amount('100.00')) # Should output: +10000#1
