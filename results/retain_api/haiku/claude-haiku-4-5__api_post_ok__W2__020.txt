import ledger

def record(acct, cents, note):
    """Posts cents to account acct with memo note.
    
    Encodes cents in wire format: <sign><digits>#<check>
    where check = (sum of decimal digits in digits) mod 10
    
    Returns the new entry id from ledger.post
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value for digit processing
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    # Construct wire-encoded amount
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    return ledger.post(acct, encoded_amount, note)
