def store(acct, wire):
    """
    Posts a wire-encoded amount string to an account in the ledger.
    
    Args:
        acct: The account identifier
        wire: The wire-encoded amount string (e.g., '+1230#6')
    
    Returns:
        The new entry id from the ledger write operation
    """
    from ledger import write
    
    entry_id = write(acct, wire, 'wire')
    return entry_id
