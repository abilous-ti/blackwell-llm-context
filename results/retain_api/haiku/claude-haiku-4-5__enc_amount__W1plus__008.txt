def encode_amount(dollars: str) -> str:
    """
    Encodes a monetary amount in dollars to the team's private wire format.
    
    Format: <sign><digits>#<check>
    - <sign>: '+' or '-'
    - <digits>: amount in minor units (cents), no decimal point
    - <check>: Luhn-style digit = (sum of decimal digits in <digits>) mod 10
    
    Examples:
    - '12.30' -> '+1230#6'
    - '-0.07' -> '-7#7'
    """
    # Parse the dollar amount
    if dollars.startswith('-'):
        sign = '-'
        amount_str = dollars[1:]
    else:
        sign = '+' if not dollars.startswith('+') else '+'
        amount_str = dollars.lstrip('+')
    
    # Handle the decimal point
    if '.' in amount_str:
        parts = amount_str.split('.')
        dollars_part = parts[0] if parts[0] else '0'
        cents_part = parts[1] if len(parts) > 1 else '0'
        
        # Pad or truncate cents to 2 digits
        cents_part = (cents_part + '00')[:2]
    else:
        dollars_part = amount_str if amount_str else '0'
        cents_part = '00'
    
    # Convert to minor units (cents)
    minor_units = int(dollars_part) * 100 + int(cents_part)
    digits = str(minor_units)
    
    # Calculate check digit (sum of all digits mod 10)
    check_digit = sum(int(d) for d in digits) % 10
    
    # Return encoded amount
    return f"{sign}{digits}#{check_digit}"
