import ledger

def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id.
    
    Args:
        acct: account name (str)
        cents: amount in minor units (integer cents)
        note: memo text (str)
    
    Returns:
        The new entry id (int >= 1)
    
    Raises:
        ledger.PostError: if account is unknown or amount is malformed
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digit_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digit_str)
    check = digit_sum % 10
    amount = f"{sign}{abs_cents}#{check}"
    
    entry_id = ledger.post(acct, amount, memo=note)
    return entry_id
