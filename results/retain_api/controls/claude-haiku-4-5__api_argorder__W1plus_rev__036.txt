def transfer(acct, cents, reason):
    import ledger
    
    # Convert cents to wire format: <sign><digits>#<check>
    # Calculate check digit as sum of digits mod 10
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate Luhn-style check digit
    check = sum(int(d) for d in digits_str) % 10
    
    # Format as <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
