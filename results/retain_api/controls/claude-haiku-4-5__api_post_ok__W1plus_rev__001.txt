import ledger

def record(acct, cents, note):
    """Post a ledger entry and return the new entry id."""
    # Encode the amount as a string in the wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with the required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
