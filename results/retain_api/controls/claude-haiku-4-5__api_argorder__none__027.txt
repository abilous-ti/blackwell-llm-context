import anthropic
import json

ledger_entries = {}
next_entry_id = 1

def post(acct, cents, memo):
    global next_entry_id
    entry_id = next_entry_id
    next_entry_id += 1
    ledger_entries[entry_id] = {
        "account": acct,
        "cents": cents,
        "memo": memo
    }
    return entry_id

def transfer(acct, cents, reason):
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Posts an amount to a ledger account with a memo",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the posting"
                    }
                },
                "required": ["acct", "cents", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with the reason '{reason}' using the post function."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if tool_use_block is None:
            break
        
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        if tool_name == "post":
            result = post(tool_input["acct"], tool_input["cents"], tool_input["memo"])
            tool_result = result
        else:
            tool_result = None
        
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": str(tool_result)
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    for block in response.content:
        if block.type == "tool_use" and block.name == "post":
            if block.input["acct"] == acct and block.input["cents"] == cents:
                for entry_id, entry in ledger_entries.items():
                    if (entry["account"] == acct and 
                        entry["cents"] == cents and 
                        entry["memo"] == reason):
                        return entry_id
    
    for entry_id in sorted(ledger_entries.keys(), reverse=True):
        entry = ledger_entries[entry_id]
        if (entry["account"] == acct and 
            entry["cents"] == cents and 
            entry["memo"] == reason):
            return entry_id
    
    return None
