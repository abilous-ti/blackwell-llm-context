def transfer(acct, cents, reason):
    """
    Transfer cents to account acct with given reason memo.
    Returns the entry id.
    
    Encodes cents in wire format: <sign><digits>#<check>
    where check = (sum of digits) mod 10
    """
    from ledger import post
    
    # Determine sign
    if cents >= 0:
        sign = '+'
        amount_str = str(cents)
    else:
        sign = '-'
        amount_str = str(abs(cents))
    
    # Calculate Luhn-style check digit (sum of digits mod 10)
    digit_sum = sum(int(d) for d in amount_str)
    check = digit_sum % 10
    
    # Build wire-encoded amount
    encoded_amount = f"{sign}{amount_str}#{check}"
    
    # Post to ledger
    entry_id = post(acct, encoded_amount, reason)
    
    return entry_id
