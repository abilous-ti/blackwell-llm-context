```python
import ledger

def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id.
    
    Args:
        acct: Account name string
        cents: Amount in minor units (integer cents)
        note: Memo string for the transaction
    
    Returns:
        The new integer entry id (>=1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    check_digit = sum(int(d) for d in digits_str) % 10
    amount = f"{sign}{abs_cents}#{check_digit}"
    
    entry_id = ledger.post(acct, amount, memo=note)
    return entry_id
```
