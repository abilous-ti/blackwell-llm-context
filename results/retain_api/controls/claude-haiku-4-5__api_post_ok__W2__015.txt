import ledger

def record(acct, cents, note):
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit (sum of digits mod 10)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Construct encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return entry id
    entry_id = ledger.post(acct, encoded_amount, note)
    return entry_id
