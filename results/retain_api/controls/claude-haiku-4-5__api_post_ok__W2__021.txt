import ledger


def record(acct, cents, note):
    """
    Post a monetary amount to an account and return the entry id.
    
    Args:
        acct: Account identifier
        cents: Integer amount in cents (can be positive or negative)
        note: Memo string
    
    Returns:
        The new entry id from ledger.post
    """
    # Calculate the sum of digits for the Luhn check
    digits_str = str(abs(cents))
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Format the amount string
    amount_encoded = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return the entry id
    return ledger.post(acct, amount_encoded, note)
