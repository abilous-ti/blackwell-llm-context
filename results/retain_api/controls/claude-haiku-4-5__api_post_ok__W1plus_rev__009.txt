import ledger


def record(acct, cents, note):
    """Post a ledger entry and return the new entry id.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        note: Memo text
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If posting fails
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to digits string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    # Encode amount as required format: <sign><digits>#<check>
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger with required keyword-only memo argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
