import anthropic

ledger_code = """
# Simulated ledger module
_entries = {}
_next_id = 1

def post(acct, cents, note):
    global _next_id
    entry_id = _next_id
    _entries[entry_id] = {
        'account': acct,
        'amount': cents,
        'memo': note,
        'id': entry_id
    }
    _next_id += 1
    return entry_id
"""

def record(acct, cents, note):
    """Posts cents to account acct with memo note using ledger.post, returns new entry id."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a Python code assistant. Given a ledger module with a post function, write Python code that:
1. Posts {cents} cents to account '{acct}' with memo '{note}' using ledger.post
2. Returns the new entry id

The ledger module has already been imported and has a post(acct, cents, note) function that returns the entry id.

Write ONLY the Python code (no explanation). The code should be a single statement or expression that returns the entry id.

Example format:
ledger.post('checking', 100, 'deposit')
"""
            }
        ]
    )
    
    code_response = message.content[0].text.strip()
    
    exec(ledger_code, globals())
    
    result = eval(f"ledger.{code_response}" if not code_response.startswith("ledger.") else code_response, {"ledger": type('module', (), {'post': post})()})
    
    return result


if __name__ == "__main__":
    exec(ledger_code, globals())
    
    entry_id = record("checking", 10000, "monthly deposit")
    print(f"New entry id: {entry_id}")
    
    entry_id2 = record("savings", 50000, "transfer")
    print(f"New entry id: {entry_id2}")
