import anthropic


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Posts a ledger entry",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Memo for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account {acct} with reason '{reason}' using the post function and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for content_block in response.content:
            if content_block.type == "tool_use":
                tool_name = content_block.name
                tool_input = content_block.input
                tool_use_id = content_block.id
                
                if tool_name == "post":
                    return tool_use_id
    
    return ""


if __name__ == "__main__":
    entry_id = transfer("checking", 10000, "Monthly transfer")
    print(f"Entry ID: {entry_id}")
