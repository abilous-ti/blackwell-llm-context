```python
import ledger

def record(acct, cents, note):
    """
    Posts a monetary amount to a ledger account.
    
    Args:
        acct: Account name (string)
        cents: Amount in minor units (integer cents)
        note: Memo/note for the transaction (string)
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Encode the amount as a wire-encoded string
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with the correct signature
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
