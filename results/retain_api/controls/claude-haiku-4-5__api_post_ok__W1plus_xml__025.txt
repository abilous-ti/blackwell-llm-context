```python
import ledger


def record(acct, cents, note):
    """Post a ledger entry and return the new entry id."""
    # Encode the amount in wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Format the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with the required API contract
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
