import anthropic
import ledger

def record(acct: str, cents: int, note: str) -> int:
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    # Call the ledger.post API with the required arguments
    # account (acct) is first positional argument
    # amount (cents) is second positional argument  
    # memo (note) is required keyword-only argument
    entry_id = ledger.post(acct, cents, memo=note)
    return entry_id
