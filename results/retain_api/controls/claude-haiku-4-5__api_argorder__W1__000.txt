
```python
import anthropic
import ledger

def transfer(acct: str, cents: int, reason: str) -> int:
    """
    Transfer funds to an account using the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents to transfer
        reason: Memo/reason for the transfer
        
    Returns:
        The new entry id from the ledger
        
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    return ledger.post(acct, cents, memo=reason)
```
