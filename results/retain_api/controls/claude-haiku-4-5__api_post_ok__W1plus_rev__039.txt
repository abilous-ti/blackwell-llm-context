import ledger

def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id."""
    # Encode the amount in the wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword-only
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
