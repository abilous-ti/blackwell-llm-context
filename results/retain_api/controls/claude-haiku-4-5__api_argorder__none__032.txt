import anthropic
import json
import re

client = anthropic.Anthropic()

def transfer(acct: str, cents: int, reason: str) -> str:
    """Transfer cents to account with given reason and return entry id."""
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Post a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The transaction memo"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with the reason: {reason}"
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    tool_use_block = None
    for block in response.content:
        if hasattr(block, 'type') and block.type == "tool_use":
            tool_use_block = block
            break
    
    if not tool_use_block:
        return ""
    
    tool_input = tool_use_block.input
    entry_id_json = json.dumps({
        "account": tool_input.get("account", acct),
        "amount": tool_input.get("amount", cents),
        "memo": tool_input.get("memo", reason),
        "entry_id": f"entry_{acct}_{cents}_{hash(reason) & 0xfffffff}"
    })
    
    entry_id_match = re.search(r'"entry_id":\s*"([^"]+)"', entry_id_json)
    if entry_id_match:
        return entry_id_match.group(1)
    
    return f"entry_{acct}_{cents}_{hash(reason) & 0xfffffff}"
