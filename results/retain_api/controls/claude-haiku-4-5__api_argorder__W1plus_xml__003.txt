```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer cents to an account using the ledger module.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id returned by ledger.post()
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Convert cents to the wire encoding format: <sign><digits>#<check>
    # where check is (sum of digits) mod 10
    
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with account first, amount second (positional),
    # and memo as required keyword-only argument
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
