```python
import ledger


def transfer(acct, cents, reason):
    """
    Transfer cents to account acct with reason as memo.
    Returns the entry id.
    """
    # Encode the amount as a string in the wire format: <sign><digits>#<check>
    # cents is an integer (can be positive or negative)
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Construct the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, amount second, and memo as keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
