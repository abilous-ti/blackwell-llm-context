import anthropic
import json
import re

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using Claude to generate the properly encoded amount.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (minor units)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger
    """
    
    def encode_amount(amount_cents: int) -> str:
        """Encode an amount in cents to the wire format."""
        sign = '+' if amount_cents >= 0 else '-'
        digits = str(abs(amount_cents))
        check = sum(int(d) for d in digits) % 10
        return f"{sign}{digits}#{check}"
    
    def decode_amount(encoded: str) -> int:
        """Decode a wire-format amount string."""
        match = re.match(r'^([+-])(\d+)#(\d)$', encoded)
        if not match:
            raise ValueError(f"Invalid amount format: {encoded}")
        
        sign, digits, check = match.groups()
        calculated_check = sum(int(d) for d in digits) % 10
        
        if int(check) != calculated_check:
            raise ValueError(f"Check digit mismatch for {encoded}")
        
        amount = int(digits)
        if sign == '-':
            amount = -amount
        
        return amount
    
    client = anthropic.Anthropic()
    
    encoded_amount = encode_amount(cents)
    
    tools = [
        {
            "name": "post_ledger_entry",
            "description": "Post a transfer to the ledger system",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": f"The amount in wire format: {encoded_amount}"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the transfer"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"""You are a ledger posting system. Post a transfer with these details:
- Account: {acct}
- Amount (in wire format): {encoded_amount}
- Reason: {reason}

Use the post_ledger_entry tool to complete this transfer. The amount is already in the correct wire format."""
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    for block in response.content:
        if block.type == "tool_use":
            if block.name == "post_ledger_entry":
                tool_input = block.input
                
                decoded_amount = decode_amount(tool_input["amount"])
                
                simulated_entry_id = f"TXN-{hash(f'{acct}-{decoded_amount}-{reason}') % 1000000:06d}"
                
                return simulated_entry_id
    
    raise ValueError("No tool use found in response")


if __name__ == "__main__":
    entry_id = transfer("ACC123", 1230, "Payment for services")
    print(f"Entry ID: {entry_id}")
    
    entry_id2 = transfer("ACC456", -750, "Refund")
    print(f"Entry ID 2: {entry_id2}")
    
    entry_id3 = transfer("ACC789", 7, "Small adjustment")
    print(f"Entry ID 3: {entry_id3}")
