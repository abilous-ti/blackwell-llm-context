def transfer(acct, cents, reason):
    """
    Transfer `cents` to account `acct` with memo `reason`.
    
    Args:
        acct: account identifier string
        cents: integer amount in minor units (cents)
        reason: memo string
    
    Returns:
        entry id (integer >= 1) on success
    
    Raises:
        ledger.PostError on failure
    """
    import ledger
    
    # Encode amount as wire format: <sign><digits>#<check>
    # where check = (sum of decimal digits) mod 10
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format: <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with required keyword argument memo=
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
