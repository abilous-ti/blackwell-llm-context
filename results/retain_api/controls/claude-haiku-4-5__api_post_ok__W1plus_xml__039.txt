```python
import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        note: Memo/description for the transaction
        
    Returns:
        The new entry id from the ledger
        
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value
    abs_cents = abs(cents)
    
    # Convert to string of digits
    digits_str = str(abs_cents)
    
    # Calculate check digit (sum of digits mod 10)
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Construct wire-encoded amount
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
