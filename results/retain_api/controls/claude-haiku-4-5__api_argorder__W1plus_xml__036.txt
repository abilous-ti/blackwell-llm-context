```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account name (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for transfer (string)
    
    Returns:
        Entry id from ledger.post()
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Convert cents to wire encoding format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    check_digit = sum(int(d) for d in digits_str) % 10
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with account first, encoded amount second, memo as keyword arg
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
