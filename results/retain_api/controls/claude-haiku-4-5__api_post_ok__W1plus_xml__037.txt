```python
import ledger

def record(acct, cents, note):
    """
    Posts an amount to an account and returns the new entry id.
    
    Args:
        acct: Account identifier (str)
        cents: Amount in minor units (integer cents)
        note: Memo/note for the transaction (str)
    
    Returns:
        The new integer entry id (>=1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Encode cents to wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with required keyword argument
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
```
