```python
import ledger


def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    # Encode cents into the wire format: <sign><digits>#<check>
    # where check is (sum of digits) mod 10
    
    if cents >= 0:
        sign = '+'
        amount_str = str(cents)
    else:
        sign = '-'
        amount_str = str(-cents)
    
    # Calculate check digit: sum of all decimal digits mod 10
    digit_sum = sum(int(d) for d in amount_str)
    check = digit_sum % 10
    
    # Build the encoded amount string
    encoded_amount = f"{sign}{amount_str}#{check}"
    
    # Call ledger.post with the required signature:
    # ledger.post(account, amount, *, memo)
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
