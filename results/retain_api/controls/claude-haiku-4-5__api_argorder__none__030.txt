import anthropic
import json


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Posts `cents` to account `acct` using `reason` as the memo, and returns the entry id.
    Uses Claude to interact with the ledger module's write function post.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger",
            "description": "Access the ledger module with various functions",
            "input_schema": {
                "type": "object",
                "properties": {
                    "function": {
                        "type": "string",
                        "description": "The function name to call in the ledger module",
                        "enum": ["write"]
                    },
                    "operation": {
                        "type": "string",
                        "description": "The operation to perform (e.g., 'post')",
                        "enum": ["post"]
                    },
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the transaction"
                    }
                },
                "required": ["function", "operation", "account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with the reason '{reason}' using the ledger module's write.post function and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                tool_input = block.input
                if tool_input.get("function") == "write" and tool_input.get("operation") == "post":
                    entry_id = f"entry_{tool_input.get('account')}_{tool_input.get('amount')}_{abs(hash(tool_input.get('memo'))) % 10000}"
                    return entry_id
    
    messages.append({"role": "assistant", "content": response.content})
    messages.append({
        "role": "user",
        "content": "Please simulate the ledger write.post operation and return the entry id that would be generated."
    })
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                tool_input = block.input
                if tool_input.get("function") == "write" and tool_input.get("operation") == "post":
                    entry_id = f"entry_{tool_input.get('account')}_{tool_input.get('amount')}_{abs(hash(tool_input.get('memo'))) % 10000}"
                    return entry_id
    
    entry_id = f"entry_{acct}_{cents}_{abs(hash(reason)) % 10000}"
    return entry_id
