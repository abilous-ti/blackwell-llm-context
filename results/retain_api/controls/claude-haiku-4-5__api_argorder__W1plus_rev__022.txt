```python
import ledger


def transfer(acct, cents, reason):
    """
    Transfer funds to an account via the ledger.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer (string)
    
    Returns:
        The entry id (integer >= 1) from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Encode the amount according to the wire format: <sign><digits>#<check>
    # where check digit is (sum of decimal digits) mod 10
    
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of all decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Construct the encoded amount string
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with the correct signature:
    # post(account: str, amount, *, memo: str)
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
