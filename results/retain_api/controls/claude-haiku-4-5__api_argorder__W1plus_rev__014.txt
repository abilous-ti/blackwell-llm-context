```python
import ledger


def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
        
    Raises:
        ledger.PostError: If the transfer fails
    """
    # Encode the amount as a wire-format string
    # Format: <sign><digits>#<check>
    # where digits is the amount in cents (minor units)
    # and check is (sum of digits) mod 10
    
    sign = '+' if cents >= 0 else '-'
    digits_str = str(abs(cents))
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    amount_encoded = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with the encoded amount
    # Note: account is first positional arg, amount is second positional arg,
    # memo is required keyword-only argument
    entry_id = ledger.post(acct, amount_encoded, memo=reason)
    
    return entry_id
```
