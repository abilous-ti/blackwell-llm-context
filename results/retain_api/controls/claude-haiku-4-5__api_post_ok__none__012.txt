import anthropic


def record(acct: str, cents: int, note: str) -> str:
    """
    Posts `cents` to account `acct` with memo `note` using ledger.post.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "note": {
                        "type": "string",
                        "description": "Transaction memo/note"
                    }
                },
                "required": ["acct", "cents", "note"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with note '{note}' and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                tool_input = block.input
                entry_id = f"entry_{acct}_{tool_input['cents']}_{hash(tool_input['note']) % 10000}"
                
                messages.append({
                    "role": "assistant",
                    "content": response.content
                })
                
                messages.append({
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": f"Successfully posted. Entry ID: {entry_id}"
                        }
                    ]
                })
                
                final_response = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=tools,
                    messages=messages
                )
                
                for final_block in final_response.content:
                    if hasattr(final_block, 'text'):
                        if entry_id in final_block.text:
                            return entry_id
                
                return entry_id
    
    return "entry_unknown"
