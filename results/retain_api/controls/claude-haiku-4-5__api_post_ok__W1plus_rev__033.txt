import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id.
    
    Args:
        acct: Account identifier string
        cents: Amount in minor units (integer cents)
        note: Memo/description for the transaction
    
    Returns:
        The new integer entry id (>=1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value
    abs_cents = abs(cents)
    
    # Convert to string of digits
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of all decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Encode as wire format: <sign><digits>#<check>
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
