```python
import ledger


def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Convert cents to the wire encoding format
    # Format: <sign><digits>#<check>
    # where check = (sum of decimal digits) mod 10
    
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit as sum of all digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Construct the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger with required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
