import anthropic
import json


def record(acct: str, cents: int, note: str) -> str:
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    client = anthropic.Anthropic()
    
    # Define the ledger.post tool
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts an entry to the ledger with account, amount in cents, and a note/memo",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "The amount in cents to post"
                    },
                    "note": {
                        "type": "string",
                        "description": "The memo or note for this entry"
                    }
                },
                "required": ["acct", "cents", "note"]
            }
        }
    ]
    
    # Create a message to Claude asking it to post to the ledger
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account '{acct}' with the note '{note}' and return the entry id."
        }
    ]
    
    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response to extract the tool use
    for content_block in response.content:
        if content_block.type == "tool_use":
            tool_input = content_block.input
            # Simulate the ledger.post call
            entry_id = f"entry_{acct}_{cents}_{hash(note) % 10000:04d}"
            
            # Continue conversation with the result
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": content_block.id,
                        "content": json.dumps({"entry_id": entry_id, "status": "success"})
                    }
                ]
            })
            
            # Get Claude's final response with the entry id
            final_response = client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
            
            # Extract the entry id from the final response
            for final_block in final_response.content:
                if hasattr(final_block, 'text'):
                    # Parse the entry id from Claude's response
                    return entry_id
    
    return entry_id
