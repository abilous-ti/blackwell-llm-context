```python
import ledger

def transfer(acct, cents, reason):
    """
    Post a transfer to an account.
    
    Args:
        acct: Account identifier string
        cents: Amount in minor units (integer cents)
        reason: Memo describing the transfer
    
    Returns:
        The entry id returned by ledger.post()
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Encode cents as wire format: <sign><digits>#<check>
    # where check = (sum of decimal digits) mod 10
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the amount string
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with correct signature
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
