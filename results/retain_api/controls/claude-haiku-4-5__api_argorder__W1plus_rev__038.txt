```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts cents to account acct using reason as the memo.
    Returns the entry id.
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check_sum = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check_sum}"
    
    entry_id = ledger.post(acct, amount, memo=reason)
    return entry_id
```
