import anthropic
import json

client = anthropic.Anthropic()

# Keep track of ledger entries
ledger_entries = {}
next_entry_id = 1


def post(acct: str, cents: int, reason: str) -> int:
    """Post an entry to the ledger and return the entry id."""
    global next_entry_id
    
    entry_id = next_entry_id
    next_entry_id += 1
    
    ledger_entries[entry_id] = {
        "account": acct,
        "amount": cents,
        "memo": reason
    }
    
    return entry_id


def transfer(acct: str, cents: int, reason: str) -> int:
    """Transfer cents to account acct using reason as the memo, and return the entry id."""
    return post(acct, cents, reason)


def process_tool_call(tool_name: str, tool_input: dict):
    """Process tool calls from Claude."""
    if tool_name == "post":
        acct = tool_input.get("acct")
        cents = tool_input.get("cents")
        reason = tool_input.get("reason")
        return post(acct, cents, reason)
    elif tool_name == "transfer":
        acct = tool_input.get("acct")
        cents = tool_input.get("cents")
        reason = tool_input.get("reason")
        return transfer(acct, cents, reason)
    else:
        return f"Unknown tool: {tool_name}"


def main():
    """Example usage with Claude."""
    tools = [
        {
            "name": "post",
            "description": "Post an entry to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account name"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "reason": {
                        "type": "string",
                        "description": "Memo for the transaction"
                    }
                },
                "required": ["acct", "cents", "reason"]
            }
        },
        {
            "name": "transfer",
            "description": "Transfer cents to an account using reason as memo",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account name"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "reason": {
                        "type": "string",
                        "description": "Memo for the transaction"
                    }
                },
                "required": ["acct", "cents", "reason"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": "Transfer 5000 cents to savings account for emergency fund. Then transfer 2500 cents to checking account for monthly bills."
        }
    ]
    
    print("User: Transfer 5000 cents to savings account for emergency fund. Then transfer 2500 cents to checking account for monthly bills.")
    print()
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Process the response
        if response.stop_reason == "tool_use":
            # Find tool use blocks
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    
                    print(f"Claude calls: {tool_name}({tool_input})")
                    result = process_tool_call(tool_name, tool_input)
                    print(f"Result: {result}")
                    print()
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": str(result)
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            # End of conversation
            for block in response.content:
                if hasattr(block, "text"):
                    print(f"Claude: {block.text}")
            break
    
    print("\nLedger entries:")
    for entry_id, entry in ledger_entries.items():
        print(f"  Entry {entry_id}: {entry['account']} {entry['amount']} cents - {entry['memo']}")


if __name__ == "__main__":
    main()
