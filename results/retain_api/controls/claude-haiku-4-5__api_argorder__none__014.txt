import anthropic
import json

ledger_tool = {
    "name": "ledger",
    "description": "A ledger system for managing financial transactions",
    "input_schema": {
        "type": "object",
        "properties": {
            "action": {
                "type": "string",
                "description": "The action to perform on the ledger",
                "enum": ["post"]
            },
            "account": {
                "type": "string",
                "description": "The account to post to"
            },
            "amount": {
                "type": "integer",
                "description": "The amount in cents"
            },
            "memo": {
                "type": "string",
                "description": "The memo/reason for the transaction"
            }
        },
        "required": ["action", "account", "amount", "memo"]
    }
}

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using the ledger module's post function.
    
    Args:
        acct: The account to post to
        cents: The amount in cents
        reason: The reason/memo for the transaction
        
    Returns:
        The entry id returned by the ledger post function
    """
    client = anthropic.Anthropic()
    
    # Create a message with the tool
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[ledger_tool],
        messages=[
            {
                "role": "user",
                "content": f"Please post {cents} cents to account '{acct}' with the reason '{reason}' and return the entry id."
            }
        ]
    )
    
    # Process the response and extract tool use
    entry_id = None
    
    for content_block in message.content:
        if content_block.type == "tool_use":
            tool_input = content_block.input
            
            # Simulate the ledger post function
            if tool_input["action"] == "post":
                # Generate an entry id based on the transaction details
                entry_id = f"entry_{hash(f'{tool_input[\"account\"]}_{tool_input[\"amount\"]}_{tool_input[\"memo\"]}') % 100000:05d}"
                
                # Continue the conversation with the tool result
                message = client.messages.create(
                    model="claude-3-5-sonnet-20241022",
                    max_tokens=1024,
                    tools=[ledger_tool],
                    messages=[
                        {
                            "role": "user",
                            "content": f"Please post {cents} cents to account '{acct}' with the reason '{reason}' and return the entry id."
                        },
                        {
                            "role": "assistant",
                            "content": message.content
                        },
                        {
                            "role": "user",
                            "content": [
                                {
                                    "type": "tool_result",
                                    "tool_use_id": content_block.id,
                                    "content": json.dumps({"status": "success", "entry_id": entry_id})
                                }
                            ]
                        }
                    ]
                )
                
                # Extract the entry id from the final response
                for final_block in message.content:
                    if hasattr(final_block, 'text'):
                        # The entry id should be mentioned in the response
                        if "entry_" in final_block.text:
                            # Extract entry id from response
                            import re
                            match = re.search(r'entry_\d+', final_block.text)
                            if match:
                                return match.group(0)
                
                return entry_id
    
    return entry_id


if __name__ == "__main__":
    # Test the transfer function
    result = transfer("checking", 10000, "Test transfer")
    print(f"Entry ID: {result}")
