def transfer(acct, cents, reason):
    """Transfer cents to account, using reason as memo. Returns entry id."""
    import ledger
    
    # Convert cents to wire format: <sign><digits>#<check>
    # Where check = sum of decimal digits mod 10
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    amount_str = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword
    entry_id = ledger.post(acct, amount_str, memo=reason)
    return entry_id
