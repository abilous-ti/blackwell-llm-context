def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post operation
    """
    from ledger import post
    
    # Determine sign based on cents value
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value for processing
    abs_cents = abs(cents)
    
    # Convert to string of digits
    digits_str = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Format as <sign><digits>#<check>
    amount_encoded = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, amount_encoded, reason)
    
    return entry_id
