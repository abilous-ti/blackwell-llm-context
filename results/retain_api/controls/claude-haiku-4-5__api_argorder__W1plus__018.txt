def transfer(acct, cents, reason):
    import ledger
    
    # Convert cents to the wire-encoding format
    # Calculate checksum: sum of decimal digits mod 10
    digits_str = str(abs(cents))
    checksum = sum(int(d) for d in digits_str) % 10
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Format as <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{checksum}"
    
    # Call ledger.post with required keyword argument memo
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
