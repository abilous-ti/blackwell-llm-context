```python
def transfer(acct, cents, reason):
    import ledger
    
    # Convert cents to wire format: <sign><digits>#<check>
    # cents is an integer, digits are the absolute value
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format: <sign><digits>#<check>
    amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword-only
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
