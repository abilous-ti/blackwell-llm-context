import anthropic


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Posts cents to account acct using reason as the memo via Claude's tool use.
    Returns the entry id.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts an entry to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo for the entry"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with reason '{reason}' and return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    while response.stop_reason == "tool_use":
        # Find the tool use block
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        # Simulate the ledger post function
        tool_input = tool_use_block.input
        entry_id = f"entry_{hash((tool_input['account'], tool_input['amount'], tool_input['memo'])) % 1000000}"
        
        # Add the assistant's response and tool result to messages
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": entry_id
                }
            ]
        })
        
        # Get the next response
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # Extract the entry id from the final response
    for block in response.content:
        if hasattr(block, "text"):
            text = block.text
            # Try to find entry_id in the response
            if "entry_" in text:
                # Extract entry id from response
                import re
                match = re.search(r'entry_\d+', text)
                if match:
                    return match.group()
    
    # Return a default entry id based on the input
    return f"entry_{hash((acct, cents, reason)) % 1000000}"
