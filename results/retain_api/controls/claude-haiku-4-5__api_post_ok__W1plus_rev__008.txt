import ledger


def record(acct, cents, note):
    """Post a ledger entry and return the new entry id."""
    # Encode cents as wire format: <sign><digits>#<check>
    # where check = (sum of digits) mod 10
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(-cents)
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Construct encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger with required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
