import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id."""
    sign = '+' if cents >= 0 else '-'
    amount_abs = abs(cents)
    digits = str(amount_abs)
    checksum = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{checksum}"
    return ledger.post(acct, encoded_amount, memo=note)
