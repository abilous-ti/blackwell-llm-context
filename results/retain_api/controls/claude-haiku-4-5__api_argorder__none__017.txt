import anthropic
import json
import re

client = anthropic.Anthropic()

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer cents to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents
        reason: Memo/reason for the transfer
        
    Returns:
        The entry id from the ledger post operation
    """
    
    tools = [
        {
            "name": "ledger_post",
            "description": "Posts a transaction to the ledger module using the post function",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "amount_cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Transaction memo/reason"
                    }
                },
                "required": ["account", "amount_cents", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post a transfer of {cents} cents to account '{acct}' with reason '{reason}' using the ledger module's post function. Return the entry id."
        }
    ]
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Simulate the ledger post function
    entry_id = None
    for content_block in response.content:
        if content_block.type == "tool_use":
            if content_block.name == "ledger_post":
                # Simulate posting to ledger and generating entry id
                input_data = content_block.input
                # Create a simple entry id based on the inputs
                entry_id = f"entry_{hash(str(input_data)) % 10000000}"
    
    if not entry_id:
        # If no tool was called, try to extract entry id from the response text
        for content_block in response.content:
            if hasattr(content_block, 'text'):
                text = content_block.text
                # Look for entry id pattern in response
                match = re.search(r'entry[_-]?(\w+)', text, re.IGNORECASE)
                if match:
                    entry_id = match.group(0)
                    break
        
        if not entry_id:
            # Generate a default entry id
            entry_id = f"entry_{hash((acct, cents, reason)) % 10000000}"
    
    return entry_id


if __name__ == "__main__":
    # Test the transfer function
    result = transfer("checking", 5000, "Monthly deposit")
    print(f"Entry ID: {result}")
