```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer cents to an account using the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo describing the transfer
    
    Returns:
        The entry id returned by ledger.post
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Convert cents to wire encoding format: +<digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    
    # Calculate check digit as sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in str(abs_cents))
    check = digit_sum % 10
    
    # Format as wire encoding
    amount = f"{sign}{abs_cents}#{check}"
    
    # Call ledger.post with required keyword argument for memo
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
