import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id.
    
    Args:
        acct: Account name/identifier
        cents: Amount in cents (integer)
        note: Memo/description for the transaction
        
    Returns:
        New entry id (integer >= 1)
        
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Encode cents to wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with the required signature
    entry_id = ledger.post(acct, amount, memo=note)
    return entry_id
