```python
import ledger

def transfer(acct, cents, reason):
    """Transfer cents to an account with a given reason.
    
    Args:
        acct: Account identifier (str)
        cents: Amount in cents (int)
        reason: Memo/reason for the transfer (str)
    
    Returns:
        Entry ID from ledger.post()
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    return ledger.post(acct, amount, memo=reason)
```
