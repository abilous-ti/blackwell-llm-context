import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id.
    
    Args:
        acct: Account name
        cents: Amount in cents (integer)
        note: Memo for the transaction
        
    Returns:
        The new entry id (>=1)
        
    Raises:
        ledger.PostError: On posting failure
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    return ledger.post(acct, amount, memo=note)
