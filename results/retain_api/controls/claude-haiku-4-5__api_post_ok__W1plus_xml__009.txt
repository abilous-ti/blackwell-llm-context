```python
import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id.
    
    Args:
        acct: Account identifier string
        cents: Amount in minor units (integer cents)
        note: Memo/description for the transaction
        
    Returns:
        The new integer entry id (>=1)
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Construct wire-encoded amount
    amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with required keyword argument for memo
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
```
