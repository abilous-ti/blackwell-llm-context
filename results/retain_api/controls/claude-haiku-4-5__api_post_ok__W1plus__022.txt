import ledger

def encode_amount(cents):
    """Encode an integer cent amount to wire format."""
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    return f"{sign}{digits}#{check}"

def record(acct, cents, note):
    """Post a monetary amount to an account and return the entry id."""
    encoded_amount = encode_amount(cents)
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    return entry_id
