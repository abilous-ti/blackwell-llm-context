```python
import ledger

def transfer(acct, cents, reason):
    # Convert cents to the wire-encoding format
    # Format: <sign><digits>#<check>
    
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit as sum of decimal digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, amount second, and memo as keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
