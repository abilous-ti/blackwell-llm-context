
```python
import ledger

def transfer(acct, cents, reason):
    """
    Post a transfer to an account in the ledger.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer (string)
    
    Returns:
        The new entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    return ledger.post(acct, cents, memo=reason)
```
