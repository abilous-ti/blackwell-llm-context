```python
import ledger


def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    
    Args:
        acct: Account name
        cents: Amount in cents (integer)
        note: Memo text
    
    Returns:
        The new entry id from ledger.post
    """
    sign = '+' if cents >= 0 else '-'
    amount_abs = abs(cents)
    digits = str(amount_abs)
    check = sum(int(d) for d in digits) % 10
    encoded_amount = f"{sign}{digits}#{check}"
    
    return ledger.post(acct, encoded_amount, memo=note)
```
