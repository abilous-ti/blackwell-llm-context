def transfer(acct, cents, reason):
    """
    Posts a monetary transfer to the ledger.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo/description for the transfer
    
    Returns:
        The entry id from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    import ledger
    
    # Determine the sign based on cents value
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value for digits
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Format the amount string as <sign><digits>#<check>
    amount_str = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with the required signature
    entry_id = ledger.post(acct, amount_str, memo=reason)
    
    return entry_id
