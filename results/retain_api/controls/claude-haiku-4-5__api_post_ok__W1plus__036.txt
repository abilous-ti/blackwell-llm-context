import ledger


def encode_amount(cents):
    """Encode an integer amount in cents to wire format."""
    sign = '+' if cents >= 0 else '-'
    digits_str = str(abs(cents))
    check = sum(int(d) for d in digits_str) % 10
    return f"{sign}{digits_str}#{check}"


def record(acct, cents, note):
    """Post cents to account acct with memo note. Returns the new entry id."""
    encoded_amount = encode_amount(cents)
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    return entry_id
