```python
import ledger

def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    # Encode cents as wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format as wire encoding
    amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with correct signature
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
```
