import ledger


def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    """
    # Encode the amount according to the wire-encoding invariant
    # Format: <sign><digits>#<check>
    # where <digits> is amount in minor units (cents)
    # and <check> is (sum of decimal digits) mod 10
    
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits)
    check = digit_sum % 10
    
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
