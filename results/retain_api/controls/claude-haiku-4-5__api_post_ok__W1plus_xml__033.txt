import ledger


def record(acct, cents, note):
    sign = '+' if cents >= 0 else '-'
    amount_abs = abs(cents)
    digits_str = str(amount_abs)
    check = sum(int(d) for d in digits_str) % 10
    encoded_amount = f"{sign}{amount_abs}#{check}"
    
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    return entry_id
