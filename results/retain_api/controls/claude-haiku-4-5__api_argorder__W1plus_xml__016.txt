```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer funds to an account.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id of the posted transaction
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Determine sign based on cents value
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Encode amount in wire format: <sign><digits>#<check>
    amount = f"{sign}{digits}#{check}"
    
    # Post to ledger with required memo keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
