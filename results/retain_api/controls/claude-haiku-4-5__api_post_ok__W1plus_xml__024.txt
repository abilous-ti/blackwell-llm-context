```python
import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id."""
    # Encode the amount in wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    check_digit = sum(int(d) for d in digits_str) % 10
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger with required keyword-only memo argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
