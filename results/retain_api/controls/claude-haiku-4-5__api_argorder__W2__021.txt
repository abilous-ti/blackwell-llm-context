def transfer(acct, cents, reason):
    """
    Transfer funds to an account by posting to the ledger.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry id from the ledger post
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format as wire-encoded string
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger
    entry_id = post(acct, encoded_amount, reason)
    
    return entry_id
