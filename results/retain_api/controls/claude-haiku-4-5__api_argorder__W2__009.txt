import ledger

def transfer(acct, cents, reason):
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value for digit processing
    abs_cents = abs(cents)
    
    # Convert to string digits
    digits_str = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the amount string
    amount_str = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return entry id
    entry_id = ledger.post(acct, amount_str, reason)
    
    return entry_id
