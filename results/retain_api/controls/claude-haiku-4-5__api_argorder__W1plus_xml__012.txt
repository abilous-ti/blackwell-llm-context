```python
def transfer(acct, cents, reason):
    import ledger
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit (sum of digits mod 10)
    check = sum(int(d) for d in digits_str) % 10
    
    # Encode amount in wire format: <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with correct signature
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
