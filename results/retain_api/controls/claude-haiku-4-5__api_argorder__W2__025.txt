import anthropic
import json
import re


def calculate_check_digit(digits_str: str) -> str:
    """Calculate Luhn-style check digit."""
    digit_sum = sum(int(d) for d in digits_str)
    return str(digit_sum % 10)


def encode_amount(cents: int) -> str:
    """Encode amount in wire format: <sign><digits>#<check>"""
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    check = calculate_check_digit(digits_str)
    return f"{sign}{digits_str}#{check}"


def decode_amount(encoded: str) -> int:
    """Decode wire format back to cents, validating check digit."""
    match = re.match(r'^([+-])(\d+)#(\d)$', encoded)
    if not match:
        raise ValueError(f"Invalid encoded amount format: {encoded}")
    
    sign, digits_str, check_str = match.groups()
    expected_check = calculate_check_digit(digits_str)
    
    if check_str != expected_check:
        raise ValueError(f"Check digit mismatch for {encoded}")
    
    cents = int(digits_str)
    if sign == '-':
        cents = -cents
    
    return cents


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer funds to an account using Claude as an AI backbone.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (can be positive or negative)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger
    """
    client = anthropic.Anthropic()
    
    # Encode the amount in wire format
    encoded_amount = encode_amount(cents)
    
    # Create a message to Claude asking it to help process the transfer
    # Claude will understand the transfer intent and return a simulated entry ID
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a ledger system assistant. Process this transfer request:
- Account: {acct}
- Amount (wire-encoded): {encoded_amount}
- Reason: {reason}

Respond with a JSON object containing:
{{"entry_id": "a unique entry identifier for this transfer"}}

The entry_id should be a string in format "entry_XXXXXXXX" where X are hexadecimal characters."""
            }
        ]
    )
    
    # Parse the response
    response_text = message.content[0].text
    try:
        # Extract JSON from response
        json_match = re.search(r'\{[^{}]*"entry_id"[^{}]*\}', response_text)
        if json_match:
            response_json = json.loads(json_match.group())
            entry_id = response_json.get("entry_id")
            if entry_id:
                return entry_id
    except (json.JSONDecodeError, AttributeError):
        pass
    
    # Fallback: generate entry ID if parsing fails
    import hashlib
    data = f"{acct}{encoded_amount}{reason}".encode()
    hash_hex = hashlib.md5(data).hexdigest()[:8]
    return f"entry_{hash_hex}"


if __name__ == "__main__":
    # Test the transfer function
    entry_id = transfer("account_123", 1230, "Payment for services")
    print(f"Transfer completed with entry ID: {entry_id}")
    
    # Test with negative amount
    entry_id2 = transfer("account_456", -750, "Refund")
    print(f"Refund processed with entry ID: {entry_id2}")
