```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger module.
    
    Args:
        acct: Account identifier (str)
        cents: Amount in cents (int)
        reason: Memo/reason for the transfer (str)
    
    Returns:
        The entry id (int >= 1) from the ledger post operation
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Convert cents to the wire-encoded format
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    check_digit = sum(int(d) for d in digits_str) % 10
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger with required keyword argument for memo
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
