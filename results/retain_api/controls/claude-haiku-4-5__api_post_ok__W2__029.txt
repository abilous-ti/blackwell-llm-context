def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id."""
    # Convert cents to the wire-encoding format
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format: <sign><digits>#<check>
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return the entry id
    entry_id = ledger.post(acct, encoded_amount, note)
    return entry_id
