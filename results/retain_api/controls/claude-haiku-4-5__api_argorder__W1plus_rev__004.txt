```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts a transfer to the ledger.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post
    
    Raises:
        ledger.PostError: If the post fails
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value for encoding
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn check digit: sum of digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Format amount string: <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with required memo keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
