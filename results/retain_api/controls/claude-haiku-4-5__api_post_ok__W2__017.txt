import ledger

def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    
    cents is an integer in minor units (cents).
    Encodes it as a wire string with format: <sign><digits>#<check>
    where check = (sum of decimal digits) mod 10
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit: sum of all digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Encode as wire format
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return the entry id
    entry_id = ledger.post(acct, encoded_amount, note)
    return entry_id
