import ledger


def record(acct, cents, note):
    # Calculate the check digit using Luhn-style sum mod 10
    digits_str = str(abs(cents))
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Encode the amount according to wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    encoded_amount = f"{sign}{abs(cents)}#{check_digit}"
    
    # Post to ledger and return the entry id
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    return entry_id
