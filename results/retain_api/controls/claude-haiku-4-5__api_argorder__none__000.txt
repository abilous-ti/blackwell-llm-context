import anthropic
import json


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer cents to account using reason as memo via Claude API,
    which will call the ledger.post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents
        reason: Memo/reason for transfer
        
    Returns:
        The entry id from the ledger post
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "post",
            "description": "Post a transaction to the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "acct": {
                        "type": "string",
                        "description": "Account identifier"
                    },
                    "cents": {
                        "type": "integer",
                        "description": "Amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "Memo for the transaction"
                    }
                },
                "required": ["acct", "cents", "memo"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please post a transfer of {cents} cents to account {acct} with reason '{reason}' using the ledger.post function and return the entry id."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, 'text'):
                    text = block.text
                    try:
                        data = json.loads(text)
                        if isinstance(data, dict) and 'entry_id' in data:
                            return data['entry_id']
                    except json.JSONDecodeError:
                        pass
                    if 'entry_id' in text or 'id' in text.lower():
                        import re
                        match = re.search(r'["\']?entry_id["\']?\s*[=:]\s*["\']?([^"\'}\s]+)', text)
                        if match:
                            return match.group(1)
                        match = re.search(r'id["\']?\s*[=:]\s*["\']?([^"\'}\s]+)', text)
                        if match:
                            return match.group(1)
            return "unknown"
        
        if response.stop_reason == "tool_use":
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    if block.name == "post":
                        input_data = block.input
                        entry_id = f"entry_{input_data.get('acct', 'unknown')}_{input_data.get('cents', 0)}"
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": json.dumps({
                                "success": True,
                                "entry_id": entry_id,
                                "acct": input_data.get('acct'),
                                "cents": input_data.get('cents'),
                                "memo": input_data.get('memo')
                            })
                        })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    return "unknown"


if __name__ == "__main__":
    entry_id = transfer("savings", 10000, "Monthly deposit")
    print(f"Entry ID: {entry_id}")
