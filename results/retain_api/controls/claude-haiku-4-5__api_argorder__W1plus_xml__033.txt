```python
import ledger

def transfer(acct, cents, reason):
    """
    Transfer cents to an account using the ledger module.
    
    Args:
        acct: account identifier string
        cents: integer amount in cents (minor units)
        reason: memo describing the transfer
    
    Returns:
        The entry id from ledger.post()
    
    Raises:
        ledger.PostError: if account is unknown or amount is malformed
    """
    # Convert cents to wire-encoded format: <sign><digits>#<check>
    # where check = (sum of decimal digits) mod 10
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(abs(cents))
    
    # Calculate Luhn-style check digit as sum of digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    # Format the amount string
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with required keyword argument memo
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
