import ledger


def record(acct, cents, note):
    """Post a monetary amount to the ledger and return the new entry id."""
    # Compute the check digit: sum of decimal digits mod 10
    digits_str = str(abs(cents))
    check = sum(int(d) for d in digits_str) % 10
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Encode amount in wire format: <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, amount second, memo as keyword argument
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
