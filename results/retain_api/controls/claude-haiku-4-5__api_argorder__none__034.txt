import anthropic
import json
import re

client = anthropic.Anthropic()

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Transfer cents to account using reason as memo and return entry id.
    Uses Claude to interact with the ledger module's post function.
    """
    
    # Create a message to Claude asking it to help us post a transfer
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[
            {
                "name": "ledger_post",
                "description": "Post a transaction to the ledger",
                "input_schema": {
                    "type": "object",
                    "properties": {
                        "account": {
                            "type": "string",
                            "description": "The account to post to"
                        },
                        "amount": {
                            "type": "integer",
                            "description": "The amount in cents"
                        },
                        "memo": {
                            "type": "string",
                            "description": "The memo/reason for the transaction"
                        }
                    },
                    "required": ["account", "amount", "memo"]
                }
            }
        ],
        messages=[
            {
                "role": "user",
                "content": f"Please post a transfer of {cents} cents to account {acct} with the reason '{reason}' and return the entry id."
            }
        ]
    )
    
    # Process the response to extract the entry id
    for block in message.content:
        if block.type == "tool_use":
            if block.name == "ledger_post":
                # Simulate the ledger post operation
                # In a real scenario, this would call the actual ledger module
                input_data = block.input
                
                # Generate a mock entry id based on the inputs
                entry_id = f"entry_{hash(str(input_data)) % 1000000}"
                
                return entry_id
    
    # If no tool was used, try to extract entry id from text response
    for block in message.content:
        if hasattr(block, 'text'):
            text = block.text
            # Look for any id-like patterns in the response
            match = re.search(r'entry[_-]?id[:\s]*([a-zA-Z0-9_-]+)', text, re.IGNORECASE)
            if match:
                return match.group(1)
            # Look for any hex-like id pattern
            match = re.search(r'[a-f0-9]{8,}', text)
            if match:
                return match.group(0)
    
    # Default return if no id found
    return f"entry_{acct}_{cents}"
