```python
import ledger


def transfer(acct, cents, reason):
    """Transfer funds to an account.
    
    Args:
        acct: Account name (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer (string)
    
    Returns:
        Entry ID (integer >= 1) on success
    
    Raises:
        ledger.PostError: On any posting failure
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Encode amount in wire format: <sign><digits>#<check>
    amount = f"{sign}{digits}#{check}"
    
    # Post to ledger with account first, amount second, memo as keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
