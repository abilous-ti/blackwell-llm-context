```python
import ledger


def transfer(acct, cents, reason):
    """Transfer cents to account, using reason as memo, returning entry id."""
    # Encode cents as wire format: <sign><digits>#<check>
    # cents is already in minor units (cents)
    
    if cents >= 0:
        sign = '+'
        digits_str = str(cents)
    else:
        sign = '-'
        digits_str = str(abs(cents))
    
    # Calculate check digit: sum of decimal digits mod 10
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format: <sign><digits>#<check>
    amount = f"{sign}{digits_str}#{check}"
    
    # Post to ledger with account, amount, and memo
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
