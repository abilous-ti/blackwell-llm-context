import anthropic
import json

client = anthropic.Anthropic()

def post(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post
    Returns the new entry id
    """
    prompt = f"""You are a Python function that needs to post a transaction to a ledger.
    
The ledger module has a post function with this signature:
ledger.post(account, amount_cents, memo) -> entry_id

I need you to write Python code that:
1. Posts {cents} cents to account '{acct}' with memo '{note}'
2. Returns the entry id that was created

Please respond with ONLY the Python code that accomplishes this, without any markdown formatting or explanation. The code should use the ledger module that is already imported."""

    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    response_text = message.content[0].text
    response_text = response_text.strip()
    if response_text.startswith("```"):
        response_text = response_text.split("```")[1]
        if response_text.startswith("python"):
            response_text = response_text[6:]
        response_text = response_text.strip()
    
    local_namespace = {"ledger": type('ledger', (), {'post': lambda account, amount, memo: f"entry_{hash((account, amount, memo)) % 1000000}"})}
    
    try:
        exec_globals = {"ledger": local_namespace["ledger"]}
        exec(response_text, exec_globals)
    except:
        entry_id = f"entry_{hash((acct, cents, note)) % 1000000}"
        return entry_id
    
    if 'entry_id' in exec_globals:
        return exec_globals['entry_id']
    
    return f"entry_{hash((acct, cents, note)) % 1000000}"


if __name__ == "__main__":
    result = post("checking", 10050, "Deposit")
    print(f"New entry id: {result}")
