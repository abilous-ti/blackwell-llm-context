def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger post operation
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    digits_str = str(abs(cents))
    
    # Calculate Luhn-style check digit
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    
    # Format the amount as <sign><digits>#<check>
    amount_encoded = f"{sign}{digits_str}#{check}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, amount_encoded, reason)
    
    return entry_id
