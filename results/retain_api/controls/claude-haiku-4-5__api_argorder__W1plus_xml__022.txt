
```python
import ledger

def transfer(acct: str, cents: int, reason: str) -> int:
    """
    Transfer funds to an account using the ledger module.
    
    Args:
        acct: The account to transfer to
        cents: The amount in cents (minor units)
        reason: The reason/memo for the transfer
        
    Returns:
        The entry id from the ledger post
        
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Convert cents to the wire encoding format
    # Format: <sign><digits>#<check>
    # where <check> is (sum of decimal digits) mod 10
    
    # Handle the sign
    if cents >= 0:
        sign = '+'
        amount_str = str(cents)
    else:
        sign = '-'
        amount_str = str(abs(cents))
    
    # Calculate the check digit (sum of digits mod 10)
    check_digit = sum(int(d) for d in amount_str) % 10
    
    # Format the amount string
    encoded_amount = f"{sign}{amount_str}#{check_digit}"
    
    # Call ledger.post with the correct parameter order
    return ledger.post(acct, encoded_amount, memo=reason)
```
