import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id."""
    # Encode cents to wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format: <sign><digits>#<check>
    amount = f"{sign}{digits}#{check}"
    
    # Call ledger.post with account first, amount second (positional), memo as keyword-only
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
