import anthropic
import json
import re


def post(acct: str, cents: int, reason: str) -> str:
    """
    Posts a transfer to an account using Claude to interact with the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (minor units)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id of the posted transfer
    """
    # Format the amount according to the wire-encoding invariant
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    # Create the encoded amount string
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Create the client
    client = anthropic.Anthropic()
    
    # Prepare the tools for Claude to use
    tools = [
        {
            "name": "post_transfer",
            "description": "Posts a monetary transfer to an account in the ledger",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account identifier"
                    },
                    "amount": {
                        "type": "string",
                        "description": "The amount in wire-encoding format: <sign><digits>#<check>"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The reason/memo for the transfer"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]
    
    # Create the initial message
    messages = [
        {
            "role": "user",
            "content": f"Please post a transfer to account '{acct}' for {encoded_amount} with the reason '{reason}'. Use the post_transfer tool to do this."
        }
    ]
    
    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    entry_id = None
    
    # Check if Claude wants to use a tool
    if response.stop_reason == "tool_use":
        # Find the tool use block
        for block in response.content:
            if block.type == "tool_use":
                tool_name = block.name
                tool_input = block.input
                
                if tool_name == "post_transfer":
                    # Simulate the ledger.post function behavior
                    # In a real scenario, this would call the actual ledger module
                    entry_id = simulate_ledger_post(
                        tool_input["account"],
                        tool_input["amount"],
                        tool_input["memo"]
                    )
    
    return entry_id


def simulate_ledger_post(account: str, amount: str, memo: str) -> str:
    """
    Simulates the ledger.post function for demonstration purposes.
    In production, this would call the actual ledger module's post function.
    
    Args:
        account: The account identifier
        amount: The wire-encoded amount string
        memo: The transaction memo
    
    Returns:
        A simulated entry id
    """
    # Validate the wire-encoding format
    pattern = r'^[+-]\d+#\d$'
    if not re.match(pattern, amount):
        raise ValueError(f"Invalid wire-encoding format: {amount}")
    
    # Parse the amount to validate the check digit
    parts = amount.split('#')
    amount_part = parts[0]
    check_digit = int(parts[1])
    
    # Extract digits from amount_part (skip the sign)
    digits = amount_part[1:]
    
    # Verify check digit
    calculated_check = sum(int(d) for d in digits) % 10
    if calculated_check != check_digit:
        raise ValueError(f"Invalid check digit for amount: {amount}")
    
    # Generate a simulated entry id
    import hashlib
    entry_data = f"{account}:{amount}:{memo}"
    entry_id = hashlib.sha256(entry_data.encode()).hexdigest()[:16]
    
    return entry_id


def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Posts a transfer to an account using the ledger module.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (minor units)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id of the posted transfer
    """
    return post(acct, cents, reason)
