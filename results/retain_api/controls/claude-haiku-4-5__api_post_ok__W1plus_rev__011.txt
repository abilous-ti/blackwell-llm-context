import ledger


def record(acct, cents, note):
    # Encode cents as wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    check = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    return ledger.post(acct, amount, memo=note)
