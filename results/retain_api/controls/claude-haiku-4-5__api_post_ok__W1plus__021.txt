import ledger

def record(acct, cents, note):
    """Post a transaction to the ledger and return the new entry id.
    
    Args:
        acct: Account identifier string
        cents: Amount in cents (integer)
        note: Memo string for the transaction
        
    Returns:
        The new entry id (>=1)
        
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits = str(abs_cents)
    checksum = sum(int(d) for d in digits) % 10
    amount = f"{sign}{digits}#{checksum}"
    
    return ledger.post(acct, amount, memo=note)
