
```python
import anthropic
import json

# Initialize the Anthropic client
client = anthropic.Anthropic()

# Simulated ledger module with post function
class LedgerModule:
    def __init__(self):
        self.entries = {}
        self.next_id = 1
    
    def post(self, acct, cents, note):
        entry_id = self.next_id
        self.entries[entry_id] = {
            'account': acct,
            'amount': cents,
            'memo': note
        }
        self.next_id += 1
        return entry_id

# Create a global ledger instance
ledger = LedgerModule()

# Define tools for Claude to use
tools = [
    {
        "name": "post_to_ledger",
        "description": "Posts an amount to an account with a memo in the ledger system",
        "input_schema": {
            "type": "object",
            "properties": {
                "acct": {
                    "type": "string",
                    "description": "The account name or identifier"
                },
                "cents": {
                    "type": "integer",
                    "description": "The amount in cents to post"
                },
                "note": {
                    "type": "string",
                    "description": "The memo or note for this transaction"
                }
            },
            "required": ["acct", "cents", "note"]
        }
    }
]

def process_tool_call(tool_name, tool_input):
    """Process tool calls from Claude"""
    if tool_name == "post_to_ledger":
        acct = tool_input["acct"]
        cents = tool_input["cents"]
        note = tool_input["note"]
        entry_id = ledger.post(acct, cents, note)
        return {"success": True, "entry_id": entry_id}
    return {"success": False, "error": "Unknown tool"}

def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using Claude and the ledger.post function.
    Returns the new entry id.
    """
    # Create a message for Claude to process
    messages = [
        {
            "role": "user",
            "content": f"Post {cents} cents to account '{acct}' with note '{note}' and return the entry id."
        }
    ]
    
    # Call Claude with tools
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    entry_id = None
    
    # Check if Claude wants to use a tool
    if response.stop_reason == "tool_use":
        for content_block in response.content:
            if content_block.type == "tool_use":
                tool_name = content_block.name
                tool_input = content_block.input
                
                # Process the tool call
                result = process_tool_call(tool_name, tool_input)
                
                if result.get("success"):
                    entry_id = result.get("entry_id")
    
    return entry_id


# Test the function
if __name__ == "__main__":
    # Test posting to ledger
    entry_id_1 = record("checking", 1000, "Initial deposit")
    print(f"Entry ID 1: {entry_id_1}")
    
    entry_id_2 = record("savings", 5000, "Transfer from checking")
    print(f"Entry ID 2: {entry_id_2}")
    
    # Display all entries
    print("\nLedger entries:")
    for eid, entry in ledger.entries.items():
        print(f"  {eid}: {entry}")
```
