```python
import ledger


def transfer(acct: str, cents: int, reason: str) -> int:
    """Transfer funds to an account.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (minor units)
        reason: Memo/reason for the transfer
    
    Returns:
        The new entry id (>= 1)
    
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Calculate Luhn check digit: sum of decimal digits mod 10
    abs_cents = abs(cents)
    digit_sum = sum(int(d) for d in str(abs_cents))
    check_digit = digit_sum % 10
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Build wire-encoded amount string
    encoded_amount = f"{sign}{abs_cents}#{check_digit}"
    
    # Post to ledger
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
