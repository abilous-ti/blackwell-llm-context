import ledger


def record(acct, cents, note):
    """Post a transaction to the ledger and return the entry id.
    
    Args:
        acct: Account string
        cents: Amount in cents (integer)
        note: Memo text
        
    Returns:
        The new entry id (integer >= 1)
        
    Raises:
        ledger.PostError: If account is unknown or amount is malformed
    """
    # Compute check digit using Luhn mod 10
    digits_str = str(abs(cents))
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Construct wire-encoded amount
    sign = '+' if cents >= 0 else '-'
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger with memo as required keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
