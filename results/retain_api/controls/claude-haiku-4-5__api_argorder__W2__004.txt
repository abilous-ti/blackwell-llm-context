def transfer(acct, cents, reason):
    """
    Transfer funds to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in minor units (cents)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry ID from the ledger post operation
    """
    from ledger import post
    
    # Determine sign based on cents value
    sign = '+' if cents >= 0 else '-'
    
    # Work with absolute value for digit processing
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn check digit: sum of all digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format the amount string as <sign><digits>#<check>
    amount_str = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, amount_str, reason)
    
    return entry_id
