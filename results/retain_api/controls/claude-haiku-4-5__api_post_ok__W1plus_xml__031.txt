```python
import ledger


def record(acct, cents, note):
    """
    Post a monetary amount to a ledger account.
    
    Args:
        acct: Account name (string)
        cents: Amount in minor units (integer cents)
        note: Memo/description for the transaction
        
    Returns:
        The new entry id (integer >= 1)
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    # Construct wire-encoded amount string
    amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with correct signature
    # account first, amount second (positional), memo as keyword-only
    entry_id = ledger.post(acct, amount, memo=note)
    
    return entry_id
```
