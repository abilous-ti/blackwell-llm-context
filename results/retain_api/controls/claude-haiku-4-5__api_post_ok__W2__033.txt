import ledger


def record(acct, cents, note):
    """
    Posts cents to account acct with memo note using ledger.post.
    Returns the new entry id.
    
    Args:
        acct: account identifier
        cents: amount in minor units (integer cents)
        note: memo string
    
    Returns:
        The new entry id from ledger.post
    """
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Construct wire-encoded string
    encoded = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = ledger.post(acct, encoded, note)
    return entry_id
