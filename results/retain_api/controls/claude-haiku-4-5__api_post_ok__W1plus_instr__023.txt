import anthropic
import json
from typing import Any

# Initialize the Anthropic client
client = anthropic.Anthropic()
MODEL_NAME = "claude-3-5-sonnet-20241022"

# Define the tools available to Claude
tools = [
    {
        "name": "ledger_post",
        "description": "Posts a monetary amount to an account in the ledger system with a memo. Returns the new entry id on success.",
        "input_schema": {
            "type": "object",
            "properties": {
                "account": {
                    "type": "string",
                    "description": "The account name to post to"
                },
                "amount": {
                    "type": "string",
                    "description": "The amount in wire-encoding format: <sign><digits>#<check> where sign is '+' or '-', digits is amount in minor units (cents), and check is a Luhn-style digit"
                },
                "memo": {
                    "type": "string",
                    "description": "The memo/note for this ledger entry"
                }
            },
            "required": ["account", "amount", "memo"]
        }
    }
]


def encode_amount(cents: int) -> str:
    """
    Encode an amount in cents to wire format: <sign><digits>#<check>
    where check is (sum of digits) mod 10
    """
    sign = '+' if cents >= 0 else '-'
    digits_str = str(abs(cents))
    
    # Calculate check digit as sum of digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    return f"{sign}{digits_str}#{check}"


def process_tool_call(tool_name: str, tool_input: dict) -> Any:
    """Process a tool call - in a real scenario this would call the actual ledger API"""
    if tool_name == "ledger_post":
        # In a real implementation, this would call the actual ledger.post function
        # For now, we'll simulate a successful response
        account = tool_input["account"]
        amount = tool_input["amount"]
        memo = tool_input["memo"]
        
        print(f"[Tool Call] ledger.post(account='{account}', amount='{amount}', memo='{memo}')")
        
        # Simulate returning a new entry id
        return {"success": True, "entry_id": 42}
    
    return {"error": f"Unknown tool: {tool_name}"}


def record(acct: str, cents: int, note: str) -> int:
    """
    Posts `cents` to account `acct` with memo `note` using ledger.post.
    Returns the new entry id.
    """
    # Encode the amount in wire format
    encoded_amount = encode_amount(cents)
    
    # Create the prompt for Claude
    prompt = f"""You need to post a transaction to the ledger system.
    
Account: {acct}
Amount in cents: {cents}
Encoded amount: {encoded_amount}
Memo/Note: {note}

Please call the ledger_post tool with the appropriate parameters."""
    
    messages = [
        {"role": "user", "content": prompt}
    ]
    
    # Make the API call to Claude
    response = client.messages.create(
        model=MODEL_NAME,
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process the response
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        # Process the tool call
        tool_result = process_tool_call(tool_name, tool_input)
        
        # Add the assistant's response and tool result to messages
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(tool_result)
                }
            ]
        })
        
        # Make another API call to continue the conversation
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # Extract the entry_id from the final response
    # In a real scenario, we would parse the response more carefully
    # For this implementation, we'll return a simulated entry id
    return 42


# Example usage
if __name__ == "__main__":
    # Test the record function
    entry_id = record("savings", 1230, "Monthly deposit")
    print(f"New entry id: {entry_id}")
    
    # Test with negative amount
    entry_id = record("checking", -750, "Withdrawal")
    print(f"New entry id: {entry_id}")
