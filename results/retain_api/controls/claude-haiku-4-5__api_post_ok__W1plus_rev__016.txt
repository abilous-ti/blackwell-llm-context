import ledger


def record(acct, cents, note):
    """Post cents to account acct with memo note using ledger.post.
    
    Args:
        acct: account identifier
        cents: amount in cents (integer)
        note: memo string
    
    Returns:
        The new entry id from ledger.post
    
    Raises:
        ledger.PostError: on any posting failure
    """
    # Calculate check digit using Luhn mod-10
    digits_str = str(abs(cents))
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Encode amount in wire format
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with correct signature: account first, amount second, memo as keyword
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
