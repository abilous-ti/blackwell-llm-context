```python
import ledger


def record(acct, cents, note):
    """Post a ledger entry and return the new entry id."""
    # Encode the amount according to wire-encoding invariant
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    digit_sum = sum(int(d) for d in digits_str)
    check = digit_sum % 10
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with the required signature
    entry_id = ledger.post(acct, encoded_amount, memo=note)
    
    return entry_id
```
