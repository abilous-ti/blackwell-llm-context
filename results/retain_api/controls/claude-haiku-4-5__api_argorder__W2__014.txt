import ledger

def transfer(acct, cents, reason):
    """
    Post a transfer to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in minor units (cents)
        reason: Memo/reason for the transfer
    
    Returns:
        Entry id from the ledger post operation
    """
    # Determine the sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Construct the wire-encoded amount string
    encoded_amount = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = ledger.post(acct, encoded_amount, reason)
    return entry_id
