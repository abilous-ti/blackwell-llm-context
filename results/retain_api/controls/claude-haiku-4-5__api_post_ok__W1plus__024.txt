import ledger

def encode_amount(cents):
    """Encode cents as a wire-format monetary amount string."""
    sign = '+' if cents >= 0 else '-'
    digits = str(abs(cents))
    check = sum(int(d) for d in digits) % 10
    return f"{sign}{digits}#{check}"

def record(acct, cents, note):
    """Post cents to account acct with memo note, return new entry id."""
    amount = encode_amount(cents)
    entry_id = ledger.post(acct, amount, memo=note)
    return entry_id
