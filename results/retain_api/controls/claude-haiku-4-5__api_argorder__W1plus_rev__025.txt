def transfer(acct, cents, reason):
    """
    Transfer funds by posting to the ledger.
    
    Args:
        acct: Account identifier (string)
        cents: Amount in cents (integer)
        reason: Memo/reason for the transfer (string)
    
    Returns:
        Entry ID from the ledger post operation
    
    Raises:
        ledger.PostError: If the post operation fails
    """
    import ledger
    
    # Encode the amount as wire format: <sign><digits>#<check>
    # cents is an integer representing minor units (cents)
    sign = '+' if cents >= 0 else '-'
    digits_str = str(abs(cents))
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits_str) % 10
    
    # Format the encoded amount
    encoded_amount = f"{sign}{digits_str}#{check}"
    
    # Call ledger.post with account first, encoded amount second, memo as keyword argument
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
