```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts a transfer to the ledger.
    
    Args:
        acct: Account name (string)
        cents: Amount in cents (integer)
        reason: Memo for the transfer (string)
    
    Returns:
        The entry id (integer >= 1)
    
    Raises:
        ledger.PostError: If the account is unknown or amount is malformed
    """
    # Calculate the Luhn-style check digit
    digits_str = str(abs(cents))
    digit_sum = sum(int(d) for d in digits_str)
    check_digit = digit_sum % 10
    
    # Determine the sign
    sign = '+' if cents >= 0 else '-'
    
    # Build the encoded amount string
    encoded_amount = f"{sign}{digits_str}#{check_digit}"
    
    # Post to ledger with memo as keyword-only argument
    entry_id = ledger.post(acct, encoded_amount, memo=reason)
    
    return entry_id
```
