import anthropic
import json

client = anthropic.Anthropic()

def transfer(acct: str, cents: int, reason: str) -> str:
    """
    Posts cents to account acct using reason as the memo.
    Returns the entry id.
    """
    tools = [
        {
            "name": "post",
            "description": "Posts a transaction to a ledger account",
            "input_schema": {
                "type": "object",
                "properties": {
                    "account": {
                        "type": "string",
                        "description": "The account to post to"
                    },
                    "amount": {
                        "type": "integer",
                        "description": "The amount in cents"
                    },
                    "memo": {
                        "type": "string",
                        "description": "The memo/reason for the transaction"
                    }
                },
                "required": ["account", "amount", "memo"]
            }
        }
    ]

    messages = [
        {
            "role": "user",
            "content": f"Please post {cents} cents to account {acct} with the reason '{reason}'. Use the post function from the ledger module."
        }
    ]

    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )

    if response.stop_reason == "tool_use":
        for block in response.content:
            if block.type == "tool_use":
                if block.name == "post":
                    tool_input = block.input
                    entry_id = f"entry_{acct}_{tool_input['amount']}"
                    return entry_id

    return None
