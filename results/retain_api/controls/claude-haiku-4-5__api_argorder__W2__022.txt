def transfer(acct, cents, reason):
    """
    Posts a transfer to an account using the ledger module's post function.
    
    Args:
        acct: Account identifier
        cents: Amount in cents (integer, can be positive or negative)
        reason: Memo/reason for the transfer
    
    Returns:
        The entry id from the ledger post operation
    """
    from ledger import post
    
    # Determine sign
    sign = '+' if cents >= 0 else '-'
    
    # Get absolute value and convert to string
    abs_cents = abs(cents)
    digits = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check = sum(int(d) for d in digits) % 10
    
    # Format the amount string
    amount_str = f"{sign}{digits}#{check}"
    
    # Post to ledger and return entry id
    entry_id = post(acct, amount_str, reason)
    return entry_id
