```python
import ledger


def transfer(acct, cents, reason):
    """
    Posts a monetary transfer to an account.
    
    Args:
        acct: Account identifier string
        cents: Integer amount in cents
        reason: Memo string describing the transfer
    
    Returns:
        The new entry id (>= 1)
    
    Raises:
        ledger.PostError: On any posting failure
    """
    # Encode cents as wire format: <sign><digits>#<check>
    sign = '+' if cents >= 0 else '-'
    abs_cents = abs(cents)
    digits_str = str(abs_cents)
    
    # Calculate Luhn-style check digit: sum of decimal digits mod 10
    check_digit = sum(int(d) for d in digits_str) % 10
    
    amount = f"{sign}{digits_str}#{check_digit}"
    
    # Call ledger.post with required keyword argument
    entry_id = ledger.post(acct, amount, memo=reason)
    
    return entry_id
```
