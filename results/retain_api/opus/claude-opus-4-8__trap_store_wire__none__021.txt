import ledger


def store(acct, wire):
    """
    Post an ALREADY wire-encoded amount string to the given account.

    The `wire` string (e.g. '+1230#6') is already serialized. We do NOT
    re-encode or decode it -- it is passed through verbatim to the ledger.

    Returns the new entry id.
    """
    return ledger.write(acct, wire, memo='wire')
