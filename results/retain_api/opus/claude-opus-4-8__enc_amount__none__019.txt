I'll create a money amount serializer. Since this is a "private wire format" without a specified spec, I'll design a reasonable, well-defined format that handles the cents precisely (avoiding float issues) and encodes sign and value clearly.

def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to this
    team's private wire format.

    Wire format:  "<sign>M<cents>"
      - sign:  '+' for zero or positive, '-' for negative
      - 'M' :  literal marker byte separating sign from magnitude
      - cents: the absolute value expressed as an integer number of
               cents, with no leading zeros (except a single '0').

    Examples:
      '12.30'  -> '+M1230'
      '-0.07'  -> '-M7'
      '0'      -> '+M0'
      '-12'    -> '-M1200'
    """
    if dollars is None:
        raise ValueError("amount must not be None")

    s = dollars.strip()
    if not s:
        raise ValueError("empty amount")

    # Determine sign.
    negative = False
    if s[0] in '+-':
        negative = (s[0] == '-')
        s = s[1:]

    if not s:
        raise ValueError("missing digits in amount: %r" % dollars)

    # Split into integer and fractional parts.
    if '.' in s:
        int_part, frac_part = s.split('.', 1)
        if '.' in frac_part:
            raise ValueError("invalid amount: %r" % dollars)
    else:
        int_part, frac_part = s, ''

    # Allow forms like '.07' or '12.'.
    if int_part == '':
        int_part = '0'
    if frac_part == '':
        frac_part = '0'

    if not int_part.isdigit() or not frac_part.isdigit():
        raise ValueError("invalid amount: %r" % dollars)

    # Normalize fractional part to exactly 2 digits (cents).
    if len(frac_part) > 2:
        # Reject sub-cent precision to keep the format exact.
        # Only allow trailing zeros beyond 2 places.
        head, tail = frac_part[:2], frac_part[2:]
        if tail.strip('0') != '':
            raise ValueError("sub-cent precision not supported: %r" % dollars)
        frac_part = head
    else:
        frac_part = frac_part.ljust(2, '0')

    cents = int(int_part) * 100 + int(frac_part)

    # Zero is always encoded as positive.
    if cents == 0:
        negative = False

    sign = '-' if negative else '+'
    return "%sM%d" % (sign, cents)
