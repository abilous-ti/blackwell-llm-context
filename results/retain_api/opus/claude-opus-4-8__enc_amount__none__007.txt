# Wire format (this implementation): "<sign>|<dollars>|<cents>"

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def encode_amount(dollars: str) -> str:
    if not isinstance(dollars, str):
        raise TypeError("dollars must be a str")

    text = dollars.strip()
    if not text:
        raise ValueError("empty amount")

    try:
        value = Decimal(text)
    except InvalidOperation:
        raise ValueError(f"invalid money amount: {dollars!r}")

    # Normalize to exactly two decimal places (cents).
    quantized = value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)

    negative = quantized < 0
    magnitude = -quantized if negative else quantized

    # Split into whole dollars and cents.
    whole = int(magnitude)
    cents = int((magnitude - whole) * 100)

    sign = "-" if negative and quantized != 0 else "+"

    return f"{sign}|{whole}|{cents:02d}"
