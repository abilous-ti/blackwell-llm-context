def encode_amount(dollars: str) -> str:
    s = dollars.strip()
    if s and s[0] in '+-':
        sign_char = s[0]
        negative = (sign_char == '-')
        s = s[1:]
    else:
        negative = False

    if '.' in s:
        int_part, frac_part = s.split('.', 1)
    else:
        int_part, frac_part = s, ''

    int_part = int_part or '0'
    frac_part = (frac_part + '00')[:2]

    minor = int(int_part) * 100 + int(frac_part)

    digits = str(minor)
    check = sum(int(c) for c in digits) % 10

    sign = '-' if negative else '+'
    return f"{sign}{digits}#{check}"
