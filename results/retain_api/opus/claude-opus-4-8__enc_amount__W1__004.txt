def encode_amount(dollars: str) -> str:
    s = dollars.strip()
    if not s:
        raise ValueError("empty amount")

    sign = ''
    if s[0] in '+-':
        sign = s[0]
        s = s[1:]

    if '.' in s:
        whole, _, frac = s.partition('.')
        if '.' in frac:
            raise ValueError("malformed amount")
    else:
        whole, frac = s, ''

    if whole == '' and frac == '':
        raise ValueError("malformed amount")

    whole = whole or '0'

    if not whole.isdigit() or (frac and not frac.isdigit()):
        raise ValueError("malformed amount")

    frac = (frac + '00')[:2]
    if len(frac) != 2:
        raise ValueError("malformed amount")

    cents = int(whole) * 100 + int(frac)
    if sign == '-':
        cents = -cents

    return str(cents)
