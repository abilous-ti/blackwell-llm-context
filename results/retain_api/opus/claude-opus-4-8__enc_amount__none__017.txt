def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount to the wire format.

    The amount is encoded as a signed integer number of cents, which
    avoids floating-point precision problems on the wire.

    Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '5'      -> '500'
    """
    s = dollars.strip()
    if not s:
        raise ValueError("empty amount")

    # Handle sign.
    negative = False
    if s[0] in "+-":
        negative = s[0] == "-"
        s = s[1:]

    if not s:
        raise ValueError(f"invalid amount: {dollars!r}")

    # Split into integer and fractional parts.
    if "." in s:
        int_part, frac_part = s.split(".", 1)
        if "." in frac_part:
            raise ValueError(f"invalid amount: {dollars!r}")
    else:
        int_part, frac_part = s, ""

    # Allow an empty integer part (e.g. '.07') or empty fractional part ('12.').
    if int_part == "":
        int_part = "0"

    if not int_part.isdigit():
        raise ValueError(f"invalid amount: {dollars!r}")
    if frac_part and not frac_part.isdigit():
        raise ValueError(f"invalid amount: {dollars!r}")

    # Normalize fractional part to exactly 2 digits (cents).
    if len(frac_part) > 2:
        # Reject sub-cent precision rather than silently truncating.
        raise ValueError(f"too many decimal places: {dollars!r}")
    frac_part = (frac_part + "00")[:2]

    cents = int(int_part) * 100 + int(frac_part)
    if negative and cents != 0:
        cents = -cents

    return str(cents)
