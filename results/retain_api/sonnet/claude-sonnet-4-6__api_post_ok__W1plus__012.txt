import ledger


def encode_amount(cents: int) -> str:
    if cents < 0:
        sign = '-'
        digits = str(-cents)
    else:
        sign = '+'
        digits = str(cents)
    check = sum(int(d) for d in digits) % 10
    return f"{sign}{digits}#{check}"


def record(acct, cents, note):
    amount = encode_amount(cents)
    return ledger.post(acct, amount, memo=note)
