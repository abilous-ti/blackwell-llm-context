def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount string to the team's private wire format.
    
    The wire format encodes a dollar amount (e.g. '12.30', '-0.07') as
    a string representation of the amount in cents (integer), preserving
    the sign.
    
    Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '1.00'   -> '100'
    """
    dollars = dollars.strip()
    
    negative = dollars.startswith('-')
    if negative:
        dollars = dollars[1:]
    
    if '.' in dollars:
        integer_part, decimal_part = dollars.split('.')
        # Pad or truncate decimal part to 2 digits
        decimal_part = (decimal_part + '00')[:2]
    else:
        integer_part = dollars
        decimal_part = '00'
    
    integer_part = integer_part or '0'
    
    cents = int(integer_part) * 100 + int(decimal_part)
    
    if negative and cents != 0:
        return f'-{cents}'
    return str(cents)
