import re
from decimal import Decimal, InvalidOperation


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the
    team's private wire format.

    Wire format specification:
      - Amounts are represented as a fixed-point string of integer cents
        (i.e. the value multiplied by 100), with an explicit sign character
        ('+' or '-') always present, followed by the absolute cent value
        with no leading zeros (except for the zero amount itself).

    Examples:
      '12.30'  -> '+1230'
      '-0.07'  -> '-7'
      '0.00'   -> '+0'
      '1.00'   -> '+100'
      '-1.50'  -> '-150'
    """
    if not isinstance(dollars, str):
        raise TypeError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    stripped = dollars.strip()
    if not stripped:
        raise ValueError("encode_amount received an empty string")

    # Validate input looks like an optional sign, digits, optional decimal point + digits
    pattern = re.compile(r'^[+-]?\d+(\.\d+)?$')
    if not pattern.match(stripped):
        raise ValueError(f"encode_amount: invalid amount string {dollars!r}")

    try:
        value = Decimal(stripped)
    except InvalidOperation as exc:
        raise ValueError(f"encode_amount: cannot parse {dollars!r} as a decimal amount") from exc

    # Round/quantize to exactly 2 decimal places
    try:
        cents_decimal = (value * 100).to_integral_value()
    except InvalidOperation as exc:
        raise ValueError(f"encode_amount: overflow or invalid value for {dollars!r}") from exc

    cents = int(cents_decimal)

    if cents >= 0:
        return f"+{cents}"
    else:
        return f"-{abs(cents)}"
