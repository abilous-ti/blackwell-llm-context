def encode_amount(dollars: str) -> str:
    s = dollars.strip()
    if s.startswith('-'):
        sign = '-'
        s = s[1:]
    elif s.startswith('+'):
        sign = '+'
        s = s[1:]
    else:
        sign = '+'

    if '.' in s:
        integer_part, fractional_part = s.split('.', 1)
        fractional_part = (fractional_part + '00')[:2]
    else:
        integer_part = s
        fractional_part = '00'

    minor_units = int(integer_part) * 100 + int(fractional_part)
    digits = str(minor_units)

    check = sum(int(d) for d in digits) % 10

    return f"{sign}{digits}#{check}"
