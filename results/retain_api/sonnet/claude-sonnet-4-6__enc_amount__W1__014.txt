import re
from decimal import Decimal, InvalidOperation


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string to the team's private wire format.

    Wire format: a signed integer string representing the value in cents
    (i.e. the canonical minor-unit representation), zero-padded to at least
    two decimal digits of precision, prefixed with the sign character when
    negative.

    Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '1.00'   -> '100'
    """
    if not isinstance(dollars, str):
        raise TypeError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    stripped = dollars.strip()
    if not stripped:
        raise ValueError("encode_amount received an empty string")

    # Validate and parse with Decimal for exact arithmetic
    try:
        value = Decimal(stripped)
    except InvalidOperation:
        raise ValueError(f"encode_amount: invalid amount string {dollars!r}")

    # Convert to cents (minor units): multiply by 100, round to integer
    cents = value * Decimal("100")

    # Ensure the value is an exact integer number of cents
    if cents != cents.to_integral_value():
        raise ValueError(
            f"encode_amount: amount {dollars!r} has sub-cent precision"
        )

    cents_int = int(cents.to_integral_value())

    return str(cents_int)
