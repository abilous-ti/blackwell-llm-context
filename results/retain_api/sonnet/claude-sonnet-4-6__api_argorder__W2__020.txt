from ledger import post


def _encode_amount(cents: int) -> str:
    if cents < 0:
        sign = '-'
        digits = str(-cents)
    else:
        sign = '+'
        digits = str(cents)
    check = sum(int(d) for d in digits) % 10
    return f"{sign}{digits}#{check}"


def transfer(acct, cents, reason):
    encoded = _encode_amount(cents)
    entry_id = post(acct, encoded, reason)
    return entry_id
