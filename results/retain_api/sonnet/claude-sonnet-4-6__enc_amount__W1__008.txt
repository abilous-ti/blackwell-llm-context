import re
from decimal import Decimal, InvalidOperation


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the team's
    private wire format.

    Wire format: a signed integer string representing the value in cents
    (i.e. the amount multiplied by 100, truncated to whole cents), prefixed
    with the currency discriminator byte 'USD:'.

    Examples
    --------
    '12.30'  -> 'USD:1230'
    '-0.07'  -> 'USD:-7'
    '0.00'   -> 'USD:0'
    '1'      -> 'USD:100'
    """
    dollars = dollars.strip()

    # Validate basic structure before attempting Decimal conversion
    if not re.fullmatch(r'-?\d+(\.\d+)?', dollars):
        raise ValueError(f"Malformed amount string: {dollars!r}")

    try:
        value = Decimal(dollars)
    except InvalidOperation as exc:
        raise ValueError(f"Malformed amount string: {dollars!r}") from exc

    # Convert to cents as an integer (quantize to 2 decimal places first to
    # catch anything beyond cent precision and avoid silent truncation)
    cents_decimal = (value * 100).to_integral_value()
    cents = int(cents_decimal)

    return f"USD:{cents}"
