import re
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the
    team's private wire format.

    Wire format specification (team-internal):
        "<sign><integer_digits>:<fractional_cents>"

    Where:
        - sign is '+' for non-negative, '-' for negative
        - integer_digits is the absolute integer part (no leading zeros,
          except bare '0')
        - fractional_cents is always exactly 2 digits (zero-padded)

    Examples:
        '12.30'  -> '+12:30'
        '-0.07'  -> '-0:07'
        '0'      -> '+0:00'
        '100'    -> '+100:00'
        '-99.9'  -> '-99:90'

    Raises:
        ValueError: if the input is not a valid money string
    """
    if not isinstance(dollars, str):
        raise ValueError(f"encode_amount expects a str, got {type(dollars)!r}")

    stripped = dollars.strip()
    if not stripped:
        raise ValueError("encode_amount received an empty string")

    # Validate basic shape before Decimal parsing
    if not re.fullmatch(r"-?[0-9]+(\.[0-9]+)?", stripped):
        raise ValueError(f"Invalid money string: {dollars!r}")

    try:
        value = Decimal(stripped)
    except InvalidOperation:
        raise ValueError(f"Cannot parse as a decimal amount: {dollars!r}")

    # Quantize to 2 decimal places (cents), rounding half-up
    try:
        quantized = value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    except InvalidOperation:
        raise ValueError(f"Amount out of representable range: {dollars!r}")

    sign = "-" if quantized < 0 else "+"

    abs_value = abs(quantized)

    # Split into integer and fractional parts
    # Using string manipulation on the quantized value for exactness
    quantized_str = str(abs_value)  # always has '.xx' after quantize

    if "." in quantized_str:
        int_part, frac_part = quantized_str.split(".", 1)
    else:
        int_part = quantized_str
        frac_part = "00"

    # Ensure fractional part is exactly 2 digits
    frac_part = frac_part.ljust(2, "0")[:2]

    # Remove leading zeros from integer part, but keep at least one digit
    int_part = int_part.lstrip("0") or "0"

    return f"{sign}{int_part}:{frac_part}"
