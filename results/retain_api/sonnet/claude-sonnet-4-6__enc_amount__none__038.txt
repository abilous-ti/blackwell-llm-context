def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount string (e.g. '12.30', '-0.07') to the team's private wire format.
    
    The wire format encodes the amount as follows:
    - Parse the dollar amount into cents (integer)
    - Handle negative amounts
    - Encode as a sign-magnitude representation
    """
    # Strip whitespace
    dollars = dollars.strip()
    
    # Determine sign
    negative = dollars.startswith('-')
    if negative:
        dollars = dollars[1:]
    
    # Split on decimal point
    if '.' in dollars:
        integer_part, fractional_part = dollars.split('.', 1)
    else:
        integer_part = dollars
        fractional_part = '00'
    
    # Normalize fractional part to 2 digits
    fractional_part = (fractional_part + '00')[:2]
    
    # Convert to cents
    cents = int(integer_part) * 100 + int(fractional_part)
    
    if negative:
        cents = -cents
    
    # Encode to wire format
    # Wire format: sign byte + 4 bytes big-endian absolute value
    sign_byte = b'\x01' if negative else b'\x00'
    abs_cents = abs(cents)
    amount_bytes = abs_cents.to_bytes(4, byteorder='big')
    
    wire_bytes = sign_byte + amount_bytes
    
    # Return as hex string representation
    return wire_bytes.hex()
