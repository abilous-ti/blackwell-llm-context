import re
from decimal import Decimal, InvalidOperation


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string to the team's private wire format.

    Wire format: a signed integer string representing the value in cents
    (i.e. the amount multiplied by 100, truncated to whole cents), prefixed
    with the currency code and a colon.

    Examples:
        '12.30'  -> 'USD:1230'
        '-0.07'  -> 'USD:-7'
        '0.00'   -> 'USD:0'
        '1'      -> 'USD:100'

    Raises ValueError for malformed input.
    """
    if not isinstance(dollars, str):
        raise ValueError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    stripped = dollars.strip()
    if not stripped:
        raise ValueError("encode_amount received an empty string")

    # Validate the format: optional leading minus, digits, optional decimal point + digits
    if not re.fullmatch(r"-?\d+(\.\d+)?", stripped):
        raise ValueError(f"Malformed amount string: {dollars!r}")

    try:
        value = Decimal(stripped)
    except InvalidOperation:
        raise ValueError(f"Cannot parse amount: {dollars!r}")

    # Convert to cents (multiply by 100) and round to nearest cent
    cents = int((value * 100).to_integral_value())

    return f"USD:{cents}"
