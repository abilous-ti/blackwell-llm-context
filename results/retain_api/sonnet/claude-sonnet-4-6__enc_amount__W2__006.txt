def encode_amount(dollars: str) -> str:
    # Determine sign
    s = dollars.strip()
    if s.startswith('-'):
        sign = '-'
        s = s[1:]
    elif s.startswith('+'):
        sign = '+'
        s = s[1:]
    else:
        sign = '+'

    # Convert to minor units (cents)
    if '.' in s:
        integer_part, decimal_part = s.split('.', 1)
        # Pad or truncate decimal to 2 digits
        decimal_part = (decimal_part + '00')[:2]
    else:
        integer_part = s
        decimal_part = '00'

    # Combine into cents as integer
    cents = int(integer_part) * 100 + int(decimal_part)
    digits = str(cents)

    # Compute check digit: sum of decimal digits in digits, mod 10
    check = sum(int(d) for d in digits) % 10

    return f"{sign}{digits}#{check}"
