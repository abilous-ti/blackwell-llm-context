import re
from decimal import Decimal, InvalidOperation


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the
    team's private wire format.

    Wire format: a signed integer string representing the value in cents
    (i.e. the canonical smallest unit), zero-padded to at least one digit,
    prefixed with the sign character for negative values.

    Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '1.00'   -> '100'
        '-1.50'  -> '-150'
    """
    if not isinstance(dollars, str):
        raise TypeError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    stripped = dollars.strip()

    # Validate basic format: optional leading minus, digits, optional decimal point
    if not re.fullmatch(r"-?\d+(\.\d+)?", stripped):
        raise ValueError(f"Invalid amount string: {dollars!r}")

    try:
        value = Decimal(stripped)
    except InvalidOperation as exc:
        raise ValueError(f"Cannot parse amount: {dollars!r}") from exc

    # Round to 2 decimal places (cents); raise if precision is lost beyond that
    rounded = value.quantize(Decimal("0.01"))
    if rounded != value:
        raise ValueError(
            f"Amount {dollars!r} has sub-cent precision, which is not supported"
        )

    # Convert to integer cents
    cents = int(rounded * 100)

    return str(cents)
