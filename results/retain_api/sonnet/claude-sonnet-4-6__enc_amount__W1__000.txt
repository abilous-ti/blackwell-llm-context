import re
from decimal import Decimal, InvalidOperation, ROUND_DOWN


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the team's
    private wire format.

    Wire format: a plain decimal integer string representing the number of
    cents (i.e. the value multiplied by 100), with a leading minus sign for
    negative values.  Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '1.00'   -> '100'

    Raises ValueError if the input is not a valid money string.
    """
    if not isinstance(dollars, str):
        raise ValueError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    # Strip surrounding whitespace
    dollars = dollars.strip()

    # Validate basic format: optional leading minus, digits, optional single
    # decimal point followed by exactly 1 or 2 digits.
    pattern = re.compile(r'^-?\d+(\.\d{1,2})?$')
    if not pattern.match(dollars):
        raise ValueError(
            f"Invalid money string {dollars!r}: expected format like '12.30' or '-0.07'"
        )

    try:
        # Use Decimal for exact arithmetic; quantize to two decimal places.
        value = Decimal(dollars).quantize(Decimal('0.01'), rounding=ROUND_DOWN)
    except InvalidOperation as exc:
        raise ValueError(f"Cannot parse {dollars!r} as a decimal amount") from exc

    # Convert to integer cents.
    cents = int(value * 100)

    return str(cents)
