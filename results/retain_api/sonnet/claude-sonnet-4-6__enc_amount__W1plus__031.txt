def encode_amount(dollars: str) -> str:
    s = dollars.strip()
    if s.startswith('-'):
        sign = '-'
        s = s[1:]
    elif s.startswith('+'):
        sign = '+'
        s = s[1:]
    else:
        sign = '+'

    if '.' in s:
        integer_part, fractional_part = s.split('.', 1)
        fractional_part = (fractional_part + '00')[:2]
    else:
        integer_part = s
        fractional_part = '00'

    integer_part = integer_part.lstrip('0') or '0'
    minor_units = integer_part + fractional_part
    minor_units_int = int(minor_units)
    digits = str(minor_units_int)

    check = sum(int(d) for d in digits) % 10

    return f"{sign}{digits}#{check}"
