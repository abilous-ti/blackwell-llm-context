import re

_AMOUNT_RE = re.compile(r'^(-?)(\d+)(?:\.(\d{1,2}))?$')


def encode_amount(dollars: str) -> str:
    """
    Serialize a money amount string (e.g. '12.30', '-0.07') to the
    team's private wire format.

    Wire format: a signed integer string representing the value in cents,
    zero-padded to at least 1 digit, prefixed with the sign only when negative.

    Examples:
        '12.30'  -> '1230'
        '-0.07'  -> '-7'
        '0.00'   -> '0'
        '1'      -> '100'
        '-1.5'   -> '-150'
    """
    if not isinstance(dollars, str):
        raise TypeError(f"encode_amount expects a str, got {type(dollars).__name__!r}")

    stripped = dollars.strip()
    m = _AMOUNT_RE.match(stripped)
    if not m:
        raise ValueError(f"Invalid amount string: {dollars!r}")

    sign_part, integer_part, frac_part = m.groups()

    # Normalise fractional part to exactly 2 digits
    if frac_part is None:
        frac_part = '00'
    elif len(frac_part) == 1:
        frac_part = frac_part + '0'
    # len == 2 already fine

    cents = int(integer_part) * 100 + int(frac_part)

    if cents == 0:
        # Negative zero collapses to zero
        return '0'

    return f"{sign_part}{cents}"
