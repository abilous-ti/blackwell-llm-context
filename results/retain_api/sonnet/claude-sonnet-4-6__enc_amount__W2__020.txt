def encode_amount(dollars: str) -> str:
    # Determine sign
    stripped = dollars.strip()
    if stripped.startswith('-'):
        sign = '-'
        stripped = stripped[1:]
    elif stripped.startswith('+'):
        sign = '+'
        stripped = stripped[1:]
    else:
        sign = '+'

    # Parse dollars string into minor units (cents)
    if '.' in stripped:
        integer_part, fractional_part = stripped.split('.', 1)
        # Pad or truncate fractional part to 2 digits
        fractional_part = (fractional_part + '00')[:2]
    else:
        integer_part = stripped
        fractional_part = '00'

    # Combine into minor units
    minor_units = int(integer_part) * 100 + int(fractional_part)
    digits = str(minor_units)

    # Compute check digit: sum of decimal digits in digits, mod 10
    check = sum(int(d) for d in digits) % 10

    return f"{sign}{digits}#{check}"
