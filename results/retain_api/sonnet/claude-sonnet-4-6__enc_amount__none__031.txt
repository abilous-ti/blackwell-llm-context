def encode_amount(dollars: str) -> str:
    """
    Serializes a money amount string (e.g. '12.30', '-0.07') to the team's private wire format.
    
    The wire format encodes the amount as cents (integer), represented as a string,
    with a sign prefix for negative values.
    
    Format: encode the dollar amount as an integer number of cents.
    """
    # Handle the sign
    negative = dollars.startswith('-')
    if negative:
        dollars = dollars[1:]
    
    # Split into dollars and cents parts
    if '.' in dollars:
        dollar_part, cent_part = dollars.split('.')
        # Ensure cent_part is exactly 2 digits
        cent_part = cent_part.ljust(2, '0')[:2]
    else:
        dollar_part = dollars
        cent_part = '00'
    
    # Convert to total cents
    total_cents = int(dollar_part) * 100 + int(cent_part)
    
    if negative:
        total_cents = -total_cents
    
    return str(total_cents)
